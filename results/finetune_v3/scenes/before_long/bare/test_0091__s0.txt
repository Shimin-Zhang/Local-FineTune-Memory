import bpy

# ---------------- MATERIALS ----------------
def make_mat(name, color, emission=0.0, rough=0.7, metal=0.0):
    m = bpy.data.materials.new(name)
    m.use_nodes = True
    bsdf = m.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = (*color, 1)
    bsdf.inputs["Roughness"].default_value = rough
    bsdf.inputs["Metallic"].default_value = metal
    if emission > 0:
        bsdf.inputs["Emission Color"].default_value = (1.0, 0.85, 0.55, 1)
        bsdf.inputs["Emission Strength"].default_value = emission
    return m

mat_ground   = make_mat("Ground",   (0.18, 0.14, 0.10), rough=0.95)
mat_brick    = make_mat("Brick",    (0.45, 0.18, 0.12), rough=0.85)
mat_brick2   = make_mat("Brick2",   (0.52, 0.22, 0.14), rough=0.85)
mat_mortar   = make_mat("Mortar",   (0.60, 0.58, 0.55), rough=0.95)

mat_steel    = make_mat("Steel",    (0.35, 0.35, 0.38), rough=0.30, metal=0.85)

# ---------------- HELPERS ----------------
def box(name, loc, size, mat, rot=(0,0,0)):
    bpy.ops.mesh.primitive_cube_add(size=1, location=loc)
    o = bpy.context.active_object
    o.name = name
    o.scale = (size[0]/2, size[1]/2, size[2]/2)
    o.rotation_euler = Euler(rot)
    o.data.materials.append(mat)
    bpy.ops.object.transform_apply(scale=True)
    return o

def cyl(name, loc, r, depth, mat, rot=(0,0,0), verts=16):
    bpy.ops.mesh.primitive_cylinder_add(radius=r, depth=depth, location=loc, vertices=verts)
    o = bpy.context.active_object
    o.name = name
    o.rotation_euler = Euler(rot)
    o.data.materials.append(mat)
    return o

# 3 floors, front 2 floors complete, back 1 floor, some walls missing
def brick_wall(name, loc, w, h, d, mat, rot=(0,0,0)):
    o = box(name, loc, (w, h, d), mat, rot)
    return o

bx, by = -8, -6   # block center
bw, bd = 12, 10   # width, depth
floor_h = 3.0
floors  = 3

# Floor slabs (concrete)
for f in range(floors + 1):
    z = f * floor_h
    if f == floors:  # top slab only partial
        box(f"Slab{f}", (bx, by, z), (bw, bd*0.6, 0.25), mat_concrete)

# Walls – front (y = by - bd/2), back (y = by + bd/2), left (x = bx - bw/2), right (x = bx + bw/2)
wall_t = 0.35
for f in range(floors):
    z0 = f * floor_h + 0.125
    zc = z0 + floor_h / 2
    # Front wall – all floors
    brick_wall(f"Front{f}", (bx, by - bd/2 + wall_t/2, zc), bw, floor_h, wall_t, mat_brick)
    # Back wall – floors 0,1 full; floor 2 partial (left half)
    if f < 2:
        brick_wall(f"Back{f}", (bx, by + bd/2 - wall_t/2, zc), bw, floor_h, wall_t, mat_brick2)
    else:
        brick_wall(f"Back{f}", (bx - bw/4, by + bd/2 - wall_t/2, zc), bw/2, floor_h, wall_t, mat_brick2)
    # Left wall
    brick_wall(f"Left{f}", (bx - bw/2 + wall_t/2, by, zc), wall_t, floor_h, bd, mat_brick)
    # Right wall – floors 0,1; floor 2 partial (back half)
    if f < 2:
        brick_wall(f"Right{f}", (bx + bw/2 - wall_t/2, by, zc), wall_t, floor_h, bd, mat_brick2)
    else:
        brick_wall(f"Right{f}", (bx + bw/2 - wall_t/2, by + bd/4, zc), wall_t, floor_h, bd/2, mat_brick2)

# Windows (dark insets) on front wall

# ---------------- SCAFFOLDING ----------------
def scaffold_post(x, y, h):
    cyl(f"ScafPost_{x:.1f}_{y:.1f}", (x, y, h/2), 0.06, h, mat_scaffold)

def scaffold_hbar(x1, y1, x2, y2, z):
    dx, dy = x2-x1, y2-y1
    L = math.sqrt(dx*dx + dy*dy)
    mid = ((x1+x2)/2, (y1+y2)/2, z)
    ang = math.atan2(dy, dx)
    o = box(f"ScafBar_{x1:.1f}_{y1:.1f}_{z:.1f}", mid, (L, 0.08, 0.08), mat_scaffold, (0, 0, ang))
    return o

# Posts around the block at 3m intervals
sx0, sy0 = bx - bw/2 - 0.5, by - bd/2 - 0.5
sx1, sy1 = bx + bw/2 + 0.5, by + bd/2 + 0.5
steps_x = 4
steps_y = 3
for i in range(steps_x + 1):
    x = sx0 + i * (sx1 - sx0) / steps_x
    for j in range(steps_y + 1):
        y = sy0 + j * (sy1 - sy0) / steps_y
        # only scaffold on floors 0 and 1 (up to 6m)
        h = 6.0 if (i in (0, steps_x) or j in (0, steps_y)) else 4.5
        scaffold_post(x, y, h)

# Horizontal bars at each level
for lvl in [1.5, 3.0, 4.5, 6.0]:
    for i in range(steps_x):
        x1 = sx0 + i * (sx1 - sx0) / steps_x
        x2 = sx0 + (i+1) * (sx1 - sx0) / steps_x
        scaffold_hbar(x1, sy0, x2, sy0, lvl)
        scaffold_hbar(x1, sy1, x2, sy1, lvl)
    for j in range(steps_y):
        y1 = sy0 + j * (sy1 - sy0) / steps_y
        y2 = sy0 + (j+1) * (sy1 - sy0) / steps_y
        scaffold_hbar(sx0, y1, sx0, y2, lvl)
        scaffold_hbar(sx1, y1, sx1, y2, lvl)

# ---------------- STACKS OF BRICKS ----------------
def brick_stack(cx, cy, rows=5, cols=4, layers=3):
    bw2, bh2, bd2 = 0.22, 0.09, 0.11
    for l in range(layers):
        for r in range(rows):
            for c in range(cols):
                x = cx + c * (bw2 + 0.01)
                y = cy + r * (bd2 + 0.01)
                z = l * (bh2 + 0.01) + bh2/2
                m = mat_brick if (l + r + c) % 2 == 0 else mat_brick2
                box(f"Brick_{cx:.1f}_{cy:.1f}_{l}_{r}_{c}", (x, y, z), (bw2, bh2, bd2), m)

# Pallet under stacks
def pallet(cx, cy):
    box(f"Pallet_{cx:.1f}_{cy:.1f}", (cx, cy, 0.05), (1.2, 0.8, 0.1), mat_pallet)

pallet(5, -10)

pallet(3, -11)
