import bpy, bmesh, math, random
from mathutils import Vector
random.seed(1754)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
def clay(rgb, em=0, ru=.9):
    m = bpy.data.materials.new("c"); m.use_nodes = True
    n = next(x for x in m.node_tree.nodes if x.type == "BSDF_PRINCIPLED")
    n.inputs["Base Color"].default_value = (*rgb, 1); n.inputs["Roughness"].default_value = ru
    if em: n.inputs["Emission Color"].default_value = (*rgb, 1); n.inputs["Emission Strength"].default_value = em
    return m
sand, sand2, stone, stone2 = clay((.32, .22, .12)), clay((.26, .18, .11)), clay((.55, .42, .27)), clay((.42, .32, .21))
cap, cap2, roof, papyrus = clay((.62, .45, .25)), clay((.5, .36, .22)), clay((.36, .26, .17)), clay((.3, .3, .12))
ob, fire, dark = clay((.4, .3, .2)), clay((1, .45, .08), 14), clay((.05, .03, .03))
def mk(bm, m, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("h"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("h", me); o.location = at; o.rotation_euler = rot
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def blk(at, s, m, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * s[0], v.co.y * s[1], (v.co.z + .5) * s[2]))
    return mk(bm, m, at, rot)
def col(at, r1, r2, h, n, m):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r1, radius2=r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return mk(bm, m, at)
def flame(kind, c, w, at, aim=None):
    L = bpy.data.lights.new("f", kind); L.color = c; L.energy = w
    o = bpy.data.objects.new("f", L); o.location = at; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=10, radius1=7, radius2=12.5, depth=3)
for v in bm.verts:
    if v.co.z < 0: v.co += Vector((random.uniform(-1.5, 1.5), random.uniform(-1.5, 1.5), random.uniform(-1.5, .3)))
    else: v.co.z += random.uniform(-.1, .1)
    v.co.z -= 1.5
mk(bm, sand)
blk((0, 1, 0), (17, 12, .3), stone2)
for i in range(3): blk((0, 1, .3 + i * .12), (15.6 - i * .8, 10.6 - i * .8, .12), stone)
# papyrus-bud columns in two rows: fluted shaft, a bud of radiating petals, then a capital
def papyrus(x, y):
    col((x, y, .66), .5, .38, 4.4, 8, stone)
    for k in range(8):
        a = k * math.pi / 4
        col((x + .15 * math.cos(a), y + .15 * math.sin(a), 5.06), .16, .02, 1.5, 4, papyrus).rotation_euler = (math.sin(a) * .45, -math.cos(a) * .45, 0)
    col((x, y, 5.06), .2, .2, 1.5, 6, papyrus)
    blk((x, y, 6.5), (1.5, 1.5, .35), cap); blk((x, y, 6.85), (1.1, 1.1, .3), cap2)
for i in range(6):
    for y in (-3.0, 3.0): papyrus(-6.0 + i * 2.4, y)
# roof slabs: some present, some open to the noon sun, with an architrave band
for i in range(5):
    for y in (-3.0, 3.0):
        if (i, y) in {(1, -3.0), (2, 3.0), (3, -3.0), (0, 3.0)}: continue
        blk((-4.8 + i * 2.4, y, 7.15), (2.35, 1.7, .5), roof)
for y in (-3.0, 3.0): blk((0, y, 6.85), (12.2, .45, .3), stone2)
blk((0, 0, 7.15), (12.2, .5, .5), stone2)
# the back wall of the hall, with a doorway and a sun disc
blk((0, 6.8, .66), (12.2, .8, 6.6), stone2)
blk((0, 6.35, .66), (2.2, .3, 3.6), dark)

blk((0, 6.0, 5.0), (2.8, .1, .15), cap)
# braziers burning on either side of the doorway
for x in (-1.8, 1.8):
    col((x, 5.2, .66), .35, .5, .8, 6, stone2); col((x, 5.2, 1.46), .3, .02, .7, 5, fire)

# obelisk at the front, a broken column and a fallen drum
col((0, -7.5, .3), .9, .5, 6.5, 4, stone, (0, 0, math.pi / 4)); col((0, -7.5, 6.8), .5, .02, 1.4, 4, cap, (0, 0, math.pi / 4))
col((8, -5, .3), .6, .6, 1.8, 8, stone2)
col((-8.5, -4, .3), .6, .6, 1.3, 8, stone2)
for i in range(5):
    blk((-10 + i * 1.0, -8.5 + (i % 2) * .4, .3), (1.1, 1.1, .7), stone2, (0, 0, i * .5))

flame("SUN", (.35, .45, 1), 2.2, (-16, 16, 9), (0, 0, 2))
w = bpy.data.worlds.new("w"); sc.world = w; w.use_nodes = True
next(n for n in w.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.02, .025, .05, 1)

co = bpy.data.objects.new("c", cam); sc.collection.objects.link(co)
t = Vector((0, 0, 3)); co.location = t + Vector((-6, -52, 34))
co.rotation_euler = (t - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
