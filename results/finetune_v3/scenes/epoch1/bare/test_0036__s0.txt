def dye(name, rgb, glow=0, rough=.9):
    m = bpy.data.materials.new(name); m.use_nodes = True
    p = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    p.inputs["Base Color"].default_value = (*rgb, 1); p.inputs["Roughness"].default_value = rough
    if glow: p.inputs["Emission Color"].default_value = (*rgb, 1); p.inputs["Emission Strength"].default_value = glow
    Y[name] = m
for k, v in dict(soil=(.2, .14, .1), frost=(.4, .42, .5), wall=(.36, .3, .22), tile=(.12, .12, .14), wood=(.25, .15, .08),
                 trunk=(.14, .1, .08), orange=(.9, .32, .04), basket=(.4, .28, .12), red=(.5, .08, .06), string=(.7, .65, .55), leaf=(.3, .2, .05)).items(): dye(k, v)
dye("pane", (1, .65, .25), 14); dye("ember", (1, .4, .08), 16)
def mk(bm, m, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("y"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("y", me); o.location = at; o.rotation_euler = rot
    sc.collection.objects.link(o); o.data.materials.append(Y[m]); return o
def blk(x, y, z, w, d, h, m, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * w, v.co.y * d, (v.co.z + .5) * h))
    return mk(bm, m, (x, y, z), rot)
def cyl(x, y, z, r, h, m, n=6, r2=None):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return mk(bm, m, (x, y, z))
def ball(x, y, z, r, m, sq=1):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co.z *= sq
    return mk(bm, m, (x, y, z))
def limb(a, b, t, m):
    a, b = Vector(a), Vector(b); q = (b - a).to_track_quat('Z', 'Y')
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = q @ Vector((v.co.x * t, v.co.y * t, (v.co.z + .5) * (b - a).length)) + a
    return mk(bm, m)
def lit(x, y, z, c, w):
    L = bpy.data.lights.new("l", "POINT"); L.color = c; L.energy = w
    o = bpy.data.objects.new("l", L); o.location = (x, y, z); sc.collection.objects.link(o)

blk(0, 9.5, 3.9, 11.2, 5.2, .15, "wood")
