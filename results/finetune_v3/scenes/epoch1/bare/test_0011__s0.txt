random.seed(1755)

sc = bpy.context.scene
def dye(rgb, glow=0):
    m = bpy.data.materials.new("d"); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*rgb, 1); b.inputs["Roughness"].default_value = .9
    if glow: b.inputs["Emission Color"].default_value = (*rgb, 1); b.inputs["Emission Strength"].default_value = glow
    return m
mud, mud2, stone, stone2, grass, water, bank, willow, reed, coat, hat, skin, lampm, birdm = (dye(c) for c in (
    (.3,.17,.1),(.22,.12,.08),(.3,.3,.33),(.2,.2,.24),(.1,.2,.08),(.1,.2,.3),(.2,.14,.08),(.12,.28,.12),(.5,.4,.15),(.1,.2,.35),(.5,.45,.3),(.55,.35,.25),(.2,.1,.05),(.8,.8,.85)))
lamp = dye((1,.7,.3), 14)
def put(bm, m, at=(0,0,0), rot=(0,0,0)):
    me = bpy.data.meshes.new("p"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("p", me); o.location = at; o.rotation_euler = rot
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def cube(at, d, m, rot=(0,0,0)):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x*d[0], v.co.y*d[1], (v.co.z+.5)*d[2]))
    return put(bm, m, at, rot)
def cyl(at, r, h, n, m, r2=None, rot=(0,0,0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h/2
    return put(bm, m, at, rot)
def ball(at, r, m, sq=(1,1,1)):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co = Vector((v.co.x*sq[0], v.co.y*sq[1], v.co.z*sq[2]))
    return put(bm, m, at)
def rod(a, b, t, m):
    a, b = Vector(a), Vector(b); d = b-a; q = d.to_track_quat('Z', 'Y')
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = q @ Vector((v.co.x*t, v.co.y*t, (v.co.z+.5)*d.length)) + a
    return put(bm, m)
def glow(c, w, at):
    L = bpy.data.lights.new("g", "POINT"); L.color = c; L.energy = w
    o = bpy.data.objects.new("g", L); o.location = at; sc.collection.objects.link(o)
# base: stream bed with banks, the stream as a dark slab between them
bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
for v in bm.verts:
    v.co = Vector((v.co.x*22, v.co.y*17, v.co.z*3))

    else: v.co.z += random.uniform(-.08,.08)

    cube((x,0,z-1.2), (1.1,4.7,2.4), stone if i%2 else stone2, (0,0,0))
for i in range(7): cube((-5.6+i*1.8, 2.4, 4.8), (1.6,.5,.6), stone2); cube((-5.6+i*1.8, -2.4, 4.8), (1.6,.5,.6), stone2)
# mud nests under the arch: stacked lumps with a hole
for i, (x, h) in enumerate(((-4.2,2.6),(-2.2,3.4),(0,3.9),(2.2,3.4),(4.2,2.6))):

# willow overhanging the bank
cyl((-8.5, 6.5, 0), .8, 3.2, 6, bank, .5)
for k in range(9):
    a = k*.7; rod((-8.5+math.cos(a)*1.6, 6.5+math.sin(a)*1.6, 3.2), (-8.5+math.cos(a)*2.4, 6.5+math.sin(a)*2.4, 1.0), .16, willow)

for i in range(18):

    for s in (-1, 1): cube((x-s*.2, y, z+.1), (.5,.05,.4), birdm, (0,0,d)).rotation_euler.y = d + s*.7

for c, w, p in (((.45,.6,1), 4.5, (-26,-14,9)), ((.9,.5,.4), 2.2, (22,20,12))):
