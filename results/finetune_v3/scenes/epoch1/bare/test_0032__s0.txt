def pigment(rgb, em=0):
    m = bpy.data.materials.new("p"); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*rgb, 1); b.inputs["Roughness"].default_value = .9
    if em: b.inputs["Emission Color"].default_value = (*rgb, 1); b.inputs["Emission Strength"].default_value = em
    return m
mud, mud2, stone, stone2, roof = pigment((.22, .15, .1)), pigment((.12, .09, .07)), pigment((.6, .58, .52)), pigment((.42, .4, .38)), pigment((.3, .1, .12))
gold, glass, water, wood, iron = pigment((1, .7, .15), 10), pigment((1, .75, .3), 7), pigment((.1, .6, .7), 3), pigment((.28, .16, .09)), pigment((.12, .12, .14))
cloth = [pigment(c) for c in ((.6, .08, .1), (.1, .2, .6), (.8, .7, .15))]
def mesh(bm, m, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("m"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("m", me); o.location, o.rotation_euler = at, rot
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def box(at, s, m, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * s[0], v.co.y * s[1], (v.co.z + .5) * s[2]))
    return mesh(bm, m, at, rot)
def col(at, r, h, n, m, r2=None, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return mesh(bm, m, at, rot)
def gable(at, w, d, h, m):
    bm = bmesh.new(); p = [bm.verts.new(c) for c in ((-w / 2, -d / 2, 0), (w / 2, -d / 2, 0), (w / 2, d / 2, 0), (-w / 2, d / 2, 0), (-w / 2, 0, h), (w / 2, 0, h))]
    for f in ((0, 1, 5, 4), (2, 3, 4, 5), (0, 4, 3), (1, 2, 5), (0, 1, 2, 3)): bm.faces.new([p[i] for i in f])
    bmesh.ops.recalc_face_normals(bm, faces=bm.faces); return mesh(bm, m, at)

    o = bpy.data.objects.new("g", L); o.location = at; sc.collection.objects.link(o)
