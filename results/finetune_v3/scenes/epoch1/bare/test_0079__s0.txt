def cast(bm, k, at=(0, 0, 0), rot=(0, 0, 0)):

def blk(x, y, z, w, d, h, k, rz=0, top=1):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts:
        f = top if v.co.z > 0 else 1
        v.co = Vector((v.co.x * w * f, v.co.y * d * f, (v.co.z + .5) * h))
    return cast(bm, k, (x, y, z), (0, 0, rz))
def tube(x, y, z, r, h, n, k, rot=(0, 0, 0), r2=None):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return cast(bm, k, (x, y, z), rot)
def beam(a, b, w, k):
    a, b = Vector(a), Vector(b); q = (b - a).to_track_quat('Z', 'Y')
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = q @ Vector((v.co.x * w, v.co.y * w, (v.co.z + .5) * (b - a).length)) + a
    return cast(bm, k)
