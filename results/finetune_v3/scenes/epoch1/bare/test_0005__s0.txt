from mathutils import Vector
random.seed(1510)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
def tone(rgb, em=0):
    m = bpy.data.materials.new("t"); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*rgb, 1); b.inputs["Roughness"].default_value = .8
    if em: b.inputs["Emission Color"].default_value = (*rgb, 1); b.inputs["Emission Strength"].default_value = em
    return m
sea, snow, rock, pine, plank, red, hull, glass, pump, white, steel = (tone((.04, .13, .2)), tone((.55, .6, .7)), tone((.17, .18, .22)), tone((.05, .17, .12)), tone((.3, .2, .12)), tone((.5, .07, .05)), tone((.7, .68, .6)), tone((1, .75, .35), 7), tone((.6, .08, .06)), tone((.75, .75, .72)), tone((.3, .32, .35)))
def shape(kind, m, at, size, rot=(0, 0, 0), n=6):
    bm = bmesh.new()
    if kind == "box": bmesh.ops.create_cube(bm, size=1)
    elif kind == "cyl": bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=.5, radius2=.5, depth=1)
    else: bmesh.ops.create_icosphere(bm, subdivisions=1, radius=.5)
    for v in bm.verts:
        if kind != "ico": v.co.z += .5
        v.co += Vector((random.uniform(-.03, .03), random.uniform(-.03, .03), random.uniform(-.03, .03)))
    me = bpy.data.meshes.new("s"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("s", me); o.location = at; o.scale = size; o.rotation_euler = rot
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def ridge(m, at, r, h, sq=.8):
    o = shape("cyl", m, at, (r, r, h), n=5)
    for v in o.data.vertices: v.co.z = max(v.co.z - h / 2, 0) * sq
    return o
def lamp(kind, c, w, at, aim=None):
    L = bpy.data.lights.new("l", kind); L.color = c; L.energy = w
    o = bpy.data.objects.new("l", L); o.location = at; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
