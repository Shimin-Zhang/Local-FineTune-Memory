from mathutils import Vector
random.seed(1587)

sc = bpy.context.scene
mats = {}
def colour(name, rgb, glow=0):
    m = bpy.data.materials.new(name); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*rgb, 1); b.inputs["Roughness"].default_value = .9
    if glow: b.inputs["Emission Color"].default_value = (*rgb, 1); b.inputs["Emission Strength"].default_value = glow
    mats[name] = m
for k, v in dict(earth=(.12, .08, .06), pine=(.04, .14, .09), trunk=(.2, .12, .08), red=(.7, .05, .04), white=(.8, .78, .72),
                 stone=(.35, .36, .38), moss=(.1, .22, .08), thatch=(.1, .12, .07), dark=(.03, .02, .02), leaf=(.1, .3, .1)).items(): colour(k, v)
colour("lantern", (1, .7, .3), 16); colour("shrine", (1, .5, .15), 12)
def put(bm, k, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("m"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("o", me); o.location, o.rotation_euler = at, rot
    sc.collection.objects.link(o); o.data.materials.append(mats[k]); return o
def slab(at, s, k, rz=0, top=1):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts:
        f = top if v.co.z > 0 else 1
        v.co = Vector((v.co.x * s[0] * f, v.co.y * s[1] * f, (v.co.z + .5) * s[2]))
    return put(bm, k, at, (0, 0, rz))
def cone(at, r, h, n, k, r2=0, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return put(bm, k, at, rot)
def blob(at, r, k, sq=1):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co *= random.uniform(.85, 1.15); v.co.z *= sq
    return put(bm, k, at)
def light(kind, c, w, at, aim=None):
    l = bpy.data.lights.new("l", kind); l.color, l.energy = c, w
    o = bpy.data.objects.new("l", l); o.location = at; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()

    at(0, 0, .25, 1.3, .6, .5, "white"); at(.65, 0, .5, .55, .45, .4, "white"); at(.85, 0, .9, .3, .3, .12, "dark")
    for sd in (-1, 1):
        at(.55, sd * .18, .85, .12, .1, .35, "white"); at(.1, sd * .17, 0, .2, .15, .3, "white"); at(-.6, sd * .12, 0, .2, .15, .3, "white")
    at(-.9, 0, .6, .6, .2, .2, "white", a + .3)
for i, (x, y) in enumerate(((-4.6, -6.5), (4.2, -3.8), (-4.4, -1.0), (4.0, 1.7), (-3.8, 4.4), (3.6, 7.0))):
