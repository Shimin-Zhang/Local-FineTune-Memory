import bpy, bmesh, random, math
from mathutils import Vector
random.seed(1232)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
def pigment(rgb, glow=0):
    m = bpy.data.materials.new("p"); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*rgb, 1); b.inputs["Roughness"].default_value = .95
    if glow: b.inputs["Emission Color"].default_value = (*rgb, 1); b.inputs["Emission Strength"].default_value = glow
    return m
def add(bm, m, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("p"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("p", me); o.location = at; o.rotation_euler = rot
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def box(w, d, h, m, at, rot=(0, 0, 0), taper=1):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts:
        k = taper if v.co.z > 0 else 1
        v.co = Vector((v.co.x * w * k, v.co.y * d * k, (v.co.z + .5) * h))
    return add(bm, m, at, rot)
def pipe(r, h, m, at, n=6, r2=None):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return add(bm, m, at)
def mound(r, m, at, sq=.45):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co *= random.uniform(.85, 1.15); v.co.z *= sq
    return add(bm, m, at)
def lamp(at, rgb, w):
    L = bpy.data.lights.new("l", "POINT"); L.color = rgb; L.energy = w
    o = bpy.data.objects.new("l", L); o.location = at; sc.collection.objects.link(o)
sand, spoil, spoil2, rust, tin, dark, door = pigment((.45, .17, .06)), pigment((.38, .12, .05)), pigment((.3, .09, .05)), pigment((.4, .12, .05)), pigment((.55, .5, .45)), pigment((.02, .02, .03)), pigment((.1, .22, .18))
win, beacon, fire = pigment((1, .7, .25), 14), pigment((1, .1, .05), 14), pigment((1, .45, .1), 16)
bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=10, radius1=7, radius2=14.5, depth=3.2)
for v in bm.verts:
    if v.co.z < 0: v.co += Vector((random.uniform(-1.2, 1.2), random.uniform(-1.2, 1.2), random.uniform(-1.4, .3)))
    else: v.co.z += random.uniform(-.1, .1)
    v.co.z -= 1.6
add(bm, sand)
# the hillside: a flat-topped ridge along the back with dugout doors, a lit window and a chimney
hx, hy = 0, 8
for i in range(7):
    a = math.radians(-15 + i * 6)
    box(6.4, 3.6, 4.6, spoil if i % 2 else spoil2, (hx + math.cos(a) * 11.2, hy - math.sin(a) * 11.2, 0), (0, 0, a), .75)
box(15, 3.2, 4.6, spoil, (hx, hy + 1.2, 0), taper=.8)
for k, x in enumerate((-5, -1.6, 1.8, 5.2)):
    box(1.7, .3, 2.0, door if k != 2 else dark, (x, hy - .6, 0), taper=.9)
    box(1.9, .3, .3, tin, (x, hy - .65, 2.0))
box(1.0, .2, .8, win, (1.8, hy - .65, .9)); lamp((1.8, hy - 2.2, 1.4), (1, .7, .25), 500)
pipe(.5, 1.6, rust, (-8, hy + 1.5, 4.2), 6, .35); box(1.2, .3, .2, tin, (-8, hy - .6, 4.2))
# spoil mounds with pipes sticking up from the ground
for i, (x, y, r) in enumerate(((-8, -3, 3.4), (7, -5, 3.0), (-2, -8, 2.4), (10, 3, 2.2), (-11, 4, 2.0), (2, -1, 1.8))):
    mound(r, spoil if i % 2 else spoil2, (x, y, 0), .5)
for i, (x, y, h) in enumerate(((-8, -3, 2.6), (7, -5, 2.2), (2, -1, 1.6), (-2, -8, 1.4), (10, 3, 1.8), (-11, 4, 1.5), (-5, 2, 1.2), (4, 3, 1.4), (9, -9, 1.0), (-9, -8, 1.1))):
    pipe(.42, h, rust, (x, y, .2), 6, .36)
    pipe(.5, .3, tin, (x, y, .2 + h), 6, .5)
# the tin sign: a board on two posts with a painted arrow
for sx in (-1, 1): box(.2, .2, 3.2, tin, (3 + sx * 2.2, -6, 0))
box(5.4, .2, 2.4, tin, (3, -6, 1.2), taper=.95)
box(3.2, .1, .5, dark, (2.6, -6.15, 2.0)); box(.5, .1, 1.0, dark, (4.4, -6.15, 2.0)); box(.5, .1, .5, dark, (4.9, -6.15, 2.25))
box(3.2, .1, .25, dark, (2.6, -6.15, 1.35)); box(1.4, .1, .25, dark, (1.8, -6.15, 1.7))
# a dugout camp: tin shed, a barrel, a campfire ring and a beacon pole
box(4.0, 3.0, 2.2, tin, (-4, -6.5, 0), taper=.9); box(4.4, 3.4, .2, rust, (-4, -6.5, 2.2), taper=.8)
box(.9, .15, 1.7, dark, (-4, -8.05, 0)); box(.9, .1, .5, win, (-2.2, -8.05, 1.2)); lamp((-2.2, -8.8, 1.4), (1, .7, .25), 250)
pipe(.7, 1.4, rust, (-1.0, -7.5, 0), 7); pipe(.75, .15, tin, (-1.0, -7.5, 1.4), 7)
for i in range(7):
    a = i * math.tau / 7; box(.5, .3, .4, spoil2, (6 + math.cos(a) * 1.2, -2 + math.sin(a) * 1.2, 0), (0, 0, a))
pipe(.5, .6, fire, (6, -2, 0), 5, .15); lamp((6, -2, 1.4), (1, .5, .15), 500)
pipe(.12, 6.5, tin, (11, -7, 0), 5); box(.5, .5, .5, beacon, (11, -7, 6.5)); lamp((11, -7, 7.2), (1, .1, .05), 700)
for col, w, p in (((1, .45, .25), 5.5, (-22, -12, 8)), ((.5, .35, .8), 2.5, (18, 22, 12))):
    L = bpy.data.lights.new("s", "SUN"); L.color = col; L.energy = w
    o = bpy.data.objects.new("s", L); o.location = p; sc.collection.objects.link(o)
    o.rotation_euler = (Vector((0, 0, 1)) - o.location).to_track_quat('-Z', 'Y').to_euler()
sc.world = bpy.data.worlds.new("dusk"); sc.world.use_nodes = True
next(n for n in sc.world.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.03, .012, .02, 1)
cam = bpy.data.cameras.new("cam"); cam.lens = 80; cam.clip_end = 500
co = bpy.data.objects.new("cam", cam); sc.collection.objects.link(co)
tg = Vector((0, 1, 1.5)); co.location = tg + Vector((6, -60, 38))
co.rotation_euler = (tg - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
