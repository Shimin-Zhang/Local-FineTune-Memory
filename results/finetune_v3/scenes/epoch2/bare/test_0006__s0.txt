import bpy, bmesh, math, random
from mathutils import Vector
random.seed(1477)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
mats = {}
def tone(name, rgb, glow=0, rough=.9):
    m = bpy.data.materials.new(name); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*rgb, 1); b.inputs["Roughness"].default_value = rough
    if glow: b.inputs["Emission Color"].default_value = (*rgb, 1); b.inputs["Emission Strength"].default_value = glow
    mats[name] = m
for k, v in dict(grass=(.1, .2, .06), turf=(.14, .22, .08), thatch=(.45, .33, .13), thatch2=(.33, .24, .1), wattle=(.3, .2, .1),
                 dark=(.02, .015, .01), clay=(.35, .16, .1), stone=(.25, .24, .22), iron=(.08, .08, .09), hide=(.4, .3, .2)).items(): tone(k, v)
tone("fire", (1, .45, .08), 25); tone("forge", (1, .35, .05), 14)
def make(bm, key, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("m"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("m", me); o.location = at; o.rotation_euler = rot
    sc.collection.objects.link(o); o.data.materials.append(mats[key]); return o
def cube(at, d, key, rz=0):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * d[0], v.co.y * d[1], (v.co.z + .5) * d[2]))
    return make(bm, key, at, (0, 0, rz))
def cone(at, r, h, n, key, r2=0.0, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return make(bm, key, at, rot)
def ball(at, r, key, sq=1):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co *= random.uniform(.85, 1.15); v.co.z *= sq
    return make(bm, key, at)
def glow(at, rgb, w):
    L = bpy.data.lights.new("g", "POINT"); L.color = rgb; L.energy = w
    o = bpy.data.objects.new("g", L); o.location = at; sc.collection.objects.link(o)
# high grass hill: flat crown, steep grassy skirt, rough underside
bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=10, radius1=6, radius2=14.5, depth=3.6)
for v in bm.verts:
    if v.co.z < 0: v.co += Vector((random.uniform(-1.4, 1.4), random.uniform(-1.4, 1.4), random.uniform(-1.4, .2)))
    else: v.co.z += random.uniform(-.06, .06)
    v.co.z -= 1.8
make(bm, "grass")
# ditch and bank ring: low grassy bank, sunken ditch, inner bank
for i in range(18):
    a = i * math.pi / 9
    for r, h, k in ((10.6, .9, "turf"), (8.6, .35, "turf")):
        cube((math.cos(a) * r, math.sin(a) * r, -.1), (3.0, 1.5, h), k, a + math.pi / 2)
    cube((math.cos(a) * 9.5, math.sin(a) * 9.5, -.55), (1.6, 1.8, .15), "grass", a + math.pi / 2)
# circle of thatched huts, doors facing the middle
for i in range(8):
    a = i * math.pi / 4 + .2
    x, y = math.cos(a) * 6.2, math.sin(a) * 6.2
    cone((x, y, 0), 1.7, 1.5, 8, "wattle", 1.65)
    cone((x, y, 1.3), 2.4, 2.0, 8, "thatch" if i % 2 else "thatch2", .25)
    cone((x, y, 3.2), .3, .4, 5, "thatch2", .05)
    d = Vector((math.cos(a), math.sin(a), 0))
    cube((x + d.x * 1.55, y + d.y * 1.55, 0), (.9, .15, 1.1), "dark", a + math.pi / 2)
    cube((x + d.x * 1.7, y + d.y * 1.7, 1.0), (.9, .1, .1), "hide", a + math.pi / 2)
# central gathering place: fire ring, logs, hanging pots, standards
for k in range(7):
    a = k * 2 * math.pi / 7; ball((math.cos(a) * 1.3, math.sin(a) * 1.3, .2), .35, "stone", .8)
cone((0, 0, 0), .5, .9, 5, "fire", .1); glow((0, 0, 1.2), (1, .5, .1), 900)
for k in range(3):
    a = k * 2.1; cube((math.cos(a) * 2.3, math.sin(a) * 2.3, 0), (2.2, .35, .35), "wattle", a + math.pi / 2)
for sx in (-1, 1):
    cube((sx * 2.6, 0, 0), (.2, .2, 3.2), "wattle")
cube((0, 0, 3.0), (5.4, .2, .2), "wattle"); ball((0, 0, 2.5), .5, "clay", 1.2)
# smithy: open-fronted shed with a glowing forge, anvil and tools
sx, sy = -3.2, -3.4
for k in range(3): cube((sx - 1.5 + k * 1.5, sy - 1.4, 0), (.2, .2, 2.4), "wattle")
for k in range(3): cube((sx - 1.5 + k * 1.5, sy + 1.4, 0), (.2, .2, 2.4), "wattle")
cube((sx, sy + 1.4, 2.4), (3.4, .2, .2), "wattle"); cube((sx, sy - 1.4, 2.4), (3.4, .2, .2), "wattle")
cone((sx, sy, 2.5), 3.2, 1.4, 4, "thatch", .1, (0, 0, math.pi / 4))
cube((sx, sy + .6, 0), (1.6, 1.2, 1.0), "stone"); cube((sx, sy + .6, 1.0), (1.2, .8, .1), "forge")
glow((sx, sy + .2, 1.4), (1, .4, .08), 500)
cube((sx + 2.2, sy - .2, 0), (.5, .5, .8), "iron"); cube((sx + 2.2, sy - .2, .8), (1.0, .4, .25), "iron")
cube((sx - 2.3, sy - .6, 0), (.12, 1.6, .12), "iron", .5)
# storage jars, a well, a low palisade gate post pair and hay
for i, (x, y) in enumerate(((3.6, -3.2), (4.4, -2.6), (-4.2, 3.8), (-5.0, 3.2))): cone((x, y, 0), .5, .9, 7, "clay", .3)
cone((3.0, 3.8, 0), .9, 1.0, 8, "stone", .8); cube((3.0, 3.8, 1.0), (.12, .12, 1.6), "wattle"); cube((3.0, 3.8, 2.5), (1.4, .12, .12), "wattle")
for i in range(4): ball((random.uniform(-7, 7), random.uniform(-7, 7), .3), random.uniform(.3, .5), "turf", .6)
# cold dawn: blue-green key low from the left, violet rim from behind
for c, w, p in (((.5, .75, .85), 4.5, (-24, -14, 8)), ((.6, .4, .9), 2.2, (18, 20, 10))):
    S = bpy.data.lights.new("s", "SUN"); S.color = c; S.energy = w
    o = bpy.data.objects.new("s", S); o.location = p; sc.collection.objects.link(o)
    o.rotation_euler = (Vector((0, 0, 1)) - o.location).to_track_quat('-Z', 'Y').to_euler()
wd = bpy.data.worlds.new("dawn"); sc.world = wd; wd.use_nodes = True
next(n for n in wd.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.015, .025, .035, 1)
cm = bpy.data.cameras.new("cam"); cm.lens = 80; cm.clip_end = 500
co = bpy.data.objects.new("cam", cm); sc.collection.objects.link(co)
aim = Vector((0, 0, 1.2)); co.location = aim + Vector((-16, -50, 34))
co.rotation_euler = (aim - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
