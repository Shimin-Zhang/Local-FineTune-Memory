from mathutils import Vector
random.seed(1878)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
M = {}
def paint(name, rgb, glow=0, rough=.85):
    m = bpy.data.materials.new(name); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*rgb, 1); b.inputs["Roughness"].default_value = rough
    if glow:
        b.inputs["Emission Color"].default_value = (*rgb, 1); b.inputs["Emission Strength"].default_value = glow
    M[name] = m
paint("sand", (.2, .17, .12)); paint("sand2", (.14, .12, .1)); paint("rock", (.1, .1, .12)); paint("hull", (.35, .12, .08), rough=.5)
paint("deck", (.5, .45, .38), rough=.5); paint("cable", (.9, .75, .1), rough=.5); paint("rov", (.1, .35, .6), rough=.4)
paint("arm", (.8, .8, .8), rough=.4); paint("lamp", (.9, 1, 1), 22); paint("ray", (.35, .3, .28)); paint("fish", (.2, .8, .9), 3)
paint("weed", (.05, .2, .12)); paint("window", (1, .7, .3), 8)
def build(bm, mat, loc=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("m"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("m", me); o.location = loc; o.rotation_euler = rot
    sc.collection.objects.link(o); o.data.materials.append(M[mat]); return o
def box(loc, s, mat, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * s[0], v.co.y * s[1], (v.co.z + .5) * s[2]))
    return build(bm, mat, loc, rot)
def rod(a, b, t, mat):
    a, b = Vector(a), Vector(b); d = b - a; q = d.to_track_quat('Z', 'Y')
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = q @ Vector((v.co.x * t, v.co.y * t, (v.co.z + .5) * d.length)) + a
    return build(bm, mat)
def cone(loc, r1, r2, h, n, mat, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r1, radius2=r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return build(bm, mat, loc, rot)
def pebble(loc, r, mat, sq=.6):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co *= random.uniform(.8, 1.2); v.co.z *= sq
    return build(bm, mat, loc)
def glint(loc, rgb, w):
    L = bpy.data.lights.new("l", "POINT"); L.color = rgb; L.energy = w
    o = bpy.data.objects.new("l", L); o.location = loc; sc.collection.objects.link(o)
def ray(c, yaw, pitch, s, tilt):
    c = Vector(c)
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=4, radius1=.5 * s, radius2=.5 * s, depth=1.6 * s)
    for v in bm.verts: v.co = Vector((v.co.x * 1.0, v.co.y * 2.6, v.co.z))
    build(bm, "ray", c, (pitch, 0, yaw))
    for sd in (-1, 1):
        p = c + Vector((sd * .6 * s * math.cos(yaw), sd * .6 * s * math.sin(yaw), 0))
        cone(p, .12 * s, .02, 1.0 * s, 4, "ray", (0, math.pi / 2, yaw))
    for sd in (-1, 1):
        e = c + Vector((sd * .5 * s * math.cos(yaw), sd * .5 * s * math.sin(yaw), .2 * s))
        rod(e, e + Vector((sd * 1.6 * s * math.cos(yaw + .3 * sd), sd * 1.6 * s * math.sin(yaw + .3 * sd), -.3 * s)), .06 * s, "ray")

# a sloping seabed: high at the back, falling to the front
bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=10, radius1=6, radius2=11, depth=3)
for v in bm.verts:
    if v.co.z < 0: v.co += Vector((random.uniform(-1.5, 1.5), random.uniform(-1.5, 1.5), random.uniform(-1.2, .3)))
    else: v.co.z += 1.5 + v.co.y * .22 + random.uniform(-.1, .1)
build(bm, "sand")
for i in range(14):
    pebble((random.uniform(-8, 8), random.uniform(-8, 8), 0), random.uniform(.4, 1.0), "sand2" if i % 2 else "rock", .5)
def surf(x, y): return 1.5 + y * .22
# the ship's stern above: a hull block on a crag, deck, funnel, glowing windows
box((0, 9.6, 4.2), (7.5, 4.2, 1.6), "hull"); box((0, 9.6, 5.8), (7.8, 4.4, .25), "deck")
for i in range(5): box((-2.6 + i * 1.3, 7.6, 6.0), (.9, .1, .6), "window")

glint((0, 7.0, 6.2), (1, .7, .3), 500)
# the thick cable: a sagging line of chunks from the stern down the slope to the ROV
pts = [Vector((-1.2, 7.6, 4.4))]
for k in range(1, 11):
    t = k / 10; y = 7.6 - t * 11.2; x = -1.2 + math.sin(t * 5) * .5
    pts.append(Vector((x, y, max(.55, surf(x, y) - .15))))
for a, b in zip(pts, pts[1:]): rod(a, b, .42, "cable")
for p in pts[1:]: pebble((p.x, p.y, p.z - .1), .3, "cable", 1.0)
# the ROV tool: body, skids, arms, thrusters, a lamp that glows
rv = pts[-1] + Vector((.6, -.4, 1.0))
box(rv, (1.5, 1.1, .9), "rov"); box(rv + Vector((0, 0, .9)), (1.1, .8, .15), "arm")
for sd in (-1, 1):
    box(rv + Vector((sd * .85, 0, -.3)), (.15, 1.2, .12), "arm"); cone(rv + Vector((sd * .75, 0, 1.0)), .25, .25, .2, 6, "arm", (0, math.pi / 2, 0))
    rod(rv + Vector((sd * .7, .6, -.3)), rv + Vector((sd * .7, 1.4, .4)), .08, "arm"); rod(rv + Vector((sd * .7, 1.4, .4)), rv + Vector((sd * 1.0, 1.6, .1)), .08, "arm")
    pebble(rv + Vector((sd * .7, 1.65, .05)), .18, "lamp", 1)
cone(rv + Vector((0, -.6, -.1)), .2, .2, .15, 6, "lamp", (-math.pi / 2, 0, 0))
glint(rv + Vector((0, -1.2, .2)), (.8, .95, 1), 900)

for i, (r, yaw, pitch) in enumerate([(-1, 0.4, .1, 1.5), (1.2, 3.6, -.1, 1.2), (0, 5.2, .2, 1.7)]):
    ray(rv + Vector((math.cos(yaw) * 3.0 * (1 + .3 * i), math.sin(yaw) * 3.0 * (1 + .3 * i), 1.2 + i * .5)), yaw + math.pi / 2, pitch, 1.4 + i * .3, 0)
for i in range(11):

for i in range(5): pebble((random.uniform(-8, 8), random.uniform(-8, -2), surf(0, 0) - .2), random.uniform(.3, .5), "rock", 1)
# deep blue night: cold blue key from the left, dim green rim from behind
for col, w, p in (((.3, .5, 1), 4.5, (-22, -14, 12)), ((.2, .8, .6), 2, (16, 22, 10))):
