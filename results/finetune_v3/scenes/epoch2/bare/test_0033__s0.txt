import bpy, bmesh, math, random
from mathutils import Vector
random.seed(437)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
def mk(rgb, glow=0):
    m = bpy.data.materials.new("m"); m.use_nodes = True
    p = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    p.inputs["Base Color"].default_value = (*rgb, 1); p.inputs["Roughness"].default_value = .9
    if glow: p.inputs["Emission Color"].default_value = (*rgb, 1); p.inputs["Emission Strength"].default_value = glow
    return m
turf, turf2, bone, bone2, granite, stream, moss, flower, fern = (mk((.1, .17, .08)), mk((.14, .22, .1)), mk((.62, .58, .45)), mk((.45, .42, .34)),
    mk((.28, .28, .32)), mk((.05, .22, .35), 1.2), mk((.2, .3, .1)), mk((.9, .5, .1)), mk((.1, .3, .15)))
def solid(bm, m, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("b"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("b", me); o.location, o.rotation_euler = at, rot
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def slab(at, s, m, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * s[0], v.co.y * s[1], (v.co.z + .5) * s[2]))
    return solid(bm, m, at, rot)
def spike(at, r1, r2, h, n, m, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r1, radius2=r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return solid(bm, m, at, rot)
def lump(at, r, m, sq=.7):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co = Vector((v.co.x, v.co.y, v.co.z * sq)) * random.uniform(.85, 1.15)
    return solid(bm, m, at)
def bone(a, b, r, m, r2=None):
    a, b = Vector(a), Vector(b); d = b - a
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=6, radius1=r, radius2=r2 or r, depth=d.length)
    for v in bm.verts: v.co.z += d.length / 2
    return solid(bm, m, a, d.to_track_quat('Z', 'Y').to_euler())
def beam(kind, c, w, at, aim=None):
    L = bpy.data.lights.new("l", kind); L.color, L.energy = c, w
    o = bpy.data.objects.new("l", L); o.location = at; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=10, radius1=7, radius2=13.5, depth=3.4)
for v in bm.verts:
    v.co.z -= 1.7
    if v.co.z < -1: v.co += Vector((random.uniform(-1.4, 1.4), random.uniform(-1.4, 1.4), random.uniform(-1.8, .3)))
solid(bm, turf)
# the stream: a winding ribbon of glowing water across the valley
pts = [(-12, -6), (-8, -3.5), (-4, -2), (0, -3), (4, -5), (8, -3.5), (12, -1)]
for (x0, y0), (x1, y1) in zip(pts, pts[1:]):
    slab(((x0 + x1) / 2, (y0 + y1) / 2, 0), (math.hypot(x1 - x0, y1 - y0) + .6, 1.8, .08), stream, (0, 0, math.atan2(y1 - y0, x1 - x0)))
for i in range(10):
    lump((random.uniform(-10, 10), random.uniform(-6, -1), 0), .3, granite, .6)
# the dragon's spine: a row of vertebrae with spines, and ribs curving off both sides
for i in range(10):
    x = -10 + i * 1.5; y = 4.5 + math.sin(i * .6) * 1.2
    lump((x, y, .2), .75, bone, 1.0); spike((x, y, .8), .12, .02, 1.6, 4, bone2)
    for s in (-1, 1):
        for k in range(5):
            a0, a1 = .15 + k * .28, .15 + (k + 1) * .28
            p0 = Vector((x, y + s * 1.4 * math.sin(a0), 1.0 + 1.4 * math.cos(a0)))
            p1 = Vector((x, y + s * 1.4 * math.sin(a1), 1.0 + 1.4 * math.cos(a1)))
            bone(p0, p1, .22 - k * .015, bone)
        bone((x, y, 1.0), (x, y + s * 1.4 * math.sin(.15), 2.4), .26, bone2)
# skull at the valley's end: cranium, snout, teeth, jaw, eye sockets and horns
sx, sy = 0, 9.5
lump((sx, sy, 1.5), 2.0, bone, .8); slab((sx, sy - 2.6, .8), (2.4, 3.2, 1.2), bone2, (0, 0, 0))
for i in range(7):
    for s in (-1, 1): spike((sx + s * .95, sy - 3.8 + i * .35, .8), .12, .02, .6, 4, bone, (math.pi, 0, 0))
for s in (-1, 1):
    slab((sx + s * 1.2, sy - 1.6, .1), (1.0, 3.4, .5), bone2, (0, 0, s * .25)); lump((sx + s * 1.2, sy - .3, 1.9), .5, bone2, .5)
    slab((sx + s * 1.1, sy - 1.6, 1.9), (.9, .5, .35), granite); bone((sx + s * 1.4, sy + .6, 2.0), (sx + s * 2.6, sy + 1.8, 3.6), .16, bone, .04)
# a fallen rib cage, half sunk in grass, with grass tufts over the bones
for i in range(5):
    x, y = -6 + i * 1.3, -8 + i * .3
    for s in (-1, 1): bone((x, y, .1), (x + .3, y + s * 1.6, 1.4), .16, bone2); bone((x + .3, y + s * 1.6, 1.4), (x + .8, y + s * 1.9, 1.0), .12, bone)
for i in range(70):
    x, y = random.uniform(-11, 11), random.uniform(-9, 10)
    if abs(y - (-3 + math.sin(x * .3) * 2)) < 1.6: continue
    spike((x, y, 0), .22, .02, random.uniform(.5, 1.1), 4, random.choice((turf2, fern, moss)), (random.uniform(-.4, .4), random.uniform(-.4, .4), 0))
for i in range(8): lump((random.uniform(-9, 9), random.uniform(-9, 0), .1), random.uniform(.4, .8), granite, .6)
# the standing stone: a tall slab with a capstone, moss and small offerings
slab((6, 6, 0), (2.2, .8, 6.0), granite, (0, .04, .2)); slab((6, 6, 5.6), (2.6, 1.0, .5), granite, (0, 0, .2))
lump((6.2, 5.5, .3), .6, moss, .5); slab((5.2, 5.0, 0), (.5, .4, .3), granite, (0, 0, .5))
spike((4.8, 5.4, 0), .12, .02, .5, 5, flower)
beam("POINT", (.4, .9, 1), 250, (6, 5.0, 1.5))
for i in range(5): spike((random.uniform(-10, 10), random.uniform(-9, -4), 0), .08, .02, .4, 5, flower)
# misty teal dawn: pale green key from the left, violet rim from behind
beam("SUN", (.7, 1, .8), 5, (-24, -12, 10), (0, 0, 1)); beam("SUN", (.8, .5, 1), 2.2, (22, 20, 9), (0, 0, 1))
w = bpy.data.worlds.new("w"); sc.world = w; w.use_nodes = True
next(n for n in w.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.012, .025, .03, 1)
cam = bpy.data.cameras.new("c"); cam.lens = 80; cam.clip_end = 500
co = bpy.data.objects.new("c", cam); sc.collection.objects.link(co)
t = Vector((0, 1, 2)); co.location = t + Vector((12, -58, 32))
co.rotation_euler = (t - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
