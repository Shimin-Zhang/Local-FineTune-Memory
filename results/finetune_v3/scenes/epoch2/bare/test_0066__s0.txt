import bpy, bmesh, math, random
from mathutils import Vector
random.seed(518)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
def hue(rgb, em=0, rough=.9):
    m = bpy.data.materials.new("h"); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*rgb, 1); b.inputs["Roughness"].default_value = rough
    if em: b.inputs["Emission Color"].default_value = (*rgb, 1); b.inputs["Emission Strength"].default_value = em
    return m
bank, causeway, stone, water = hue((.12, .2, .1)), hue((.28, .26, .22)), hue((.34, .35, .38)), hue((.06, .16, .2), rough=.15)
tile, wood, willow, trunk = hue((.1, .13, .17)), hue((.3, .12, .08)), hue((.25, .4, .12)), hue((.2, .14, .09))
lamp, rain, path = hue((1, .65, .25), 14), hue((.6, .75, .9), 1.2), hue((.2, .2, .18))
def make(bm, m, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("p"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("p", me); o.location, o.rotation_euler = at, rot
    sc.collection.objects.link(o); me.materials.append(m); return o
def blk(at, s, m, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * s[0], v.co.y * s[1], (v.co.z + .5) * s[2]))
    return make(bm, m, at, rot)
def cyl(at, r, h, n, m, r2=None, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return make(bm, m, at, rot)
def line(a, b, t, m):
    a, b = Vector(a), Vector(b); d = b - a; q = d.to_track_quat('Z', 'Y')
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = q @ Vector((v.co.x * t, v.co.y * t, (v.co.z + .5) * d.length)) + a
    return make(bm, m)
def glimmer(at, c, w):
    L = bpy.data.lights.new("g", "POINT"); L.color, L.energy = c, w
    o = bpy.data.objects.new("g", L); o.location = at; sc.collection.objects.link(o)
# lake slab with a rough underside
bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
for v in bm.verts:
    if v.co.z < 0: v.co.x *= .8; v.co.y *= .8; v.co += Vector((random.uniform(-.5, .5), random.uniform(-.5, .5), random.uniform(-1.5, 0)))
    v.co = Vector((v.co.x * 30, v.co.y * 20, v.co.z * 3 - 1.5))
make(bm, water)
# both banks
for sx in (-1, 1):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts:
        v.co = Vector((v.co.x * 11, v.co.y * 19, (v.co.z + .5) * 1.2))
        if v.co.z > .5: v.co.z += random.uniform(-.1, .15)
    make(bm, bank, (sx * 9.5, 0, 0))
# raised causeway on piers, with a parapet, running x from -4.5 to 4.5
blk((0, 0, .8), (10, 3.2, .5), causeway)
for i in range(7): blk((-4.5 + i * 1.5, 0, 0), (.5, 2.6, .85), stone)
for sy in (-1.5, 1.5):
    blk((0, sy, 1.3), (10, .25, .45), stone)
    for i in range(11): blk((-5 + i, sy, 1.3), (.3, .3, .6), stone)
# stone arched bridge over the causeway
for i in range(7):
    a = math.pi * i / 6; x = 1.7 * math.cos(a); z = 1.1 * math.sin(a)
    blk((x, 0, z), (.65, 3.4, .5), stone, (0, a - math.pi / 2, 0))
for sy in (-1.7, 1.7): blk((0, sy, 1.5), (4.2, .3, .4), stone)
for sx in (-1.8, 1.8): blk((sx, 0, 2.2), (.5, 3.4, .5), stone)
# roofed pavilion at the middle of the causeway
for sx in (-.9, .9):
    for sy in (-.9, .9): cyl((sx, sy, 1.3), .12, 1.8, 5, wood)
blk((0, 0, 1.3), (2.4, 2.4, .25), wood)
cyl((0, 0, 3.0), 2.4, .5, 4, tile, rot=(0, 0, math.pi / 4)); cyl((0, 0, 3.5), 1.7, .9, 4, tile, .1, (0, 0, math.pi / 4))
blk((0, 0, 3.0), (2.0, 2.0, .1), lamp)
for sy in (-1, 1): blk((0, sy * 1.1, 1.6), (1.4, .1, .6), wood)
glimmer((0, 0, 2.6), (1, .65, .25), 600)
# willows along the causeway and the banks
def willow(x, y, s):
    cyl((x, y, 0), .22 * s, 2.2 * s, 5, trunk, .15 * s)
    for k in range(7):
        a = k * .9; line((x + .3 * math.cos(a), y + .3 * math.sin(a), 2.1 * s), (x + 1.6 * math.cos(a), y + 1.6 * math.sin(a), .3), .22 * s, willow)
    line((x, y, 2.2 * s), (x + .2, y, 3.0 * s), .35 * s, willow)
for i in range(6): willow(-4.2 + i * 1.7, 2.6 + (i % 2) * .3, 1.0)
for i in range(5): willow(-3.5 + i * 1.8, -2.7 - (i % 2) * .3, .9)
for i in range(4): willow(-11 + i * 1.6, 6 + (i % 2) * 1.5, 1.3); willow(-10 + i * 1.8, -6 - (i % 2) * 1.5, 1.2)
for i in range(4): willow(10.5 - (i % 2) * .8, 5 + i * 1.5, 1.3); willow(10.5, -6 - i * 1.4, 1.2)
# a bank with a path and a stone lantern, no pagoda
blk((9.5, -8, 1.2), (4, 4, .5), path)
cyl((9.5, -8, 1.7), .5, .8, 6, stone, .4); cyl((9.5, -8, 2.5), .55, .5, 6, lamp, .45); cyl((9.5, -8, 3.0), .9, .35, 6, stone, .1)
glimmer((9.5, -8, 2.8), (1, .65, .25), 350)
# light rain, slanting
for i in range(70):
    x, y, z = random.uniform(-13, 13), random.uniform(-9, 9), random.uniform(1, 11)
    line((x, y, z), (x + .4, y + .15, z - 1.3), .04, rain)
# grey dawn: cold key from the left, faint warm rim
for c, w, p in (((.7, .8, 1), 4.5, (-26, -12, 9)), ((1, .6, .4), 1.8, (20, 20, 8))):
    L = bpy.data.lights.new("s", "SUN"); L.color, L.energy = c, w
    o = bpy.data.objects.new("s", L); o.location = p; sc.collection.objects.link(o)
    o.rotation_euler = (Vector((0, 0, 1)) - o.location).to_track_quat('-Z', 'Y').to_euler()
wd = bpy.data.worlds.new("w"); sc.world = wd; wd.use_nodes = True
next(n for n in wd.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.02, .03, .04, 1)
cm = bpy.data.cameras.new("c"); cm.lens = 80; cm.clip_end = 400
co = bpy.data.objects.new("c", cm); sc.collection.objects.link(co)
tg = Vector((0, 0, 1.5)); co.location = tg + Vector((-16, -58, 36))
co.rotation_euler = (tg - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
