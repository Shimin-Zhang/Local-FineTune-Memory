import bpy, bmesh, math, random
from mathutils import Vector
random.seed(1897)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)

def dye(name, rgb, glow=0):
    m = bpy.data.materials.new(name); m.use_nodes = True
    n = next(x for x in m.node_tree.nodes if x.type == "BSDF_PRINCIPLED")
    n.inputs["Base Color"].default_value = (*rgb, 1); n.inputs["Roughness"].default_value = .6
    if glow:
        n.inputs["Emission Color"].default_value = (*rgb, 1); n.inputs["Emission Strength"].default_value = glow
    mats[name] = m
def solid(bm, name, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new(name); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new(name, me); o.location = at; o.rotation_euler = rot
    sc.collection.objects.link(o); o.data.materials.append(mats[name]); return o
def bar(name, at, size, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * size[0], v.co.y * size[1], v.co.z * size[2]))
    return solid(bm, name, at, rot)
def rod(name, a, b, r, n=6, r2=None):
    a, b = Vector(a), Vector(b); d = b - a; q = d.to_track_quat('Z', 'Y')
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=d.length)
    for v in bm.verts: v.co = q @ Vector((v.co.x, v.co.y, v.co.z + d.length / 2)) + a
    return solid(bm, name)
def orb(name, at, r, s=(1, 1, 1)):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co = Vector((v.co.x * s[0], v.co.y * s[1], v.co.z * s[2]))
    return solid(bm, name, at)
def fin(name, at, size, rot=(0, 0, 0)):
    bm = bmesh.new(); f = bm.faces.new([bm.verts.new(p) for p in size])
    r = bmesh.ops.extrude_face_region(bm, geom=[f])
    bmesh.ops.translate(bm, vec=(0, 0, .2), verts=[g for g in r["geom"] if isinstance(g, bmesh.types.BMVert)])
    return solid(bm, name, at, rot)
def lamp(kind, rgb, w, at, aim=None):
    d = bpy.data.lights.new(kind, kind); d.color = rgb; d.energy = w
    o = bpy.data.objects.new(kind, d); o.location = at; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
def shark(x, y, z, yaw, scale, name):
    c, s = math.cos(yaw), math.sin(yaw)
    def at(dx, dy, dz): return (x + dx * c - dy * s, y + dx * s + dy * c, z + dz)
    body = bmesh.new(); bmesh.ops.create_icosphere(body, subdivisions=2, radius=1)
    for v in body.verts:
        v.co = Vector((v.co.x * 4.6 * scale, v.co.y * 1.0 * scale, v.co.z * .95 * scale))
        if v.co.x > 2: v.co.z *= .55; v.co.y *= .7
        if v.co.x < -2: v.co.y *= .15
    solid(body, name, at(0, 0, 0), (0, 0, yaw))
    fin(name, at(.2, 0, .85 * scale), [(-1.0 * scale, 0, 0), (1.0 * scale, 0, 0), (0, 0, 1.5 * scale)], (0, 0, yaw))
    fin(name, at(-3.2 * scale, 0, 0), [(-.9 * scale, 0, 0), (.9 * scale, 0, 0), (0, 0, 1.4 * scale)], (0, 0, yaw))
    for sd in (-1, 1):
        fin(name, at(-2.5 * scale, sd * .5 * scale, -.1 * scale), [(-.5 * scale, 0, 0), (.5 * scale, 0, 0), (0, sd * 1.1 * scale, 0)], (0, 0, yaw))
        orb("eye", at(3.4 * scale, sd * .55 * scale, .35 * scale), .17 * scale)
    fin("tooth", at(3.6 * scale, 0, -.5 * scale), [(-.5 * scale, 0, 0), (.5 * scale, 0, 0), (0, 0, -.35 * scale)], (0, 0, yaw))
