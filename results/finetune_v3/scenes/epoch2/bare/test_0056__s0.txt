import bpy, bmesh, math, random
from mathutils import Vector
random.seed(454)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
def pigment(rgb, glow=0):
    m = bpy.data.materials.new("p"); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*rgb, 1); b.inputs["Roughness"].default_value = .9
    if glow: b.inputs["Emission Color"].default_value = (*rgb, 1); b.inputs["Emission Strength"].default_value = glow
    return m
ash, ash2, blackstone, iron, iron2, rail, post, ember, steam, wood, sign, door = (pigment(c) for c in (
    (.16, .15, .16), (.1, .1, .11), (.07, .07, .08), (.3, .32, .35), (.2, .22, .25), (.2, .12, .08), (.12, .08, .05), (.1, .03, .02), (.35, .33, .35), (.25, .17, .1), (.7, .65, .5), (.1, .06, .04)))
lava = pigment((1, .3, .04), 14); lamp = pigment((1, .6, .2), 16)
def mesh_obj(bm, m, loc=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("m"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("m", me); o.location = loc; o.rotation_euler = rot
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def slab(m, x, y, z, w, d, h, rz=0):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * w, v.co.y * d, (v.co.z + .5) * h))
    return mesh_obj(bm, m, (x, y, z), (0, 0, rz))
def cone(m, x, y, z, r1, r2, h, n=8, jit=0):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r1, radius2=r2, depth=h)
    for v in bm.verts:
        v.co.z += h / 2
        if jit: v.co += Vector((random.uniform(-jit, jit), random.uniform(-jit, jit), random.uniform(-jit, jit)))
    return mesh_obj(bm, m, (x, y, z))
def puff(m, x, y, z, r, sq=.7):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co.z *= sq; v.co *= random.uniform(.85, 1.15)
    return mesh_obj(bm, m, (x, y, z))
def glowpt(col, e, x, y, z):
    L = bpy.data.lights.new("g", "POINT"); L.color = col; L.energy = e
    o = bpy.data.objects.new("g", L); o.location = (x, y, z); sc.collection.objects.link(o)
# ash plain, rough underside
bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=11, radius1=6, radius2=15, depth=4)
for v in bm.verts:
    if v.co.z < 0: v.co += Vector((random.uniform(-1.5, 1.5), random.uniform(-1.5, 1.5), random.uniform(-1.5, .3)))
    else: v.co.z += random.uniform(-.15, .15)
    v.co.z -= 2
mesh_obj(bm, ash)
for i in range(10): puff(ash2, random.uniform(-11, 11), random.uniform(-11, 4), .05, random.uniform(1, 2.2), .12)
# the volcano: stepped cone with a crater of glowing lava and steam
cone(blackstone, 0, 8, 0, 8, 4.2, 8, 9, .4); cone(blackstone, 0, 8, 8, 4.4, 3.2, 2.4, 9, .25)
cone(lava, 0, 8, 10.3, 2.5, 2.2, .25, 9)
for i in range(5): puff(ember, math.cos(i * 1.3) * 2.7, 8 + math.sin(i * 1.3) * 2.7, 10.2, .5, .6)
for i in range(7): puff(steam, math.sin(i * 2) * 1.2, 8 + math.cos(i * 2) * .8, 11.4 + i * 1.2, 1.0 + i * .22, .8)
glowpt((1, .35, .08), 2500, 0, 6.5, 11.5)
# the rim hut: black stone walls, corrugated iron roof with ridges, door and window
hx, hy = -4.5, -2.5
slab(blackstone, hx, hy, 0, 7, 5, 2.8)
for i in range(6): slab(ash2, hx - 3 + i * 1.2, hy - 2.52, .5, .1, .1, .1)
for i in range(14):
    slab(iron if i % 2 else iron2, hx - 3.2 + i * .5, hy, 2.8, .5, 6.2, .12).rotation_euler.x = .0
slab(iron2, hx, hy, 3.0, 7, .5, .25)
slab(door, hx + 1.5, hy - 2.52, 0, 1.1, .1, 2.0); slab(lamp, hx - 1.5, hy - 2.52, 1.0, 1.2, .1, 1.0)
glowpt((1, .6, .2), 350, hx - 1.5, hy - 3.5, 1.4)
for i in range(4): slab(iron2, hx - 3 + i * 2.2, hy - 2.7, 2.4, .15, .15, .5)
slab(iron, hx + 2.5, hy + 1.5, 3.0, .6, .6, 1.6)
# marker pole with a flag and a sign board
slab(post, 3.5, -5, 0, .25, .25, 4.2); slab(sign, 3.5, -5, 3.4, 2.2, .12, 1.0); slab(lamp, 3.5, -5.1, 3.7, 1.4, .05, .2)
glowpt((1, .6, .2), 120, 3.5, -6, 3.6)
# railed path: posts, rails and steps from the front up to the hut
for i in range(12):
    t = i / 11; px = -10 + t * 10; py = -9 + t * 5
    for s in (-1, 1):
        slab(post, px + s * .8, py + s * .3, 0, .15, .15, 1.0)
        if i < 11:
            q = (i + 1) / 11; slab(rail, px + .8 * s + .45, py + .3 * s + .225, .8, 1.0, .06, .06)
for i in range(6): slab(ash2, -4.5 + (i % 2) * .3, -5.5 + i * .5, 0, 2.2, .5, .12 + i * .0)
# boulders and a few tufts of grey scrub
for i in range(14):
    a = random.uniform(0, 6.28); r = random.uniform(9, 12.5)
    puff(ash2 if i % 2 else blackstone, math.cos(a) * r * .9, math.sin(a) * r * .6 - 1, .1, random.uniform(.4, 1.0), .7)
# harsh volcanic dusk: low red-orange key from the left, cold slate rim from behind
for col, e, p in (((1, .45, .2), 5, (-26, -14, 9)), ((.3, .4, .8), 2, (14, 24, 12))):
    L = bpy.data.lights.new("s", "SUN"); L.color = col; L.energy = e
    o = bpy.data.objects.new("s", L); o.location = p; sc.collection.objects.link(o)
    o.rotation_euler = (Vector((0, 0, 2)) - o.location).to_track_quat('-Z', 'Y').to_euler()
w = bpy.data.worlds.new("w"); sc.world = w; w.use_nodes = True
next(n for n in w.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.03, .015, .02, 1)
cam = bpy.data.cameras.new("c"); cam.lens = 80; cam.clip_end = 500
co = bpy.data.objects.new("c", cam); sc.collection.objects.link(co)
co.location = (8, -58, 36)
co.rotation_euler = (Vector((0, 3, 4)) - co.location).to_track_quat('-Z', 'Y').to_euler()
sc.camera = co
