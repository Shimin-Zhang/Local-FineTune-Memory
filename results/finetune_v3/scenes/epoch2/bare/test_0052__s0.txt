def hue(c, e=0):
    m = bpy.data.materials.new("h"); m.use_nodes = True
    n = next(x for x in m.node_tree.nodes if x.type == "BSDF_PRINCIPLED")
    n.inputs["Base Color"].default_value = (*c, 1); n.inputs["Roughness"].default_value = .8
    if e: n.inputs["Emission Color"].default_value = (*c, 1); n.inputs["Emission Strength"].default_value = e
    return m
board, under, line = hue((.12, .1, .2)), hue((.05, .04, .1)), hue((.9, .85, .7), .3)
reds = [hue(c) for c in ((.5, .12, .1), (.55, .3, .1), (.15, .3, .45), (.5, .4, .15), (.35, .15, .4), (.15, .4, .25))]
roofc = [hue(c) for c in ((.15, .1, .25), (.4, .12, .12), (.1, .25, .2))]
glass, lamp, grass, tree = hue((1, .7, .25), 7), hue((1, .85, .5), 14), hue((.1, .2, .1)), hue((.08, .22, .12))
def add(bm, m, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("j"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("j", me); o.location, o.rotation_euler = at, rot
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def slab(x, y, z, w, d, h, m, rz=0):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * w, v.co.y * d, (v.co.z + .5) * h))
    return add(bm, m, (x, y, z), (0, 0, rz))
def gable(x, y, z, w, d, h, m, rz=0):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * w, v.co.y * d, (v.co.z + .5) * h))
    for v in bm.verts:
        if v.co.z > 0: v.co.y = 0
    return add(bm, m, (x, y, z), (0, 0, rz))
def cone(x, y, z, r, h, n, m, r2=0):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return add(bm, m, (x, y, z))
def tab(x, y, z, r, m, axis):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co = Vector((v.co.x, v.co.y, v.co.z)) * (1.25, 1.25, 1)
    return add(bm, m, (x, y, z))
def lit(x, y, z, w, c=(1, .7, .3)):
    L = bpy.data.lights.new("l", "POINT"); L.color = c; L.energy = w
    o = bpy.data.objects.new("l", L); o.location = (x, y, z); sc.collection.objects.link(o)

for v in bm.verts:

add(bm, under)

# the church: nave, gable roof, tower with spire, glowing windows and a door

for i, x in enumerate((-7.5, -3.2, 1.2, 5.6, 9.0)):

    slab(x, -5.5, .5, 3.2, 3.4, h, reds[i % 6]); gable(x, -5.5, .5 + h, 3.6, 4.0, 1.5, roofc[i % 3])

w = bpy.data.worlds.new("w"); sc.world = w; w.use_nodes = True
next(n for n in w.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.015, .012, .04, 1)
cam = bpy.data.cameras.new("c"); cam.lens = 80; cam.clip_end = 400
co = bpy.data.objects.new("c", cam); sc.collection.objects.link(co)
