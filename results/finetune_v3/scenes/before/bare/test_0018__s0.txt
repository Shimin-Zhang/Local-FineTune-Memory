def create_camel(x, y, z, rot_z):
    # Body
    bpy.ops.mesh.primitive_cube_add(size=1, location=(x, y, z+1))
    body = bpy.context.active_object
    body.scale = (1.5, 0.8, 0.8)
    body.data.materials.append(mat_camels)
    
    # Hump
    bpy.ops.mesh.primitive_uv_sphere_add(radius=0.6, location=(x, y, z+1.8))
    hump = bpy.context.active_object
    hump.scale = (0.8, 0.6, 1)
    hump.data.materials.append(mat_camels)
    
    # Head/Neck
    bpy.ops.mesh.primitive_cylinder_add(radius=0.2, depth=1.5, location=(x+1.2, y, z+1.5))
    neck = bpy.context.active_object
    neck.rotation_euler = (0, math.radians(30), 0)
    neck.data.materials.append(mat_camels)
    bpy.ops.mesh.primitive_cube_add(size=0.4, location=(x+1.6, y, z+2.2))
    head = bpy.context.active_object
    head.data.materials.append(mat_camels)
    
    # Legs
    for i in range(4):
        lx = x + (0.8 if i < 2 else -0.8)
        ly = y + (0.4 if i % 2 == 0 else -0.4)
        bpy.ops.mesh.primitive_cylinder_add(radius=0.15, depth=1, location=(lx, ly, z+0.5))
        leg = bpy.context.active_object
        leg.data.materials.append(mat_camels)
        
    # Rotate group
    for obj in [body, hump, neck, head]:
        obj.rotation_euler.z = rot_z

# --- Acacias ---
def create_acacia(x, y):
    # Trunk
    bpy.ops.mesh.primitive_cylinder_add(radius=0.2, depth=3, location=(x, y, 1.5))
    trunk = bpy.context.active_object
    trunk.data.materials.append(mat_leather)
    
    # Branches/Canopy
    for i in range(5):
        angle = i * (math.pi * 2 / 5)
        bx = x + math.cos(angle) * 1.5
        by = y + math.sin(angle) * 1.5
        bpy.ops.mesh.primitive_uv_sphere_add(radius=0.8, location=(bx, by, 3.5))
        leaf = bpy.context.active_object
        leaf.scale = (1.5, 1.5, 0.5)
        leaf.data.materials.append(mat_acacia)

create_acacia(-15, -10)
create_acacia(15, -15)
create_acacia(-10, 15)
