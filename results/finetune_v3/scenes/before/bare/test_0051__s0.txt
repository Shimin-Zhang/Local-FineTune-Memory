import bpy
import math
import mathutils
from mathutils import Vector

# Clear the scene
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# Set units
bpy.context.scene.unit_settings.system = 'METRIC'
bpy.context.scene.unit_settings.scale_length = 1.0

# Materials
def create_material(name, color, metallic=0.0, roughness=0.8):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Metallic"].default_value = metallic
    bsdf.inputs["Roughness"].default_value = roughness
    return mat

mat_stone = create_material("Roman Stone", (0.65, 0.60, 0.50), 0.0, 0.9)
mat_wood = create_material("Wood", (0.35, 0.22, 0.12), 0.0, 0.85)
mat_water = create_material("Water", (0.15, 0.35, 0.55), 0.1, 0.2)
mat_brick = create_material("Brick", (0.55, 0.30, 0.20), 0.0, 0.95)
mat_roof = create_material("Tile Roof", (0.45, 0.25, 0.15), 0.0, 0.9)
mat_ground = create_material("Hillside", (0.45, 0.40, 0.25), 0.0, 1.0)
mat_aqueduct = create_material("Aqueduct Stone", (0.55, 0.50, 0.42), 0.0, 0.9)

# Helper: create a box
def create_box(name, size, location, material, rotation=(0,0,0)):
    bpy.ops.mesh.primitive_cube_add(size=1, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    bpy.ops.object.transform_apply(scale=True)
    obj.data.materials.append(material)
    return obj

# Helper: create a cylinder
def create_cylinder(name, radius, depth, location, material, rotation=(0,0,0), vertices=32):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation, vertices=vertices)
    obj = bpy.context.active_object
    obj.name = name
    obj.data.materials.append(material)
    return obj

# Helper: create a water wheel
def create_water_wheel(name, location, radius=2.5, width=0.8, material_wood=mat_wood):
    objects = []
    # Central axle
    axle = create_cylinder(name+"_axle", 0.15, width + 1.0, location, mat_wood, rotation=(math.pi/2, 0, 0))
    objects.append(axle)
    
    # Rim (outer ring) - use a torus-like approximation with cylinders
    rim_segments = 16
    for i in range(rim_segments):
        angle = i * (2 * math.pi / rim_segments)
        x = location[0] + radius * math.cos(angle)
        y = location[1] + radius * math.sin(angle)
        z = location[2]
        # Small box for rim segment
        seg = create_box(f"{name}_rim_{i}", (0.3, 0.15, width), (x, y, z), mat_wood, 
                         rotation=(0, 0, -angle))
        objects.append(seg)
    
    # Spokes
    spoke_count = 8
    for i in range(spoke_count):
        angle = i * (2 * math.pi / spoke_count)
        # Spoke from center to rim
        mid_x = location[0] + (radius/2) * math.cos(angle)
        mid_y = location[1] + (radius/2) * math.sin(angle)
        mid_z = location[2]
        spoke = create_box(f"{name}_spoke_{i}", (radius, 0.12, 0.12), (mid_x, mid_y, mid_z), mat_wood,
                           rotation=(0, 0, -angle))
        objects.append(spoke)
    
    # Paddles (buckets) on the outer rim
    paddle_count = 16
    for i in range(paddle_count):
        angle = i * (2 * math.pi / paddle_count)
        x = location[0] + (radius + 0.1) * math.cos(angle)
        y = location[1] + (radius + 0.1) * math.sin(angle)
        z = location[2]
        paddle = create_box(f"{name}_paddle_{i}", (0.2, 0.5, width - 0.1), (x, y, z), mat_wood,
                            rotation=(0, 0, -angle))
        objects.append(paddle)
    
    # Join all wheel parts
    bpy.ops.object.select_all(action='DESELECT')
    for obj in objects:
        obj.select_set(True)
    bpy.context.view_layer.objects.active = axle
    bpy.ops.object.join()
    bpy.context.active_object.name = name
    return bpy.context.active_object

bpy.ops.mesh.primitive_plane_add(size=1, location=(0, 0, 0))
hill = bpy.context.active_object
hill.name = "Hillside"
hill.scale = (60, 40, 1)
bpy.ops.object.transform_apply(scale=True)
# Tilt the plane to create slope (rotate around Y axis)
hill.rotation_euler = (0, math.radians(12), 0)
hill.data.materials.append(mat_ground)

# Adjust hill position
hill.location = (0, 0, -5)

# Aqueduct (elevated water channel) running along the top
# Aqueduct runs along X axis at the top of the slope
aqueduct_y = 15
aqueduct_z = 12
aqueduct_length = 50
aqueduct_width = 1.5
aqueduct_height = 1.0

# Aqueduct body
aqueduct_body = create_box("Aqueduct_Body", (aqueduct_length, aqueduct_width, aqueduct_height),
                           (0, aqueduct_y, aqueduct_z), mat_aqueduct)

# Aqueduct water channel (inner)
aqueduct_water = create_box("Aqueduct_Water", (aqueduct_length - 0.2, aqueduct_width - 0.4, 0.3),
                            (0, aqueduct_y, aqueduct_z + 0.4), mat_water)

# Aqueduct pillars
pillar_spacing = 5
num_pillars = int(aqueduct_length / pillar_spacing) + 1
for i in range(num_pillars):
    x = -aqueduct_length/2 + i * pillar_spacing
    pillar = create_box(f"Aqueduct_Pillar_{i}", (0.6, 0.6, aqueduct_z - 0.5),
                        (x, aqueduct_y, (aqueduct_z - 0.5)/2), mat_aqueduct)

# Two parallel rows of 8 water wheels stepped down the slope
# Row 1: y = 5, Row 2: y = -5
# Wheels step down along X axis
wheel_radius = 2.5
wheel_width = 0.8
wheels_per_row = 8
x_start = -20
x_spacing = 5

all_wheels = []
for row in range(2):
    y_pos = 5 if row == 0 else -5
    for i in range(wheels_per_row):
        x_pos = x_start + i * x_spacing
        # Step down: each wheel is lower than the previous
        z_pos = 8 - i * 1.2  # Stepping down the slope
        
        # Support structure for the wheel
        support_left = create_box(f"Support_L_{row}_{i}", (0.3, 0.3, z_pos),
                                   (x_pos - wheel_radius - 0.5, y_pos, z_pos/2), mat_stone)
        support_right = create_box(f"Support_R_{row}_{i}", (0.3, 0.3, z_pos),
                                    (x_pos + wheel_radius + 0.5, y_pos, z_pos/2), mat_stone)
        
        # Base platform
        platform = create_box(f"Platform_{row}_{i}", (wheel_radius*2 + 2, 3, 0.3),
                               (x_pos, y_pos, 0.15), mat_stone)
        
        # Water wheel
        wheel = create_water_wheel(f"WaterWheel_{row}_{i}", (x_pos, y_pos, z_pos), 
                                    radius=wheel_radius, width=wheel_width)
        all_wheels.append(wheel)
        
        # Water channel leading to each wheel
        if i < wheels_per_row - 1:
            channel = create_box(f"Channel_{row}_{i}", (x_spacing, 0.8, 0.5),
                                  (x_pos + x_spacing/2, y_pos, z_pos - wheel_radius + 0.2), mat_water)
        else:
            # Last wheel in row - channel to next row or to bakery
            pass

# Connect the two rows with cross channels
for i in range(wheels_per_row):
    x_pos = x_start + i * x_spacing
    z_pos = 8 - i * 1.2
    cross_channel = create_box(f"CrossChannel_{i}", (0.8, 10, 0.5),
                                (x_pos, 0, z_pos - wheel_radius + 0.2), mat_water)

# Bakery building at the end (bottom of the slope)
bakery_x = 22
bakery_y = 0
bakery_z = 0

# Bakery main structure
bakery_body = create_box("Bakery_Body", (12, 10, 6), (bakery_x, bakery_y, 3), mat_brick)

# Bakery roof (pyramid-like)
bpy.ops.mesh.primitive_cone_add(radius1=9, radius2=0, depth=3, location=(bakery_x, bakery_y, 6+1.5), vertices=4)
roof = bpy.context.active_object
roof.name = "Bakery_Roof"
roof.rotation_euler = (0, 0, math.radians(45))
roof.scale = (1.0, 0.8, 1.0)
bpy.ops.object.transform_apply(scale=True)
roof.data.materials.append(mat_roof)

# Bakery entrance
entrance = create_box("Bakery_Entrance", (3, 0.2, 4), (bakery_x - 6.1, bakery_y, 2), mat_stone)

# Bakery chimney
chimney = create_box("Bakery_Chimney", (1.5, 1.5, 3), (bakery_x + 3, bakery_y + 3, 6+1.5), mat_brick)

# Water channel from last wheels to bakery
for row in range(2):
    y_pos = 5 if row == 0 else -5
    last_x = x_start + (wheels_per_row - 1) * x_spacing
    last_z = 8 - (wheels_per_row - 1) * 1.2
    channel_to_bakery = create_box(f"ChannelToBakery_{row}", (bakery_x - last_x - wheel_radius, 0.8, 0.5),
                                    ((last_x + bakery_x - wheel_radius)/2, y_pos, last_z - wheel_radius + 0.2), mat_water)

# Add some terrain details - small rocks
import random

                          (x, y, z), mat_ground)

# Lighting
# Sun light
bpy.ops.object.light_add(type='SUN', location=(20, -30, 40))
sun = bpy.context.active_object
sun.name = "Sun"

# World background
world = bpy.context.scene.world
if world is None:

bpy.ops.object.camera_add(location=(35, -35, 25))
