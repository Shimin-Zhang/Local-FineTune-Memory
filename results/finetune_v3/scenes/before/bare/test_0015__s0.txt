import bpy
import math
import mathutils
import random

# --- Clean Scene ---
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete()
for collection in [bpy.data.meshes, bpy.data.materials, bpy.data.lights, bpy.data.cameras, bpy.data.worlds]:
    for item in collection:
        if item.users == 0:
            collection.remove(item)

# --- Materials ---
def make_mat(name, color, roughness=0.8, metallic=0.0, emit=None, emit_strength=0.0):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get("Principled BSDF")
    bsdf.inputs["Base Color"].default_value = (*color, 1.0)
    bsdf.inputs["Roughness"].default_value = roughness
    bsdf.inputs["Metallic"].default_value = metallic
    if emit is not None:
        bsdf.inputs["Emission Color"].default_value = (*emit, 1.0)
        bsdf.inputs["Emission Strength"].default_value = emit_strength
    return mat

mat_white   = make_mat("White",   (0.95, 0.95, 0.93), 0.6)
mat_wall    = make_mat("Wall",    (0.75, 0.65, 0.50), 0.9)
mat_floor   = make_mat("Floor",   (0.55, 0.45, 0.35), 0.95)
mat_dome    = make_mat("Dome",    (0.92, 0.92, 0.90), 0.4)
mat_cross   = make_mat("Cross",   (0.85, 0.85, 0.83), 0.3)
mat_palm_trunk = make_mat("PalmTrunk", (0.45, 0.35, 0.25), 0.95)
mat_palm_leaf  = make_mat("PalmLeaf",  (0.30, 0.50, 0.20), 0.8)
mat_bell_frame = make_mat("BellFrame", (0.50, 0.45, 0.40), 0.85)
mat_bell       = make_mat("Bell",    (0.60, 0.50, 0.30), 0.4, metallic=0.7)
mat_ground   = make_mat("Ground",  (0.50, 0.40, 0.30), 0.98)

# --- Helper ---
def add_cube(name, loc, size, mat, rot=(0,0,0)):
    bpy.ops.mesh.primitive_cube_add(size=1, location=loc, rotation=rot)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    obj.data.materials.append(mat)
    return obj

def add_cylinder(name, loc, r, depth, mat, rot=(0,0,0), verts=32):
    bpy.ops.mesh.primitive_cylinder_add(radius=r, depth=depth, location=loc, rotation=rot, vertices=verts)
    obj = bpy.context.active_object
    obj.name = name
    obj.data.materials.append(mat)
    return obj

def add_uv_sphere(name, loc, r, mat, segments=32, rings=16):
    bpy.ops.mesh.primitive_uv_sphere_add(radius=r, location=loc, segments=segments, rings=rings)
    obj = bpy.context.active_object
    obj.name = name
    obj.data.materials.append(mat)
    return obj

# Drum

        bm.verts.remove(v)
bm.to_mesh(me)
bm.free()

add_cylinder("CrossHoriz", (0, 0, 8.55), 0.08, 0.5, mat_cross, rot=(0, math.radians(90), 0), verts=8)

# --- Bell Frame (small, on west side) ---
# Two posts

# --- Palm Trees ---
def add_palm(loc, height=6, lean=0.1):
    # Trunk
    trunk = add_cylinder("PalmTrunk", (loc[0], loc[1], height/2), 0.15, height, mat_palm_trunk, verts=12)
    # Slight lean
    trunk.rotation_euler = (lean, 0, random.uniform(0, 2*math.pi))
    # Leaves
    top = mathutils.Vector((loc[0] + math.sin(lean)*height*0.5, loc[1], height))
    for i in range(8):
        angle = i * (2*math.pi/8) + random.uniform(-0.2, 0.2)
        bpy.ops.mesh.primitive_plane_add(size=1, location=(
            top.x + math.cos(angle)*1.2,
            top.y + math.sin(angle)*1.2,
            top.z - 0.3
        ))
        leaf = bpy.context.active_object
        leaf.name = f"PalmLeaf_{i}"
        leaf.scale = (1.8, 0.4, 1)
        leaf.rotation_euler = (
            random.uniform(0.3, 0.8),
            angle,
            random.uniform(0, 2*math.pi)
        )
        leaf.data.materials.append(mat_palm_leaf)

# Place palms around forecourt and near church
