def create_material(name, base_color, roughness=0.5, metallic=0.0, emission=None, emission_strength=0.0):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    links = mat.node_tree.links
    nodes.clear()
    
    output = nodes.new('ShaderNodeOutputMaterial')
    output.location = (400, 0)
    
    bsdf = nodes.new('ShaderNodeBsdfPrincipled')
    bsdf.location = (0, 0)
    bsdf.inputs['Base Color'].default_value = base_color
    bsdf.inputs['Roughness'].default_value = roughness
    bsdf.inputs['Metallic'].default_value = metallic
    
    if emission is not None:
        bsdf.inputs['Emission Color'].default_value = emission
        bsdf.inputs['Emission Strength'].default_value = emission_strength
    
    links.new(bsdf.outputs['BSDF'], output.inputs['Surface'])
    return mat

def create_chandelier(x, y, z):
    # Main frame
    bpy.ops.mesh.primitive_torus_add(major_radius=1.5, minor_radius=0.05, location=(x, y, z), major_segments=16, minor_segments=8)
    torus = bpy.context.active_object
    torus.name = "ChandelierFrame"
    torus.data.materials.append(lantern_mat)
    
    # Central stem
    bpy.ops.mesh.primitive_cylinder_add(radius=0.05, depth=2, location=(x, y, z + 1), vertices=8)
    stem = bpy.context.active_object
    stem.name = "ChandelierStem"
    stem.data.materials.append(lantern_mat)
    
    # Crystal pendants
    for j in range(12):
        angle = j * (2 * math.pi / 12)
        cx = x + 1.5 * math.cos(angle)
        cz = z
        bpy.ops.mesh.primitive_cone_add(radius1=0.15, radius2=0, depth=0.5, location=(cx, y, cz - 0.3), vertices=6)
        crystal = bpy.context.active_object
        crystal.name = f"Crystal_{j}"
        crystal.rotation_euler = (math.radians(180), 0, 0)
        crystal.data.materials.append(crystal_mat)
    
    # Crystal cluster at center
    bpy.ops.mesh.primitive_ico_sphere_add(radius=0.4, location=(x, y, z), subdivisions=1)
    center_crystal = bpy.context.active_object
    center_crystal.name = "CenterCrystal"
    center_crystal.data.materials.append(crystal_mat)
    
    # Point light for glow
    light_data = bpy.data.lights.new(f"ChandelierLight_{x}_{y}", type='POINT')
    light_data.energy = 100
    light_data.color = (0.7, 0.8, 1.0)
    light_data.shadow_soft_size = 0.3
    light_obj = bpy.data.objects.new(f"ChandelierLight_{x}_{y}", light_data)
    scene.collection.objects.link(light_obj)
    light_obj.location = (x, y, z - 0.5)
