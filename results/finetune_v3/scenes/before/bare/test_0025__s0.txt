def make_mat(name, color, rough=0.7, metal=0.0):
    m = bpy.data.materials.new(name)
    m.use_nodes = True
    bsdf = m.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Roughness"].default_value = rough
    bsdf.inputs["Metallic"].default_value = metal
    return m

# ---------------- Helpers ----------------
def add_box(name, loc, size, mat, rot=(0,0,0)):
    bpy.ops.mesh.primitive_cube_add(size=1, location=loc, rotation=rot)
    obj = bpy.context.object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.data.materials.append(mat)
    return obj

def add_cyl(name, loc, r, depth, mat, rot=(0,0,0), verts=32):
    bpy.ops.mesh.primitive_cylinder_add(radius=r, depth=depth, location=loc, rotation=rot, vertices=verts)
    obj = bpy.context.object
    obj.name = name
    obj.data.materials.append(mat)
    return obj

def add_sphere(name, loc, r, mat, segs=24):
    bpy.ops.mesh.primitive_uv_sphere_add(radius=r, location=loc, segments=segs, ring_count=segs//2)
    obj = bpy.context.object
    obj.name = name
    obj.data.materials.append(mat)
    return obj

def make_folding_chair(name, loc, rot_z=0):
    """Create a simple folding chair at loc, rotated by rot_z."""
    chair_parts = []
    # Seat
    seat = add_box(f"{name}_Seat", (0, 0, 0.45), (0.5, 0.5, 0.05), mat_leather)
    chair_parts.append(seat)
    # Backrest
    back = add_box(f"{name}_Back", (0, -0.22, 0.8), (0.5, 0.05, 0.6), mat_leather)
    chair_parts.append(back)
    # Legs (X-frame)
    for sx in [-0.2, 0.2]:
        for sy in [-0.2, 0.2]:
            leg = add_cyl(f"{name}_Leg_{sx}_{sy}", (sx, sy, 0.22), 0.03, 0.45, mat_wood, rot=(math.radians(15), math.radians(15), 0))
            chair_parts.append(leg)
    # Cross brace
    brace = add_cyl(f"{name}_Brace", (0, 0, 0.25), 0.02, 0.7, mat_wood, rot=(0, math.radians(90), 0))
    chair_parts.append(brace)

    # Parent all to an empty and rotate
    bpy.ops.object.empty_add(location=loc)
    empty = bpy.context.object
    empty.name = name
    empty.rotation_euler = (0, 0, rot_z)
    for p in chair_parts:
        p.parent = empty
        p.matrix_parent_inverse = empty.matrix_world.inverted()

    return empty

def make_bathtub(name, loc):
    parts = []
    # Tub body (elongated box with rounded ends approximation)
    tub = add_box(f"{name}_Body", (0, 0, 0.5), (2.2, 0.9, 0.8), mat_paper)
    parts.append(tub)
    # Water surface
    water = add_box(f"{name}_Water", (0, 0, 0.9), (2.0, 0.75, 0.05), mat_water)
    parts.append(water)
    # Claw feet
    for fx in [-0.9, 0.9]:
        for fy in [-0.35, 0.35]:
            foot = add_cyl(f"{name}_Foot_{fx}_{fy}", (fx, fy, 0.1), 0.08, 0.2, mat_brass)
            parts.append(foot)
            # Claw tip
            tip = add_sphere(f"{name}_Claw_{fx}_{fy}", (fx, fy, 0.05), 0.1, mat_brass)
            parts.append(tip)
    # Faucet
    faucet = add_cyl(f"{name}_Faucet", (1.1, 0, 0.9), 0.04, 0.4, mat_brass)
    parts.append(faucet)
    faucet_head = add_cyl(f"{name}_FaucetHead", (1.1, 0, 1.1), 0.06, 0.15, mat_brass, rot=(math.radians(90), 0, 0))
    parts.append(faucet_head)

    bpy.ops.object.empty_add(location=loc)
    empty = bpy.context.object
    empty.name = name
    for p in parts:
        p.parent = empty
        p.matrix_parent_inverse = empty.matrix_world.inverted()
    return empty

def make_gramophone(name, loc):
    parts = []
    # Base box
    base = add_box(f"{name}_Base", (0, 0, 0.15), (0.6, 0.5, 0.3), mat_wood_dk)
    parts.append(base)
    # Turntable
    turntable = add_cyl(f"{name}_Turntable", (0, 0, 0.32), 0.25, 0.03, mat_metal)
    parts.append(turntable)
    # Record
    record = add_cyl(f"{name}_Record", (0, 0, 0.35), 0.2, 0.01, mat_wood_dk)
    parts.append(record)
    # Horn (cone approximation using stacked cylinders)
    horn_base = add_cyl(f"{name}_HornBase", (0.3, 0, 0.4), 0.05, 0.3, mat_brass, rot=(0, math.radians(45), 0))
    parts.append(horn_base)
    # Horn cone - use a cone
    bpy.ops.mesh.primitive_cone_add(radius1=0.05, radius2=0.25, depth=0.6, location=(0.5, 0, 0.65), rotation=(0, math.radians(45), 0), vertices=32)
    horn_cone = bpy.context.object
    horn_cone.name = f"{name}_HornCone"
    horn_cone.data.materials.append(mat_brass)
    parts.append(horn_cone)
    # Arm
    arm = add_cyl(f"{name}_Arm", (-0.2, 0, 0.4), 0.02, 0.4, mat_brass, rot=(0, math.radians(90), 0))
    parts.append(arm)

    bpy.ops.object.empty_add(location=loc)
    empty = bpy.context.object
    empty.name = name
    for p in parts:
        p.parent = empty
        p.matrix_parent_inverse = empty.matrix_world.inverted()
    return empty

make_gramophone("Gramophone", (-12, -8, 0))
