import bpy

from mathutils import Vector, Euler

# Clean scene
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete()

# Clear collections
for coll in bpy.data.collections:
    if coll.name != "Scene Collection":
        bpy.data.collections.remove(coll)

# Set up world
world = bpy.context.scene.world
if world is None:
    world = bpy.data.worlds.new("World")
    bpy.context.scene.world = world
world.use_nodes = True
bg = world.node_tree.nodes.get("Background")
if bg:
    bg.inputs[0].default_value = (0.9, 0.6, 0.3, 1.0)  # warm evening sky
    bg.inputs[1].default_value = 1.0

bpy.context.scene.collection.objects.link(cam)
bpy.context.scene.camera = cam
cam.location = (12, -15, 4)
cam.rotation_euler = Euler((math.radians(75), 0, math.radians(20)))

# Trestle table
def make_trestle_table(x, y, z, length=2.0, width=0.8, height=0.9):
    # Table top
    top_data = bpy.data.meshes.new("TableTop")
    top = bpy.data.objects.new("TableTop", top_data)
    bpy.context.scene.collection.objects.link(top)
    bpy.context.view_layer.objects.active = top
    top.select_set(True)
    bpy.ops.mesh.primitive_cube_add(size=1, location=(x, y, z + height))
    top = bpy.context.active_object
    top.name = "TableTop"
    top.scale = (length/2, width/2, 0.03)
    mat_wood = bpy.data.materials.new("Wood")
    mat_wood.use_nodes = True
    b = mat_wood.node_tree.nodes.get("Principled BSDF")
    b.inputs[0].default_value = (0.5, 0.3, 0.15, 1.0)
    top.data.materials.append(mat_wood)
    
    # Legs (trestle style - A-frame)
    for lx in [-length/3, length/3]:
        for ly in [-width/3, width/3]:
            leg_data = bpy.data.meshes.new("Leg")
            leg = bpy.data.objects.new("Leg", leg_data)
            bpy.context.scene.collection.objects.link(leg)
            bpy.context.view_layer.objects.active = leg
            leg.select_set(True)
            bpy.ops.mesh.primitive_cylinder_add(radius=0.03, depth=height, location=(x+lx, y+ly, z + height/2))
            leg = bpy.context.active_object
            leg.name = "Leg"
            leg.data.materials.append(mat_wood)

make_trestle_table(0, 0, 0)

# Pilots on bench (simple humanoid figures)
def make_pilot(x, y, z, color=(0.2, 0.2, 0.5)):
    # Body
    body_data = bpy.data.meshes.new("PilotBody")
    body = bpy.data.objects.new("PilotBody", body_data)
    bpy.context.scene.collection.objects.link(body)
    bpy.context.view_layer.objects.active = body
    body.select_set(True)
    bpy.ops.mesh.primitive_cylinder_add(radius=0.12, depth=0.4, location=(x, y, z + 0.65))
    body = bpy.context.active_object
    body.name = "PilotBody"
    mat_pilot = bpy.data.materials.new("PilotCloth")
    mat_pilot.use_nodes = True
    b = mat_pilot.node_tree.nodes.get("Principled BSDF")
    b.inputs[0].default_value = (*color, 1.0)
    body.data.materials.append(mat_pilot)
    
    # Head
    head_data = bpy.data.meshes.new("PilotHead")
    head = bpy.data.objects.new("PilotHead", head_data)
    bpy.context.scene.collection.objects.link(head)
    bpy.context.view_layer.objects.active = head
    head.select_set(True)
    bpy.ops.mesh.primitive_uv_sphere_add(radius=0.08, location=(x, y, z + 0.95))
    head = bpy.context.active_object
    head.name = "PilotHead"
    mat_skin = bpy.data.materials.new("Skin")
    mat_skin.use_nodes = True
    b = mat_skin.node_tree.nodes.get("Principled BSDF")
    b.inputs[0].default_value = (0.8, 0.6, 0.5, 1.0)
    head.data.materials.append(mat_skin)
    
    # Legs
    for lx in [-0.06, 0.06]:
        leg_data = bpy.data.meshes.new("PilotLeg")
        leg = bpy.data.objects.new("PilotLeg", leg_data)
        bpy.context.scene.collection.objects.link(leg)
        bpy.context.view_layer.objects.active = leg
        leg.select_set(True)
        bpy.ops.mesh.primitive_cylinder_add(radius=0.04, depth=0.3, location=(x+lx, y, z + 0.15))
        leg = bpy.context.active_object
        leg.name = "PilotLeg"
        leg.data.materials.append(mat_pilot)

def make_tiny_plane(x, y, z, scale=0.5, color=(0.8, 0.8, 0.9)):
    # Fuselage
    fus_data = bpy.data.meshes.new("PlaneFus")
    fus = bpy.data.objects.new("PlaneFus", fus_data)
    bpy.context.scene.collection.objects.link(fus)
    bpy.context.view_layer.objects.active = fus
    fus.select_set(True)
    bpy.ops.mesh.primitive_cylinder_add(radius=0.05*scale, depth=0.3*scale, location=(x, y, z))
    fus = bpy.context.active_object
    fus.name = "PlaneFus"
    fus.rotation_euler = Euler((0, math.radians(90), 0))
    mat_plane = bpy.data.materials.new("PlaneBody")
    mat_plane.use_nodes = True
    b = mat_plane.node_tree.nodes.get("Principled BSDF")
    b.inputs[0].default_value = (*color, 1.0)
    fus.data.materials.append(mat_plane)
    
    # Wings
    wing_data = bpy.data.meshes.new("PlaneWing")
    wing = bpy.data.objects.new("PlaneWing", wing_data)
    bpy.context.scene.collection.objects.link(wing)
    bpy.context.view_layer.objects.active = wing
    wing.select_set(True)
    bpy.ops.mesh.primitive_cube_add(size=1, location=(x, y, z))
    wing = bpy.context.active_object
    wing.name = "PlaneWing"
    wing.scale = (0.15*scale, 0.02*scale, 0.04*scale)
    wing.data.materials.append(mat_plane)
    
    # Tail
    tail_data = bpy.data.meshes.new("PlaneTail")
    tail = bpy.data.objects.new("PlaneTail", tail_data)
    bpy.context.scene.collection.objects.link(tail)
    bpy.context.view_layer.objects.active = tail
    tail.select_set(True)
    bpy.ops.mesh.primitive_cube_add(size=1, location=(x - 0.15*scale, y, z))
    tail = bpy.context.active_object
    tail.name = "PlaneTail"
    tail.scale = (0.03*scale, 0.06*scale, 0.02*scale)
    tail.data.materials.append(mat_plane)
