```python
import bpy
import math
import bmesh
from mathutils import Vector

# Clear the scene
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# --- MATERIALS ---
def make_mat(name, color, rough=0.8, metal=0.0, emission=0.0, emissive_color=(0,0,0,1)):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    links = mat.node_tree.links
    bsdf = nodes.get("Principled BSDF")
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Roughness"].default_value = rough
    bsdf.inputs["Metallic"].default_value = metal
    if emission > 0:
        bsdf.inputs["Emission Color"].default_value = emissive_color
        bsdf.inputs["Emission Strength"].default_value = emission
    return mat

wood_mat = make_mat("Wood", (0.35, 0.2, 0.1, 1), rough=0.7)
dark_wood = make_mat("DarkWood", (0.2, 0.12, 0.06, 1), rough=0.8)
stone_mat = make_mat("Stone", (0.5, 0.48, 0.45, 1), rough=0.9)
bone_mat = make_mat("Bone", (0.9, 0.85, 0.75, 1), rough=0.6)
glass_mat = make_mat("Glass", (0.7, 0.8, 0.9, 1), rough=0.1, emission=0.5, emissive_color=(0.8, 0.9, 1.0, 1))
metal_mat = make_mat("Metal", (0.4, 0.4, 0.45, 1), rough=0.3, metal=0.9)
skin_mat = make_mat("Skin", (0.8, 0.6, 0.5, 1), rough=0.7)

# --- FLOOR ---
bpy.ops.mesh.primitive_cylinder_add(radius=12, depth=0.3, location=(0, 0, -0.15))
floor = bpy.context.object
floor.name = "Floor"
floor.data.materials.append(stone_mat)

# --- CENTRAL TABLE ---
bpy.ops.mesh.primitive_cylinder_add(radius=1.5, depth=0.8, location=(0, 0, 0.4))
table = bpy.context.object
table.name = "CentralTable"
table.data.materials.append(wood_mat)

# Table top (slightly larger)
bpy.ops.mesh.primitive_cylinder_add(radius=1.8, depth=0.1, location=(0, 0, 0.85))
tabletop = bpy.context.object
tabletop.name = "TableTop"
tabletop.data.materials.append(dark_wood)

# --- SKELTON ON STAND ---
# Stand
bpy.ops.mesh.primitive_cylinder_add(radius=0.15, depth=1.2, location=(0, 0, 1.5))
stand_pole = bpy.context.object
stand_pole.name = "SkeletonStandPole"
stand_pole.data.materials.append(metal_mat)

# Base of stand
bpy.ops.mesh.primitive_cylinder_add(radius=0.5, depth=0.1, location=(0, 0, 0.95))
stand_base = bpy.context.object
stand_base.name = "SkeletonStandBase"
stand_base.data.materials.append(metal_mat)

# Skeleton parts (simplified)
# Skull
bpy.ops.mesh.primitive_uv_sphere_add(radius=0.15, location=(0, 0, 2.2))
skull = bpy.context.object
skull.name = "Skull"
skull.scale = (1, 0.9, 1.1)
skull.data.materials.append(bone_mat)

# Spine (vertical column of small spheres)
for i in range(8):
    z = 1.95 - i * 0.12
    bpy.ops.mesh.primitive_uv_sphere_add(radius=0.06, location=(0, 0, z))
    vert = bpy.context.object
    vert.name = f"SpineVertebra_{i}"
    vert.data.materials.append(bone_mat)

# Ribcage (simplified as a torus-like shape)
bpy.ops.mesh.primitive_torus_add(major_radius=0.25, minor_radius=0.04, location=(0, 0, 1.75))
ribs = bpy.context.object
ribs.name = "Ribcage"
ribs.rotation_euler = (math.radians(90), 0, 0)
ribs.scale = (1, 1.2, 1)
ribs.data.materials.append(bone_mat)

# Pelvis
bpy.ops.mesh.primitive_uv_sphere_add(radius=0.12, location=(0, 0, 1.35))
pelvis = bpy.context.object
pelvis.name = "Pelvis"
pelvis.scale = (1.3, 1, 0.8)
pelvis.data.materials.append(bone_mat)

# Arms (simple cylinders)
for side in [-1, 1]:
    bpy.ops.mesh.primitive_cylinder_add(radius=0.03, depth=0.5, location=(side*0.35, 0, 1.85))
    upper_arm = bpy.context.object
    upper_arm.name = f"UpperArm_{'L' if side>0 else 'R'}"
    upper_arm.rotation_euler = (0, math.radians(20*side), 0)
    upper_arm.data.materials.append(bone_mat)

    bpy.ops.mesh.primitive_cylinder_add(radius=0.025, depth=0.4, location=(side*0.55, 0, 1.55))
    forearm = bpy.context.object
    forearm.name = f"Forearm_{'L' if side>0 else 'R'}"
    forearm.rotation_euler = (0, math.radians(15*side), 0)
    forearm.data.materials.append(bone_mat)

# Legs (simple cylinders)
for side in [-1, 1]:
    bpy.ops.mesh.primitive_cylinder_add(radius=0.04, depth=0.5, location=(side*0.12, 0, 1.05))
    thigh = bpy.context.object
    thigh.name = f"Thigh_{'L' if side>0 else 'R'}"
    thigh.rotation_euler = (0, math.radians(5*side), 0)
    thigh.data.materials.append(bone_mat)

    bpy.ops.mesh.primitive_cylinder_add(radius=0.03, depth=0.45, location=(side*0.15, 0, 0.6))
    shin = bpy.context.object
    shin.name = f"Shin_{'L' if side>0 else 'R'}"
    shin.rotation_euler = (0, math.radians(3*side), 0)
    shin.data.materials.append(bone_mat)

# --- STEEP CIRCULAR WOODEN GALLERIES ---
num_galleries = 4
gallery_spacing = 1.8
gallery_start_z = 2.5
gallery_radius_inner = 4.0
gallery_width = 1.2

for i in range(num_galleries):
    z = gallery_start_z + i * gallery_spacing
    r = gallery_radius_inner + i * 0.3  # slightly wider as we go up

    # Gallery floor (ring)
    bpy.ops.mesh.primitive_torus_add(
        major_radius=r + gallery_width/2,
        minor_radius=gallery_width/2,
        location=(0, 0, z)
    )
    gal = bpy.context.object
    gal.name = f"Gallery_{i}"
    gal.scale = (1, 1, 0.15)  # flatten to make a ring platform
    gal.data.materials.append(wood_mat)

    # Railing (outer edge)
    bpy.ops.mesh.primitive_torus_add(
        major_radius=r + gallery_width,
        minor_radius=0.05,
        location=(0, 0, z + 0.5)
    )
    rail = bpy.context.object
    rail.name = f"Railing_{i}"
    rail.data.materials.append(dark_wood)

    # Support posts (vertical columns from gallery to gallery)
    num_posts = 16
    for j in range(num_posts):
        angle = j * (2 * math.pi / num_posts)
        x = (r + gallery_width/2) * math.cos(angle)
        y = (r + gallery_width/2) * math.sin(angle)
        bpy.ops.mesh.primitive_cylinder_add(radius=0.08, depth=gallery_spacing, location=(x, y, z + gallery_spacing/2))
        post = bpy.context.object
        post.name = f"Post_{i}_{j}"
        post.data.materials.append(wood_mat)

# --- SKYLIGHT ABOVE ---
# Skylight frame
skylight_z = gallery_start_z + num_galleries * gallery_spacing + 0.5
bpy.ops.mesh.primitive_cylinder_add(radius=3, depth=0.2, location=(0, 0, skylight_z))
skylight_frame = bpy.context.object
skylight_frame.name = "SkylightFrame"
skylight_frame.data.materials.append(metal_mat)

# Skylight glass (slightly above)
bpy.ops.mesh.primitive_cylinder_add(radius=2.8, depth=0.1, location=(0, 0, skylight_z + 0.15))
skylight_glass = bpy.context.object
skylight_glass.name = "SkylightGlass"
skylight_glass.data.materials.append(glass_mat)

# Support beams for skylight
for j in range(8):
    angle = j * (2 * math.pi / 8)
    x1 = 3 * math.cos(angle)
    y1 = 3 * math.sin(angle)
    x2 = 2.8 * math.cos(angle)
    y2 = 2.8 * math.sin(angle)
    mid_x = (x1 + x2) / 2
    mid_y = (y1 + y2) / 2
    length = math.sqrt((x2-x1)**2 + (y2-y1)**2)
    bpy.ops.mesh.primitive_cylinder_add(radius=0.05, depth=length, location=(mid_x, mid_y, skylight_z))
    beam = bpy.context.object
    beam.name = f"SkylightBeam_{j}"
    beam.rotation_euler = (0, math.radians(90), -angle)
    beam.data.materials.append(metal_mat)

# --- LIGHTING ---
# Sun light from skylight
bpy.ops.object.light_add(type='SUN', location=(0, 0, skylight_z + 5))
sun = bpy.context.object
sun.name = "SkylightSun"
sun.data.energy = 3
sun.rotation_euler = (0, 0, 0)

# Ambient light
bpy.ops.object.light_add(type='POINT', location=(0, 0, 8))
ambient = bpy.context.object
ambient.name = "AmbientLight"
ambient.data.energy = 50
ambient.data.color = (1, 0.95, 0.9)

# Warm interior lights on galleries
for i in range(num_galleries):
    z = gallery_start_z + i * gallery_spacing + 1
    for j in range(4):
        angle = j * (2 * math.pi / 4) + math.pi/4
        x = (gallery_radius_inner + 0.5) * math.cos(angle)
        y = (gallery_radius_inner + 0.5) * math.sin(angle)
        bpy.ops.object.light_add(type='POINT', location=(x, y, z))
        light = bpy.context.object
        light.name = f"GalleryLight_{i}_{j}"
        light.data.energy = 10
        light.data.color = (1, 0.85, 0.6)

# Camera
bpy.ops.object.camera_add(location=(8, 8, 4))
cam = bpy.context.object
cam.name = "Camera"
# Point camera at center
direction = Vector((0, 0, 1.5)) - cam.location
rot_quat = direction.to_track_quat('-Z', 'Y')
cam.rotation_euler = rot_quat.to_euler()

# Set camera as active
bpy.context.scene.camera = cam

# Set viewport to camera
for area in bpy.context.screen.areas:
    if area.type == 'VIEW_3D':
        for space in area.spaces:
            if space.type == 'VIEW_3D':
                space.region_3d.view_perspective = 'CAMERA'

print("Historic anatomical theatre created successfully!")
```
