# Helper: create a mesh object from vertices/faces
def make_mesh_obj(name, verts, faces, loc=(0,0,0), scale=(1,1,1)):
    mesh = bpy.data.meshes.new(name)
    mesh.from_pydata(verts, [], faces)
    mesh.update()
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    obj.location = loc
    obj.scale = scale
    return obj

def make_cyl(name, radius, depth, loc=(0,0,0), segments=12, rot=(0,0,0)):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, vertices=segments, location=loc)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rot
    return obj

def make_sphere(name, radius, loc=(0,0,0), segments=16, ring=12):
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=segments, ring_count=ring, location=loc)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def make_plane(name, size, loc=(0,0,0), rot=(0,0,0)):
    bpy.ops.mesh.primitive_plane_add(size=size, location=loc)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rot
    return obj

def make_cone(name, radius, depth, loc=(0,0,0), segments=12):
    bpy.ops.mesh.primitive_cone_add(radius1=radius, radius2=0, depth=depth, vertices=segments, location=loc)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def make_torus(name, major_radius, minor_radius, loc=(0,0,0)):
    bpy.ops.mesh.primitive_torus_add(major_radius=major_radius, minor_radius=minor_radius, location=loc)
    obj = bpy.context.active_object
    obj.name = name
    return obj

# Dragonfly: small body + two pairs of wings
def make_dragonfly(name, loc, rot=(0,0,0)):
    # Body
    body = make_cyl(f"{name}_Body", radius=0.02, depth=0.15, loc=loc, segments=8)
    body.rotation_euler = rot
    body.scale = (1, 1, 1)
    
    # Head
    head = make_sphere(f"{name}_Head", radius=0.025, loc=(loc[0] + 0.08, loc[1], loc[2]), segments=8, ring=6)
    head.rotation_euler = rot
    
    # Wings: thin planes
    wing_w = 0.12
    wing_l = 0.04
    # Left upper wing
    bpy.ops.mesh.primitive_plane_add(size=1, location=(loc[0] - 0.02, loc[1] - 0.06, loc[2] + 0.01))
    w1 = bpy.context.active_object
    w1.name = f"{name}_WingLU"
    w1.scale = (wing_w/2, wing_l/2, 1)
    w1.rotation_euler = (math.radians(10), math.radians(5), rot[2])
    
    # Right upper wing
    bpy.ops.mesh.primitive_plane_add(size=1, location=(loc[0] - 0.02, loc[1] + 0.06, loc[2] + 0.01))
    w2 = bpy.context.active_object
    w2.name = f"{name}_WingRU"
    w2.scale = (wing_w/2, wing_l/2, 1)
    w2.rotation_euler = (math.radians(-10), math.radians(5), rot[2])
    
    # Left lower wing
    bpy.ops.mesh.primitive_plane_add(size=1, location=(loc[0] + 0.02, loc[1] - 0.05, loc[2] + 0.005))
    w3 = bpy.context.active_object
    w3.name = f"{name}_WingLL"
    w3.scale = (wing_w*0.8/2, wing_l*0.8/2, 1)
    w3.rotation_euler = (math.radians(15), math.radians(-5), rot[2])
    
    # Right lower wing
    bpy.ops.mesh.primitive_plane_add(size=1, location=(loc[0] + 0.02, loc[1] + 0.05, loc[2] + 0.005))
    w4 = bpy.context.active_object
    w4.name = f"{name}_WingRL"
    w4.scale = (wing_w*0.8/2, wing_l*0.8/2, 1)
    w4.rotation_euler = (math.radians(-15), math.radians(-5), rot[2])
    
    return body

# ==================== MATERIALS ====================
def set_material(obj, color, roughness=0.8, metallic=0.0):
    mat = bpy.data.materials.new(f"Mat_{obj.name}")
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = (*color, 1.0)
    bsdf.inputs["Roughness"].default_value = roughness
    bsdf.inputs["Metallic"].default_value = metallic
    obj.data.materials.clear()
    obj.data.materials.append(mat)
