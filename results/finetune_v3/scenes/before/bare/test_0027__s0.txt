def get_material(name, color, metallic=0.0, roughness=0.6):
    mat = bpy.data.materials.get(name)
    if mat is None:
        mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs["Base Color"].default_value = color
        bsdf.inputs["Metallic"].default_value = metallic
        bsdf.inputs["Roughness"].default_value = roughness
    return mat

# --- HELPER: Create a simple dragon head ---
def create_dragon_head(name, location, rotation, scale, mat_body, mat_accent):
    # Main head - elongated box
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    head = bpy.context.object
    head.name = name + "_Head"
    head.scale = (scale * 0.3, scale * 0.8, scale * 0.4)
    head.rotation_euler = rotation
    head.data.materials.append(mat_body)
    
    # Snout
    bpy.ops.mesh.primitive_cube_add(size=1, location=(location[0] + math.sin(rotation[2]) * scale * 0.5, 
                                                       location[1] + math.cos(rotation[2]) * scale * 0.5, 
                                                       location[2]))
    snout = bpy.context.object
    snout.name = name + "_Snout"
    snout.scale = (scale * 0.2, scale * 0.5, scale * 0.3)
    snout.rotation_euler = rotation
    snout.data.materials.append(mat_accent)
    
    # Horns
    for i, side in enumerate([-1, 1]):
        bpy.ops.mesh.primitive_cone_add(radius1=scale*0.1, radius2=scale*0.02, depth=scale*0.5, 
                                        location=(location[0] + side * scale * 0.2 * math.cos(rotation[2]),
                                                  location[1] + side * scale * 0.2 * math.sin(rotation[2]),
                                                  location[2] + scale * 0.3))
        horn = bpy.context.object
        horn.name = name + f"_Horn{i}"
        horn.rotation_euler = (rotation[0] + math.radians(20), rotation[1], rotation[2])
        horn.data.materials.append(mat_accent)
    
    # Eyes
    for i, side in enumerate([-1, 1]):
        bpy.ops.mesh.primitive_uv_sphere_add(radius=scale*0.05, 
                                            location=(location[0] + side * scale * 0.15 * math.cos(rotation[2]),
                                                      location[1] + side * scale * 0.15 * math.sin(rotation[2]),
                                                      location[2] + scale * 0.1))
        eye = bpy.context.object
        eye.name = name + f"_Eye{i}"
        eye.data.materials.append(gold_paint)
    
    return head

# --- HELPER: Create a dragon boat ---
def create_dragon_boats(name, location, rotation, scale, hull_mat, head_mat, accent_mat):
    # Hull - long tapered box
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    hull = bpy.context.object
    hull.name = name + "_Hull"
    hull.scale = (scale * 0.5, scale * 4.0, scale * 0.3)
    hull.rotation_euler = rotation
    hull.data.materials.append(hull_mat)
    
    # Taper the hull (make it narrower at the ends)
    bpy.ops.object.mode_set(mode='EDIT')
    bpy.ops.mesh.select_all(action='SELECT')
    bpy.ops.mesh.subdivide(number_cuts=10)
    bpy.ops.object.mode_set(mode='OBJECT')
    
    # Dragon head at the front (positive Y direction in local space)
    head_loc = (location[0] + math.sin(rotation[2]) * scale * 4.2,
                location[1] + math.cos(rotation[2]) * scale * 4.2,
                location[2] + scale * 0.2)
    create_dragon_head(name, head_loc, rotation, scale * 1.5, head_mat, accent_mat)
    
    # Tail at the back
    tail_loc = (location[0] - math.sin(rotation[2]) * scale * 4.2,
                location[1] - math.cos(rotation[2]) * scale * 4.2,
                location[2] + scale * 0.1)
    bpy.ops.mesh.primitive_cone_add(radius1=scale*0.2, radius2=scale*0.05, depth=scale*1.0,
                                    location=tail_loc)
    tail = bpy.context.object
    tail.name = name + "_Tail"
    tail.rotation_euler = (rotation[0] + math.radians(90), rotation[1], rotation[2])
    tail.data.materials.append(accent_mat)
    
    # Drummer in the bow
    drummer_loc = (location[0] + math.sin(rotation[2]) * scale * 2.0,
                   location[1] + math.cos(rotation[2]) * scale * 2.0,
                   location[2] + scale * 0.5)
    
    # Drummer body
    bpy.ops.mesh.primitive_cylinder_add(radius=scale*0.15, depth=scale*0.4, location=drummer_loc)
    drummer_body = bpy.context.object
    drummer_body.name = name + "_DrummerBody"
    drummer_body.data.materials.append(cloth_red)
    
    # Drummer head
    bpy.ops.mesh.primitive_uv_sphere_add(radius=scale*0.12, 
                                         location=(drummer_loc[0], drummer_loc[1], drummer_loc[2] + scale*0.25))
    drummer_head = bpy.context.object
    drummer_head.name = name + "_DrummerHead"
    drummer_head.data.materials.append(skin_mat)
    
    # Drum
    drum_loc = (drummer_loc[0] + math.sin(rotation[2]) * scale * 0.3,
                drummer_loc[1] + math.cos(rotation[2]) * scale * 0.3,
                drummer_loc[2] + scale * 0.1)
    bpy.ops.mesh.primitive_cylinder_add(radius=scale*0.2, depth=scale*0.15, location=drum_loc)
    drum = bpy.context.object
    drum.name = name + "_Drum"
    drum.rotation_euler = (math.radians(90), 0, rotation[2])
    drum.data.materials.append(wood_dark)
    
    # Rowers (simplified)
    for i in range(8):
        rower_y = -scale * 3.0 + i * scale * 0.7
        rower_loc = (location[0], location[1] + rower_y, location[2] + scale * 0.3)
        
        # Body
        bpy.ops.mesh.primitive_cylinder_add(radius=scale*0.1, depth=scale*0.3, location=rower_loc)
        rower_body = bpy.context.object
        rower_body.name = name + f"_Rower{i}Body"
        rower_body.data.materials.append(cloth_blue if i % 2 == 0 else cloth_yellow)
        
        # Head
        bpy.ops.mesh.primitive_uv_sphere_add(radius=scale*0.08, 
                                            location=(rower_loc[0], rower_loc[1], rower_loc[2] + scale*0.2))
        rower_head = bpy.context.object
        rower_head.name = name + f"_Rower{i}Head"
        rower_head.data.materials.append(skin_mat)
        
        # Oar
        oar_loc = (rower_loc[0] + scale * 0.2, rower_loc[1], rower_loc[2] - scale * 0.1)
        bpy.ops.mesh.primitive_cylinder_add(radius=scale*0.02, depth=scale*0.8, location=oar_loc)
        oar = bpy.context.object
        oar.name = name + f"_Oar{i}"
        oar.rotation_euler = (math.radians(30), 0, rotation[2])
        oar.data.materials.append(wood_light)
    
    return hull
