import bpy
import math
import random
from mathutils import Vector, Matrix

# Clear scene
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# Set up collections
def get_collection(name):
    if name in bpy.data.collections:
        return bpy.data.collections[name]
    col = bpy.data.collections.new(name)
    bpy.context.scene.collection.children.link(col)
    return col

canyon_col = get_collection("Canyon")
tunnel_col = get_collection("Tunnels")
props_col = get_collection("Props")

# Helper functions
def add_to_collection(obj, col):
    for c in obj.users_collection:
        c.objects.unlink(obj)
    col.objects.link(obj)

def set_material(obj, color, metallic=0.0, roughness=0.5):
    mat = bpy.data.materials.new(name=f"Mat_{obj.name}")
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = (*color, 1.0)
    bsdf.inputs["Metallic"].default_value = metallic
    bsdf.inputs["Roughness"].default_value = roughness
    obj.data.materials.append(mat)

# Canyon Wall
# Create a large curved wall
bpy.ops.mesh.primitive_plane_add(size=1, location=(0, 0, 0))
wall = bpy.context.active_object
wall.name = "CanyonWall"
# Subdivide and shape
bpy.ops.object.mode_set(mode='EDIT')
bpy.ops.mesh.subdivide(number_cuts=20)
bpy.ops.object.mode_set(mode='OBJECT')

# Shape the wall with displacement
import bmesh
bm = bmesh.new()
bm.from_mesh(wall.data)
for v in bm.verts:
    # Create a curved canyon wall shape
    x = v.co.x
    y = v.co.y
    z = v.co.z
    # Wall curves inward
    v.co.x = x * 0.5 + 2.0
    v.co.y = y * 0.8
    v.co.z = z * 3.0 + 5.0
    # Add some noise
    noise = math.sin(x * 3) * math.cos(y * 2) * 0.3
    v.co.x += noise
bm.to_mesh(wall.data)
bm.free()

# Solidify the wall
mod = wall.modifiers.new("Solidify", 'SOLIDIFY')
mod.thickness = 0.5
bpy.ops.object.modifier_apply(modifier="Solidify")

# Add subsurf for smoothness
mod = wall.modifiers.new("Subsurf", 'SUBSURF')
mod.levels = 2
mod.render_levels = 2
bpy.ops.object.modifier_apply(modifier="Subsurf")

set_material(wall, (0.45, 0.35, 0.25), 0.0, 0.9)
add_to_collection(wall, canyon_col)

# Ground
bpy.ops.mesh.primitive_plane_add(size=30, location=(0, 0, -0.1))
ground = bpy.context.active_object
ground.name = "Ground"
set_material(ground, (0.35, 0.3, 0.2), 0.0, 0.95)
add_to_collection(ground, canyon_col)

# Canyon floor rocks
for i in range(15):
    x = random.uniform(-5, 5)
    y = random.uniform(-3, 3)
    bpy.ops.mesh.primitive_ico_sphere_add(radius=random.uniform(0.2, 0.6), location=(x, y, 0))
    rock = bpy.context.active_object
    rock.name = f"CanyonRock_{i}"
    rock.scale = (random.uniform(0.8, 1.5), random.uniform(0.8, 1.5), random.uniform(0.5, 1.0))
    rock.rotation_euler = (random.uniform(0, 3.14), random.uniform(0, 3.14), random.uniform(0, 3.14))
    set_material(rock, (0.4, 0.35, 0.28), 0.0, 0.9)
    add_to_collection(rock, canyon_col)

# Tunnel mouths - small round openings in the wall
tunnel_positions = [
    (2.0, -1.5, 1.5),   # Main entry
    (2.2, 0.5, 2.5),    # Upper tunnel
    (2.1, -2.5, 3.0),   # Side tunnel
    (2.3, 1.5, 4.0),    # High tunnel
    (2.0, -0.5, 5.0),   # Top tunnel
]

for idx, pos in enumerate(tunnel_positions):
    # Create tunnel mouth (torus for rim)
    bpy.ops.mesh.primitive_torus_add(
        major_radius=0.4, minor_radius=0.08,
        major_segments=24, minor_segments=12,
        location=pos
    )
    rim = bpy.context.active_object
    rim.name = f"TunnelRim_{idx}"
    # Orient to face outward from wall
    rim.rotation_euler = (0, math.radians(90), 0)
    set_material(rim, (0.35, 0.28, 0.2), 0.0, 0.85)
    add_to_collection(rim, tunnel_col)

    # Dark interior
    bpy.ops.mesh.primitive_cylinder_add(
        radius=0.38, depth=0.3,
        location=(pos[0] + 0.15, pos[1], pos[2])
    )
    interior = bpy.context.active_object
    interior.name = f"TunnelInterior_{idx}"
    interior.rotation_euler = (0, math.radians(90), 0)
    set_material(interior, (0.05, 0.05, 0.05), 0.0, 1.0)
    add_to_collection(interior, tunnel_col)

# Ladders
def create_ladder(location, height=2.0, width=0.3):
    """Create a simple wooden ladder"""
    ladder_parts = []
    # Two side rails
    for side in [-1, 1]:
        bpy.ops.mesh.primitive_cylinder_add(
            radius=0.03, depth=height,
            location=(location[0] + side * width/2, location[1], location[2] + height/2)
        )
        rail = bpy.context.active_object
        rail.name = f"LadderRail_{side}"
        set_material(rail, (0.3, 0.2, 0.1), 0.0, 0.8)
        ladder_parts.append(rail)

    # Rungs
    num_rungs = int(height / 0.25)
    for i in range(num_rungs + 1):
        z = location[2] + i * 0.25
        bpy.ops.mesh.primitive_cylinder_add(
            radius=0.025, depth=width,
            location=(location[0], location[1], z)
        )
        rung = bpy.context.active_object
        rung.name = f"LadderRung_{i}"
        rung.rotation_euler = (0, math.radians(90), 0)
        set_material(rung, (0.3, 0.2, 0.1), 0.0, 0.8)
        ladder_parts.append(rung)

    # Join all parts
    bpy.ops.object.select_all(action='DESELECT')
    for p in ladder_parts:
        p.select_set(True)
    bpy.context.view_layer.objects.active = ladder_parts[0]
    bpy.ops.object.join()
    ladder = bpy.context.active_object
    ladder.name = "Ladder"
    add_to_collection(ladder, props_col)
    return ladder

# Ladder to main entry
create_ladder((1.5, -1.5, 0), height=1.8)
