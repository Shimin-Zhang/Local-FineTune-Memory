```python
import bpy
import math
import random
from mathutils import Vector

# Clean scene
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)
bpy.data.meshes.clear()
bpy.data.materials.clear()
bpy.data.lights.clear()

# Materials
def make_mat(name, color, roughness=0.8, metal=0.0):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get("Principled BSDF")
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Roughness"].default_value = roughness
    bsdf.inputs["Metallic"].default_value = metal
    return mat

mat_ground = make_mat("Ground", (0.15, 0.25, 0.08), roughness=0.95)
mat_tent   = make_mat("Tent",   (0.7, 0.45, 0.15), roughness=0.7)
mat_fire   = make_mat("Fire",   (1.0, 0.4, 0.05), roughness=0.3, metal=0.0)
mat_log    = make_mat("Log",    (0.35, 0.22, 0.1), roughness=0.9)
mat_pine   = make_mat("Pine",   (0.05, 0.2, 0.08), roughness=0.85)
mat_trunk  = make_mat("Trunk",  (0.25, 0.15, 0.08), roughness=0.95)
mat_stone  = make_mat("Stone",  (0.4, 0.4, 0.4), roughness=0.9)

# Ground
bpy.ops.mesh.primitive_plane_add(size=40, location=(0, 0, 0))
ground = bpy.context.active_object
ground.name = "Ground"
ground.data.materials.append(mat_ground)

# Tent (pyramid)
bpy.ops.mesh.primitive_cone_add(vertices=4, radius=1.2, depth=1.8, location=(0, 0, 0.9))
tent = bpy.context.active_object
tent.name = "Tent"
tent.rotation_euler = (0, 0, math.radians(45))
tent.data.materials.append(mat_tent)

# Campfire
# Fire pit (stone ring)
for i in range(12):
    angle = i * (2 * math.pi / 12)
    x = 1.2 * math.cos(angle)
    y = 1.2 * math.sin(angle)
    bpy.ops.mesh.primitive_uv_sphere_add(radius=0.15, location=(x, y, 0.1))
    stone = bpy.context.active_object
    stone.name = f"Stone_{i}"
    stone.scale = (1, 0.8, 0.6)
    stone.data.materials.append(mat_stone)

# Logs for fire
for i in range(4):
    angle = i * (math.pi / 2) + math.radians(20)
    bpy.ops.mesh.primitive_cylinder_add(radius=0.08, depth=0.8, location=(0.2 * math.cos(angle), 0.2 * math.sin(angle), 0.15))
    log = bpy.context.active_object
    log.name = f"FireLog_{i}"
    log.rotation_euler = (math.radians(15), 0, angle)
    log.data.materials.append(mat_log)

# Fire (flame cone)
bpy.ops.mesh.primitive_cone_add(vertices=8, radius=0.25, depth=0.6, location=(0, 0, 0.3))
fire = bpy.context.active_object
fire.name = "Fire"
fire.data.materials.append(mat_fire)
fire.data.materials[0].node_tree.nodes["Principled BSDF"].inputs["Emission Color"].default_value = (1.0, 0.4, 0.05)
fire.data.materials[0].node_tree.nodes["Principled BSDF"].inputs["Emission Strength"].default_value = 5.0

# Point light for fire
bpy.ops.object.light_add(type='POINT', location=(0, 0, 0.5))
light = bpy.context.active_object
light.name = "FireLight"
light.data.color = (1.0, 0.5, 0.2)
light.data.energy = 50
light.data.shadow_soft_size = 0.3

# Log bench
# Two logs as seat
bpy.ops.mesh.primitive_cylinder_add(radius=0.15, depth=2.0, location=(-2, 1.5, 0.3))
bench_log1 = bpy.context.active_object
bench_log1.name = "BenchLog1"
bench_log1.rotation_euler = (0, math.radians(90), 0)
bench_log1.data.materials.append(mat_log)

bpy.ops.mesh.primitive_cylinder_add(radius=0.15, depth=2.0, location=(-2, 1.9, 0.3))
bench_log2 = bpy.context.active_object
bench_log2.name = "BenchLog2"
bench_log2.rotation_euler = (0, math.radians(90), 0)
bench_log2.data.materials.append(mat_log)

# Legs
for x in [-2.8, -1.2]:
    for y in [1.3, 2.1]:
        bpy.ops.mesh.primitive_cylinder_add(radius=0.08, depth=0.3, location=(x, y, 0.15))
        leg = bpy.context.active_object
        leg.name = f"BenchLeg_{x}_{y}"
        leg.data.materials.append(mat_log)

# Pine trees
def make_pine_tree(x, y, scale=1.0):
    # Trunk
    bpy.ops.mesh.primitive_cylinder_add(radius=0.15 * scale, depth=2.0 * scale, location=(x, y, 1.0 * scale))
    trunk = bpy.context.active_object
    trunk.name = f"PineTrunk_{x}_{y}"
    trunk.data.materials.append(mat_trunk)

    # Foliage layers (cones)
    for i, (z_offset, radius, depth) in enumerate([(1.5, 0.8, 1.0), (2.2, 0.6, 0.9), (2.8, 0.4, 0.8)]):
        bpy.ops.mesh.primitive_cone_add(vertices=12, radius=radius * scale, depth=depth * scale, location=(x, y, (z_offset + depth/2) * scale))
        foliage = bpy.context.active_object
        foliage.name = f"PineFoliage_{x}_{y}_{i}"
        foliage.data.materials.append(mat_pine)

# Place trees around the campsite
tree_positions = [
    (-5, -4, 1.2), (-6, 2, 1.0), (-4, 5, 1.1),
    (5, -3, 1.3), (6, 3, 1.0), (4, 5, 1.2),
    (-3, -6, 1.1), (3, -6, 1.0), (0, -7, 1.3),
    (-7, -1, 1.0), (7, -1, 1.1), (-2, 7, 1.2), (2, 7, 1.0),
]

for x, y, s in tree_positions:
    make_pine_tree(x, y, s)

# Camera
bpy.ops.object.camera_add(location=(8, 8, 4))
camera = bpy.context.active_object
camera.name = "Camera"
camera.rotation_euler = (math.radians(60), 0, math.radians(-45))
bpy.context.scene.camera = camera

# Sun light
bpy.ops.object.light_add(type='SUN', location=(0, 0, 10))
sun = bpy.context.active_object
sun.name = "Sun"
sun.data.energy = 3
sun.rotation_euler = (math.radians(45), math.radians(30), 0)

# World background
world = bpy.data.worlds.get("World") or bpy.data.worlds.new("World")
bpy.context.scene.world = world
world.use_nodes = True
bg = world.node_tree.nodes.get("Background")
if bg:
    bg.inputs[0].default_value = (0.4, 0.5, 0.7, 1.0)
    bg.inputs[1].default_value = 1.0
```
