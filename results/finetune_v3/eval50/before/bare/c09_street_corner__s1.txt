```python
import bpy
import math
import bmesh

# Clear scene
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# Materials
def create_material(name, color):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Roughness"].default_value = 0.8
    return mat

mat_building1 = create_material("Building1", (0.7, 0.65, 0.6, 1.0))
mat_building2 = create_material("Building2", (0.55, 0.6, 0.65, 1.0))
mat_roof = create_material("Roof", (0.3, 0.25, 0.2, 1.0))
mat_street = create_material("Street", (0.2, 0.2, 0.22, 1.0))
mat_sidewalk = create_material("Sidewalk", (0.5, 0.5, 0.5, 1.0))
mat_lamp_pole = create_material("LampPole", (0.1, 0.1, 0.12, 1.0))
mat_lamp_head = create_material("LampHead", (0.9, 0.9, 0.7, 1.0))
mat_mailbox = create_material("Mailbox", (0.7, 0.1, 0.1, 1.0))
mat_bench = create_material("Bench", (0.35, 0.2, 0.1, 1.0))
mat_crosswalk = create_material("Crosswalk", (0.9, 0.9, 0.9, 1.0))

# Ground / Street
street = bpy.data.meshes.new("Street")
bpy.ops.mesh.primitive_plane_add(size=20, location=(0, 0, 0))
bpy.context.active_object.name = "Street"
bpy.context.active_object.data.materials.append(mat_street)

# Sidewalks (two strips forming an L around the corner)
# Sidewalk along X axis (left side)
bpy.ops.mesh.primitive_cube_add(size=1, location=(-6, 0, 0.05))
bpy.context.active_object.name = "Sidewalk_X"
bpy.context.active_object.scale = (8, 1.5, 0.05)
bpy.context.active_object.data.materials.append(mat_sidewalk)

# Sidewalk along Y axis (back side)
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, -6, 0.05))
bpy.context.active_object.name = "Sidewalk_Y"
bpy.context.active_object.scale = (1.5, 8, 0.05)
bpy.context.active_object.data.materials.append(mat_sidewalk)

# Building 1 (along X axis, left side)
bpy.ops.mesh.primitive_cube_add(size=1, location=(-8, 0, 3))
bpy.context.active_object.name = "Building1"
bpy.context.active_object.scale = (3, 4, 3)
bpy.context.active_object.data.materials.append(mat_building1)

# Building 1 roof
bpy.ops.mesh.primitive_cube_add(size=1, location=(-8, 0, 6.05))
bpy.context.active_object.name = "Building1_Roof"
bpy.context.active_object.scale = (3.1, 4.1, 0.1)
bpy.context.active_object.data.materials.append(mat_roof)

# Building 2 (along Y axis, back side)
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, -8, 3))
bpy.context.active_object.name = "Building2"
bpy.context.active_object.scale = (4, 3, 3)
bpy.context.active_object.data.materials.append(mat_building2)

# Building 2 roof
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, -8, 6.05))
bpy.context.active_object.name = "Building2_Roof"
bpy.context.active_object.scale = (4.1, 3.1, 0.1)
bpy.context.active_object.data.materials.append(mat_roof)

# Street Lamp (at the corner)
# Pole
bpy.ops.mesh.primitive_cylinder_add(radius=0.08, depth=3.5, location=(-2, -2, 1.75))
bpy.context.active_object.name = "Lamp_Pole"
bpy.context.active_object.data.materials.append(mat_lamp_pole)

# Lamp arm (horizontal)
bpy.ops.mesh.primitive_cylinder_add(radius=0.06, depth=0.8, location=(-1.6, -2, 3.5))
bpy.context.active_object.name = "Lamp_Arm"
bpy.context.active_object.rotation_euler = (0, math.radians(90), 0)
bpy.context.active_object.data.materials.append(mat_lamp_pole)

# Lamp head
bpy.ops.mesh.primitive_uv_sphere_add(radius=0.2, location=(-1.2, -2, 3.5))
bpy.context.active_object.name = "Lamp_Head"
bpy.context.active_object.scale = (1, 1, 0.6)
bpy.context.active_object.data.materials.append(mat_lamp_head)

# Mailbox
# Post
bpy.ops.mesh.primitive_cylinder_add(radius=0.05, depth=0.8, location=(-3, -1, 0.4))
bpy.context.active_object.name = "Mailbox_Post"
bpy.context.active_object.data.materials.append(mat_lamp_pole)

# Box
bpy.ops.mesh.primitive_cube_add(size=1, location=(-3, -1, 0.9))
bpy.context.active_object.name = "Mailbox_Box"
bpy.context.active_object.scale = (0.3, 0.2, 0.25)
bpy.context.active_object.data.materials.append(mat_mailbox)

# Bench (on sidewalk)
# Seat
bpy.ops.mesh.primitive_cube_add(size=1, location=(-1, -4, 0.45))
bpy.context.active_object.name = "Bench_Seat"
bpy.context.active_object.scale = (1.5, 0.4, 0.08)
bpy.context.active_object.data.materials.append(mat_bench)

# Backrest
bpy.ops.mesh.primitive_cube_add(size=1, location=(-1, -4.15, 0.75))
bpy.context.active_object.name = "Bench_Back"
bpy.context.active_object.scale = (1.5, 0.08, 0.5)
bpy.context.active_object.data.materials.append(mat_bench)

# Legs (4 legs)
for x_off in [-0.6, 0.6]:
    for y_off in [-0.15, 0.15]:
        bpy.ops.mesh.primitive_cylinder_add(radius=0.04, depth=0.45, location=(-1 + x_off, -4 + y_off, 0.225))
        bpy.context.active_object.name = f"Bench_Leg_{x_off}_{y_off}"
        bpy.context.active_object.data.materials.append(mat_bench)

# Crosswalk (stripes on the street)
# The street corner is at (-2, -2). Crosswalk goes along X axis (east-west)
for i in range(6):
    x_pos = -4 + i * 0.8
    bpy.ops.mesh.primitive_cube_add(size=1, location=(x_pos, -2, 0.02))
    bpy.context.active_object.name = f"Crosswalk_{i}"
    bpy.context.active_object.scale = (0.4, 2.5, 0.02)
    bpy.context.active_object.data.materials.append(mat_crosswalk)

# Add a light for visibility
bpy.ops.object.light_add(type='SUN', location=(10, 10, 20))
bpy.context.active_object.name = "Sun"
bpy.context.active_object.rotation_euler = (math.radians(45), math.radians(30), 0)
bpy.context.active_object.data.energy = 3

# Camera
bpy.ops.object.camera_add(location=(10, 10, 8))
bpy.context.active_object.name = "Camera"
# Point camera at the corner
target = bpy.data.objects.get("Street")
if target:
    direction = target.location - bpy.context.active_object.location
    rot_quat = direction.to_track_quat('Z', 'Y')
    bpy.context.active_object.rotation_euler = rot_quat.to_euler()

# Set camera as active
bpy.context.scene.camera = bpy.context.active_object

print("Street corner scene created!")
```
