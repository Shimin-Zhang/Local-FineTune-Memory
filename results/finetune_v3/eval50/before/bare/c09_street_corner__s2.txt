```python
import bpy
import math

# Clear existing scene
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# Materials
def make_mat(name, color, roughness=0.7):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Roughness"].default_value = roughness
    return mat

mat_road   = make_mat("Road",   (0.15, 0.15, 0.15), 0.9)
mat_sidewalk = make_mat("Sidewalk", (0.55, 0.55, 0.55), 0.8)
mat_bldg1  = make_mat("Bldg1",  (0.75, 0.65, 0.55), 0.85)
mat_bldg2  = make_mat("Bldg2",  (0.55, 0.60, 0.70), 0.85)
mat_roof   = make_mat("Roof",   (0.30, 0.25, 0.20), 0.9)
mat_lamp   = make_mat("Lamp",   (0.10, 0.10, 0.10), 0.5)
mat_glow   = make_mat("Glow",   (1.0, 0.9, 0.5), 0.3)
mat_mail   = make_mat("Mail",   (0.85, 0.10, 0.10), 0.6)
mat_bench  = make_mat("Bench",  (0.35, 0.22, 0.12), 0.8)
mat_cross  = make_mat("Cross",  (0.95, 0.95, 0.95), 0.8)
mat_door   = make_mat("Door",   (0.20, 0.12, 0.08), 0.7)
mat_window = make_mat("Window", (0.60, 0.75, 0.85), 0.3)

def add_box(name, size, loc, mat):
    bpy.ops.mesh.primitive_cube_add(size=1, location=loc)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    bpy.ops.object.transform_apply(scale=True)
    obj.data.materials.append(mat)
    return obj

# ---- Ground / Road ----
# Main road (east-west)
add_box("Road_EW", (60, 10, 0.2), (0, 0, -0.1), mat_road)
# Cross street (north-south)
add_box("Road_NS", (10, 60, 0.2), (0, 0, -0.1), mat_road)

# Sidewalks (slightly raised)
# North sidewalk strip
add_box("Sidewalk_N", (60, 3, 0.25), (0, 6.5, 0.125), mat_sidewalk)
# South sidewalk strip
add_box("Sidewalk_S", (60, 3, 0.25), (0, -6.5, 0.125), mat_sidewalk)
# East sidewalk strip
add_box("Sidewalk_E", (3, 60, 0.25), (6.5, 0, 0.125), mat_sidewalk)
# West sidewalk strip
add_box("Sidewalk_W", (3, 60, 0.25), (-6.5, 0, 0.125), mat_sidewalk)

# ---- Crosswalk (east-west road, across north-south street) ----
# Zebra stripes on the north-south road
for i in range(6):
    y = -3 + i * 1.2
    add_box(f"Cross_{i}", (8, 0.8, 0.05), (0, y, 0.12), mat_cross)

# ---- Building 1 (North-West corner) ----
b1x, b1y = -18, 18
# Main body
add_box("Bldg1_Body", (12, 12, 15), (b1x, b1y, 7.5), mat_bldg1)
# Roof
add_box("Bldg1_Roof", (12.5, 12.5, 0.5), (b1x, b1y, 15.25), mat_roof)
# Door (south face)
add_box("Bldg1_Door", (2, 0.3, 4), (b1x, b1y - 6.1, 2), mat_door)
# Windows (south face)
for wx in range(3):
    for wy in range(3):
        add_box(f"Bldg1_Win_{wx}_{wy}", (1.5, 0.2, 1.5),
                (b1x - 3 + wx * 3, b1y - 6.1, 5 + wy * 3), mat_window)
# Windows (east face)
for wx in range(3):
    for wy in range(3):
        add_box(f"Bldg1_WinE_{wx}_{wy}", (0.2, 1.5, 1.5),
                (b1x + 6.1, b1y - 3 + wx * 3, 5 + wy * 3), mat_window)

# ---- Building 2 (South-East corner) ----
b2x, b2y = 18, -18
add_box("Bldg2_Body", (12, 12, 12), (b2x, b2y, 6), mat_bldg2)
add_box("Bldg2_Roof", (12.5, 12.5, 0.5), (b2x, b2y, 12.25), mat_roof)
# Door (north face)
add_box("Bldg2_Door", (2, 0.3, 4), (b2x, b2y + 6.1, 2), mat_door)
# Windows (north face)
for wx in range(3):
    for wy in range(2):
        add_box(f"Bldg2_Win_{wx}_{wy}", (1.5, 0.2, 1.5),
                (b2x - 3 + wx * 3, b2y + 6.1, 5 + wy * 3), mat_window)
# Windows (west face)
for wx in range(3):
    for wy in range(2):
        add_box(f"Bldg2_WinW_{wx}_{wy}", (0.2, 1.5, 1.5),
                (b2x - 6.1, b2y - 3 + wx * 3, 5 + wy * 3), mat_window)

# ---- Street Lamp (North-East corner sidewalk) ----
lamp_x, lamp_y = 8, 8
# Pole
bpy.ops.mesh.primitive_cylinder_add(radius=0.15, depth=6, location=(lamp_x, lamp_y, 3))
pole = bpy.context.active_object
pole.name = "Lamp_Pole"
pole.data.materials.append(mat_lamp)
# Arm
add_box("Lamp_Arm", (1.5, 0.15, 0.15), (lamp_x - 0.75, lamp_y, 5.9), mat_lamp)
# Lamp head
bpy.ops.mesh.primitive_cylinder_add(radius=0.3, depth=0.3, location=(lamp_x - 1.5, lamp_y, 5.9))
head = bpy.context.active_object
head.name = "Lamp_Head"
head.data.materials.append(mat_glow)
# Point light
bpy.ops.object.light_add(type='POINT', location=(lamp_x - 1.5, lamp_y, 5.7))
light = bpy.context.active_object
light.name = "Lamp_Light"
light.data.energy = 100
light.data.color = (1.0, 0.9, 0.6)

# ---- Mailbox (South-West sidewalk) ----
mb_x, mb_y = -8, -8
# Post
bpy.ops.mesh.primitive_cylinder_add(radius=0.1, depth=1.2, location=(mb_x, mb_y, 0.6))
mb_post = bpy.context.active_object
mb_post.name = "Mailbox_Post"
mb_post.data.materials.append(mat_lamp)
# Box
add_box("Mailbox_Box", (0.8, 0.5, 0.6), (mb_x, mb_y, 1.5), mat_mail)
# Flag
add_box("Mailbox_Flag", (0.05, 0.3, 0.2), (mb_x + 0.45, mb_y, 1.7), mat_mail)

# ---- Bench (South-East sidewalk) ----
bench_x, bench_y = 8, -8
# Legs
for lx in [-1, 1]:
    add_box(f"Bench_Leg_{lx}", (0.15, 0.15, 0.5), (bench_x + lx, bench_y, 0.25), mat_bench)
# Seat
add_box("Bench_Seat", (2.5, 0.6, 0.1), (bench_x, bench_y, 0.55), mat_bench)
# Backrest
add_box("Bench_Back", (2.5, 0.1, 0.6), (bench_x, bench_y + 0.3, 0.9), mat_bench)
# Back legs
for lx in [-1, 1]:
    add_box(f"Bench_BackLeg_{lx}", (0.15, 0.15, 0.6), (bench_x + lx, bench_y + 0.3, 0.3), mat_bench)

# ---- Camera ----
bpy.ops.object.camera_add(location=(25, 25, 15))
cam = bpy.context.active_object
cam.name = "StreetCam"
cam.rotation_euler = (math.radians(65), 0, math.radians(45))
bpy.context.scene.camera = cam

# ---- Sun light ----
bpy.ops.object.light_add(type='SUN', location=(0, 0, 30))
sun = bpy.context.active_object
sun.name = "Sun"
sun.data.energy = 3
sun.rotation_euler = (math.radians(45), math.radians(20), 0)

# World background
world = bpy.context.scene.world
if world is None:
    world = bpy.data.worlds.new("World")
    bpy.context.scene.world = world
world.use_nodes = True
bg = world.node_tree.nodes["Background"]
bg.inputs[0].default_value = (0.7, 0.8, 0.95, 1.0)
bg.inputs[1].default_value = 1.0
```
