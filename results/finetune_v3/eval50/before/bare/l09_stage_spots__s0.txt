```python
import bpy
import math

# Clear existing scene objects
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# Create the white stage floor
bpy.ops.mesh.primitive_plane_add(size=10, location=(0, 0, 0))
floor = bpy.context.active_object
floor.name = "StageFloor"

# Create white material for the floor
mat = bpy.data.materials.new(name="WhiteFloor")
mat.use_nodes = True
nodes = mat.node_tree.nodes
links = mat.node_tree.links
bsdf = nodes["Principled BSDF"]
bsdf.inputs["Base Color"].default_value = (1.0, 1.0, 1.0, 1.0)
bsdf.inputs["Roughness"].default_value = 0.8
floor.data.materials.append(mat)

# Create three spotlights (red, green, blue)
colors = [
    ("RedSpot",   (1.0, 0.0, 0.0, 1.0), (-2, -3, 5)),
    ("GreenSpot", (0.0, 1.0, 0.0, 1.0), ( 0, -3, 5)),
    ("BlueSpot",  (0.0, 0.0, 1.0, 1.0), ( 2, -3, 5)),
]

for name, color, loc in colors:
    # Create light data
    light_data = bpy.data.lights.new(name=name, type='SPOT')
    light_data.energy = 500
    light_data.color = color
    light_data.spot_size = math.radians(45)   # cone angle
    light_data.spot_blend = 0.3
    light_data.shadow_soft_size = 0.1

    # Create light object
    light_obj = bpy.data.objects.new(name, light_data)
    light_obj.location = loc
    bpy.context.collection.objects.link(light_obj)

    # Aim the spotlight down toward the center of the floor
    # Direction vector from light to target (0,0,0)
    import mathutils
    direction = mathutils.Vector((0, 0, 0)) - mathutils.Vector(loc)
    rotation_quat = direction.to_track_quat('-Z', 'Y')
    light_obj.rotation_euler = rotation_quat.to_euler()

# Set up world to be dark so only the spotlights are visible
world = bpy.context.scene.world
if world is None:
    world = bpy.data.worlds.new("World")
    bpy.context.scene.world = world
world.use_nodes = True
bg_node = world.node_tree.nodes.get("Background")
if bg_node:
    bg_node.inputs[0].default_value = (0.0, 0.0, 0.0, 1.0)
    bg_node.inputs[1].default_value = 1.0

# Enable bloom for nicer glow effect
scene = bpy.context.scene
scene.view_settings.bloom = 0.5
```
