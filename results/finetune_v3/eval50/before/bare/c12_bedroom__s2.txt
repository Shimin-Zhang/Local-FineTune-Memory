```python
import bpy
import math

# Clear the scene
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# Materials
def make_mat(name, color, roughness=0.6):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Roughness"].default_value = roughness
    return mat

mat_wall   = make_mat("Wall",   (0.9, 0.9, 0.85), 0.8)
mat_floor  = make_mat("Floor",  (0.7, 0.5, 0.3),  0.7)
mat_bed    = make_mat("Bed",    (0.8, 0.7, 0.6),  0.6)
mat_matt   = make_mat("Mattress",(0.95,0.95,0.95),0.4)
mat_pill   = make_mat("Pillow", (0.9, 0.9, 0.95), 0.3)
mat_stand  = make_mat("Nightstand",(0.5,0.35,0.2),0.5)
mat_lamp   = make_mat("Lamp",   (0.9,0.85,0.6),  0.4)
mat_ward   = make_mat("Wardrobe",(0.4,0.28,0.18), 0.5)
mat_rug    = make_mat("Rug",    (0.6,0.2,0.2),   0.9)
mat_window = make_mat("Window", (0.7,0.85,0.95), 0.1)
mat_frame  = make_mat("Frame",  (0.9,0.9,0.9),   0.3)

def add_box(name, size, loc, mat, rot=(0,0,0)):
    bpy.ops.mesh.primitive_cube_add(size=1, location=loc, rotation=rot)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.data.materials.append(mat)
    return obj

# Floor
add_box("Floor", (10, 10, 0.2), (0, 0, -0.1), mat_floor)

# Walls
add_box("Wall_N", (10, 0.2, 4), (0, -5, 2), mat_wall)
add_box("Wall_S", (10, 0.2, 4), (0,  5, 2), mat_wall)
add_box("Wall_W", (0.2, 10, 4), (-5, 0, 2), mat_wall)
add_box("Wall_E", (0.2, 10, 4), ( 5, 0, 2), mat_wall)

# Window on North wall
win = add_box("Window", (3, 0.1, 2), (0, -4.95, 2.2), mat_window)
# Frame
add_box("WinFrame_T", (3.2, 0.15, 0.15), (0, -4.95, 3.25), mat_frame)
add_box("WinFrame_B", (3.2, 0.15, 0.15), (0, -4.95, 1.15), mat_frame)
add_box("WinFrame_L", (0.15, 0.15, 2.2), (-1.55, -4.95, 2.2), mat_frame)
add_box("WinFrame_R", (0.15, 0.15, 2.2), ( 1.55, -4.95, 2.2), mat_frame)
add_box("WinFrame_MidV", (0.1, 0.15, 2.2), (0, -4.95, 2.2), mat_frame)
add_box("WinFrame_MidH", (3.2, 0.15, 0.1), (0, -4.95, 2.2), mat_frame)

# Bed
add_box("BedFrame", (2.5, 4, 0.4), (2, 2, 0.2), mat_bed)
add_box("Mattress", (2.3, 3.8, 0.3), (2, 2, 0.55), mat_matt)
add_box("Pillow1", (0.9, 0.5, 0.2), (2, 3.5, 0.8), mat_pill)
add_box("Pillow2", (0.9, 0.5, 0.2), (2, 3.0, 0.8), mat_pill)
# Headboard
add_box("Headboard", (2.5, 0.2, 1.2), (2, 4, 1.0), mat_bed)

# Nightstand
add_box("Nightstand", (1, 0.8, 0.6), (0.5, 3.5, 0.3), mat_stand)
# Lamp
add_box("LampBase", (0.3, 0.3, 0.1), (0.5, 3.5, 0.65), mat_lamp)
add_box("LampPole", (0.05, 0.05, 0.4), (0.5, 3.5, 0.85), mat_lamp)
add_box("LampShade", (0.5, 0.5, 0.3), (0.5, 3.5, 1.15), mat_lamp)

# Wardrobe
add_box("Wardrobe", (2, 0.8, 2.5), (-4, -3, 1.25), mat_ward)
# Door handles
add_box("Handle1", (0.05, 0.05, 0.3), (-3.1, -3, 1.25), mat_frame)
add_box("Handle2", (0.05, 0.05, 0.3), (-3.1, -2.5, 1.25), mat_frame)

# Rug
add_box("Rug", (3, 2, 0.05), (1, 0, 0.025), mat_rug)

# Lighting
bpy.ops.object.light_add(type='SUN', location=(0, 0, 8))
sun = bpy.context.active_object
sun.rotation_euler = (math.radians(30), math.radians(20), 0)
sun.data.energy = 3

bpy.ops.object.light_add(type='POINT', location=(0.5, 3.5, 1.3))
lamp_light = bpy.context.active_object
lamp_light.data.energy = 50
lamp_light.data.color = (1.0, 0.9, 0.7)

# Camera
bpy.ops.object.camera_add(location=(4, 6, 3))
cam = bpy.context.active_object
cam.rotation_euler = (math.radians(70), 0, math.radians(160))
bpy.context.scene.camera = cam
```
