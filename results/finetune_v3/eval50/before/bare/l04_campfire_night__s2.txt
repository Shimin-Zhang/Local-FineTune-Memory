```python
import bpy
import math
import random

# Clear the scene
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete()

# Create a dark world (night sky)
world = bpy.data.worlds.new("NightWorld")
bpy.context.scene.world = world
world.use_nodes = True
bg_node = world.node_tree.nodes.get("Background")
bg_node.inputs[0].default_value = [0.001, 0.001, 0.002]  # Very dark blue-black
bg_node.inputs[1].default_value = 1.0

# Ground plane
bpy.ops.mesh.primitive_plane_add(size=20, location=(0, 0, 0))
ground = bpy.context.active_object
ground.name = "Ground"
mat_ground = bpy.data.materials.new("GroundMat")
mat_ground.use_nodes = True
bsdf = mat_ground.node_tree.nodes.get("Principled BSDF")
bsdf.inputs["Base Color"].default_value = [0.05, 0.04, 0.03, 1]  # Dark earth
bsdf.inputs["Roughness"].default_value = 1.0
ground.data.materials.append(mat_ground)

# Fire pit - ring of stones
stone_count = 12
stone_radius = 0.6
for i in range(stone_count):
    angle = (i / stone_count) * 2 * math.pi
    x = stone_radius * math.cos(angle)
    y = stone_radius * math.sin(angle)
    bpy.ops.mesh.primitive_ico_sphere_add(subdivisions=2, radius=0.12, location=(x, y, 0.08))
    stone = bpy.context.active_object
    stone.name = f"Stone_{i}"
    stone.scale = (1, 0.8, 0.6)
    stone.rotation_euler = (random.uniform(0, math.pi), random.uniform(0, math.pi), random.uniform(0, math.pi))
    mat_stone = bpy.data.materials.new(f"StoneMat_{i}")
    mat_stone.use_nodes = True
    bsdf_s = mat_stone.node_tree.nodes.get("Principled BSDF")
    bsdf_s.inputs["Base Color"].default_value = [0.15, 0.13, 0.12, 1]
    bsdf_s.inputs["Roughness"].default_value = 0.9
    stone.data.materials.append(mat_stone)

# Logs (crossed)
def create_log(start, end, radius=0.06):
    direction = (end[0] - start[0], end[1] - start[1], end[2] - start[2])
    length = math.sqrt(direction[0]**2 + direction[1]**2 + direction[2]**2)
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=length, location=((start[0]+end[0])/2, (start[1]+end[1])/2, (start[2]+end[2])/2))
    log = bpy.context.active_object
    # Align cylinder to direction
    log.rotation_euler = (0, math.atan2(direction[0], direction[2]), 0)
    # More precise alignment
    if length > 0:
        dir_vec = bpy.data.objects.new("temp", None)
        # Use quaternion from direction
        import mathutils
        direction_vec = mathutils.Vector(direction).normalized()
        quat = direction_vec.to_track_quat('Z', 'Y')
        log.rotation_euler = quat.to_euler()
    mat_log = bpy.data.materials.new("LogMat")
    mat_log.use_nodes = True
    bsdf_l = mat_log.node_tree.nodes.get("Principled BSDF")
    bsdf_l.inputs["Base Color"].default_value = [0.12, 0.07, 0.04, 1]  # Dark wood
    bsdf_l.inputs["Roughness"].default_value = 1.0
    log.data.materials.append(mat_log)
    return log

# Crossed logs in the fire pit
create_log((-0.3, -0.1, 0.15), (0.3, 0.1, 0.2))
create_log((-0.1, 0.3, 0.15), (0.1, -0.3, 0.2))
create_log((-0.2, 0.2, 0.18), (0.2, -0.2, 0.22))

# Fire flames - using emissive spheres/cones to simulate fire
fire_center = (0, 0, 0.25)

# Main fire glow - point light
light = bpy.data.lights.new("FireLight", type='POINT')
light.color = (1.0, 0.45, 0.1)  # Orange glow
light.energy = 150
light.shadow_soft_size = 0.3
light_obj = bpy.data.objects.new("FireLight", light)
bpy.context.collection.objects.link(light_obj)
light_obj.location = (0, 0, 0.35)

# Secondary smaller light for flicker effect
light2 = bpy.data.lights.new("FireLight2", type='POINT')
light2.color = (1.0, 0.3, 0.05)  # Deeper orange
light2.energy = 50
light2.shadow_soft_size = 0.2
light_obj2 = bpy.data.objects.new("FireLight2", light2)
bpy.context.collection.objects.link(light_obj2)
light_obj2.location = (0.05, 0.05, 0.25)

# Flame shapes using emissive materials
def create_flame(location, scale, color, energy):
    bpy.ops.mesh.primitive_cone_add(vertices=8, radius1=scale[0], radius2=0, depth=scale[2], location=location)
    flame = bpy.context.active_object
    flame.name = "Flame"
    flame.scale = (scale[0], scale[1], scale[2])
    mat_flame = bpy.data.materials.new("FlameMat")
    mat_flame.use_nodes = True
    # Remove default BSDF, use emission
    nodes = mat_flame.node_tree.nodes
    links = mat_flame.node_tree.links
    nodes.clear()
    emission = nodes.new('ShaderNodeEmission')
    emission.inputs[0].default_value = color
    emission.inputs[1].default_value = energy
    output = nodes.new('ShaderNodeOutputMaterial')
    links.new(emission.outputs[0], output.inputs[0])
    flame.data.materials.append(mat_flame)
    return flame

# Multiple flame cones
create_flame((0, 0, 0.35), (0.12, 0.12, 0.3), (1.0, 0.5, 0.1), 20)
create_flame((0.08, 0.05, 0.32), (0.08, 0.08, 0.25), (1.0, 0.6, 0.15), 15)
create_flame((-0.06, -0.04, 0.33), (0.09, 0.09, 0.28), (1.0, 0.45, 0.08), 18)
create_flame((0.02, -0.08, 0.3), (0.07, 0.07, 0.2), (1.0, 0.7, 0.2), 12)
create_flame((-0.05, 0.06, 0.31), (0.06, 0.06, 0.18), (1.0, 0.65, 0.25), 10)

# Ember glow at base
bpy.ops.mesh.primitive_uv_sphere_add(radius=0.15, location=(0, 0, 0.18))
embers = bpy.context.active_object
embers.name = "Embers"
embers.scale = (1, 1, 0.4)
mat_ember = bpy.data.materials.new("EmberMat")
mat_ember.use_nodes = True
nodes_e = mat_ember.node_tree.nodes
links_e = mat_ember.node_tree.links
nodes_e.clear()
emission_e = nodes_e.new('ShaderNodeEmission')
emission_e.inputs[0].default_value = (1.0, 0.3, 0.05)
emission_e.inputs[1].default_value = 8
output_e = nodes_e.new('ShaderNodeOutputMaterial')
links_e.new(emission_e.outputs[0], output_e.inputs[0])
embers.data.materials.append(mat_ember)

# A few scattered rocks around the scene
for i in range(8):
    angle = random.uniform(0, 2 * math.pi)
    dist = random.uniform(1.5, 4)
    x = dist * math.cos(angle)
    y = dist * math.sin(angle)
    size = random.uniform(0.08, 0.2)
    bpy.ops.mesh.primitive_ico_sphere_add(subdivisions=1, radius=size, location=(x, y, size * 0.5))
    rock = bpy.context.active_object
    rock.name = f"Rock_{i}"
    rock.scale = (1, random.uniform(0.7, 1.3), random.uniform(0.5, 0.8))
    rock.rotation_euler = (random.uniform(0, math.pi), random.uniform(0, math.pi), random.uniform(0, math.pi))
    mat_rock = bpy.data.materials.new(f"RockMat_{i}")
    mat_rock.use_nodes = True
    bsdf_r = mat_rock.node_tree.nodes.get("Principled BSDF")
    bsdf_r.inputs["Base Color"].default_value = [0.1, 0.09, 0.08, 1]
    bsdf_r.inputs["Roughness"].default_value = 0.95
    rock.data.materials.append(mat_rock)

# Set up camera
bpy.ops.object.camera_add(location=(2.5, 2.5, 1.2))
camera = bpy.context.active_object
camera.name = "Camera"
# Point camera at fire
direction = (0 - camera.location[0], 0 - camera.location[1], 0.3 - camera.location[2])
import mathutils
direction_vec = mathutils.Vector(direction).normalized()
rot_quat = direction_vec.to_track_quat('-Z', 'Y')
camera.rotation_euler = rot_quat.to_euler()
bpy.context.scene.camera = camera

# Render settings
scene = bpy.context.scene
scene.render.resolution_x = 1920
scene.render.resolution_y = 1080
scene.render.film_transparent = False
scene.cycles.samples = 256
scene.cycles.use_denoising = True

# Set viewport shading to rendered
bpy.context.space_data.shading.type = 'RENDERED'

print("Campfire scene created successfully!")
```
