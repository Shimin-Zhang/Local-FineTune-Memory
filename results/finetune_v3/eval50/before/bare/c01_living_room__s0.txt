```python
import bpy
import math

# Clear the scene
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# Create a collection for the living room
living_room = bpy.data.collections.new("Living Room")
bpy.context.scene.collection.children.link(living_room)

def move_to_collection(obj, collection):
    for c in obj.users_collection:
        c.objects.unlink(obj)
    collection.objects.link(obj)

# Materials
def create_material(name, color, roughness=0.5, metallic=0.0):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get("Principled BSDF")
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Roughness"].default_value = roughness
    bsdf.inputs["Metallic"].default_value = metallic
    return mat

wood_mat = create_material("Wood", (0.3, 0.15, 0.05, 1), roughness=0.6)
fabric_mat = create_material("Fabric", (0.2, 0.3, 0.5, 1), roughness=0.9)
rug_mat = create_material("Rug", (0.5, 0.2, 0.1, 1), roughness=0.95)
metal_mat = create_material("Metal", (0.7, 0.7, 0.7, 1), roughness=0.3, metallic=0.8)
shade_mat = create_material("LampShade", (0.9, 0.85, 0.7, 1), roughness=0.7)
floor_mat = create_material("Floor", (0.6, 0.5, 0.4, 1), roughness=0.7)
wall_mat = create_material("Wall", (0.9, 0.88, 0.85, 1), roughness=0.8)
tabletop_mat = create_material("TableTop", (0.1, 0.1, 0.1, 1), roughness=0.2)

# Floor
bpy.ops.mesh.primitive_plane_add(size=8, location=(0, 0, 0))
floor = bpy.context.active_object
floor.name = "Floor"
floor.data.materials.append(floor_mat)
move_to_collection(floor, living_room)

# Back wall
bpy.ops.mesh.primitive_plane_add(size=8, location=(0, -4, 2), rotation=(math.radians(90), 0, 0))
back_wall = bpy.context.active_object
back_wall.name = "BackWall"
back_wall.data.materials.append(wall_mat)
move_to_collection(back_wall, living_room)

# Left wall
bpy.ops.mesh.primitive_plane_add(size=8, location=(-4, 0, 2), rotation=(0, math.radians(90), 0))
left_wall = bpy.context.active_object
left_wall.name = "LeftWall"
left_wall.data.materials.append(wall_mat)
move_to_collection(left_wall, living_room)

# Rug (cylinder, very flat)
bpy.ops.mesh.primitive_cylinder_add(radius=2.5, depth=0.05, location=(0, 0, 0.025))
rug = bpy.context.active_object
rug.name = "Rug"
rug.data.materials.append(rug_mat)
move_to_collection(rug, living_room)

# Sofa
sofa_collection = bpy.data.collections.new("Sofa")
living_room.children.link(sofa_collection)

# Sofa base
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, -1.5, 0.25))
sofa_base = bpy.context.active_object
sofa_base.name = "SofaBase"
sofa_base.scale = (2.0, 0.8, 0.25)
sofa_base.data.materials.append(fabric_mat)
move_to_collection(sofa_base, sofa_collection)

# Sofa back
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, -1.9, 0.5))
sofa_back = bpy.context.active_object
sofa_back.name = "SofaBack"
sofa_back.scale = (2.0, 0.15, 0.5)
sofa_back.data.materials.append(fabric_mat)
move_to_collection(sofa_back, sofa_collection)

# Sofa left arm
bpy.ops.mesh.primitive_cube_add(size=1, location=(-1.0, -1.5, 0.35))
sofa_arm_l = bpy.context.active_object
sofa_arm_l.name = "SofaArmLeft"
sofa_arm_l.scale = (0.15, 0.8, 0.35)
sofa_arm_l.data.materials.append(fabric_mat)
move_to_collection(sofa_arm_l, sofa_collection)

# Sofa right arm
bpy.ops.mesh.primitive_cube_add(size=1, location=(1.0, -1.5, 0.35))
sofa_arm_r = bpy.context.active_object
sofa_arm_r.name = "SofaArmRight"
sofa_arm_r.scale = (0.15, 0.8, 0.35)
sofa_arm_r.data.materials.append(fabric_mat)
move_to_collection(sofa_arm_r, sofa_collection)

# Sofa cushions (3)
for i, x in enumerate([-0.6, 0, 0.6]):
    bpy.ops.mesh.primitive_cube_add(size=1, location=(x, -1.45, 0.55))
    cushion = bpy.context.active_object
    cushion.name = f"SofaCushion{i+1}"
    cushion.scale = (0.55, 0.6, 0.1)
    cushion.data.materials.append(fabric_mat)
    move_to_collection(cushion, sofa_collection)

# Coffee Table
table_collection = bpy.data.collections.new("CoffeeTable")
living_room.children.link(table_collection)

# Table top
bpy.ops.mesh.primitive_cylinder_add(radius=0.5, depth=0.05, location=(0, 0, 0.45))
table_top = bpy.context.active_object
table_top.name = "TableTop"
table_top.data.materials.append(tabletop_mat)
move_to_collection(table_top, table_collection)

# Table legs (4)
for dx, dy in [(-0.35, -0.35), (0.35, -0.35), (-0.35, 0.35), (0.35, 0.35)]:
    bpy.ops.mesh.primitive_cylinder_add(radius=0.02, depth=0.45, location=(dx, dy, 0.225))
    leg = bpy.context.active_object
    leg.name = f"TableLeg_{dx}_{dy}"
    leg.data.materials.append(metal_mat)
    move_to_collection(leg, table_collection)

# Floor Lamp
lamp_collection = bpy.data.collections.new("FloorLamp")
living_room.children.link(lamp_collection)

# Lamp base
bpy.ops.mesh.primitive_cylinder_add(radius=0.15, depth=0.03, location=(-2.5, -1.5, 0.015))
lamp_base = bpy.context.active_object
lamp_base.name = "LampBase"
lamp_base.data.materials.append(metal_mat)
move_to_collection(lamp_base, lamp_collection)

# Lamp pole
bpy.ops.mesh.primitive_cylinder_add(radius=0.02, depth=1.5, location=(-2.5, -1.5, 0.78))
lamp_pole = bpy.context.active_object
lamp_pole.name = "LampPole"
lamp_pole.data.materials.append(metal_mat)
move_to_collection(lamp_pole, lamp_collection)

# Lamp shade (cone)
bpy.ops.mesh.primitive_cone_add(radius1=0.25, radius2=0.15, depth=0.3, location=(-2.5, -1.5, 1.6))
lamp_shade = bpy.context.active_object
lamp_shade.name = "LampShade"
lamp_shade.data.materials.append(shade_mat)
move_to_collection(lamp_shade, lamp_collection)

# Point light inside lamp shade
bpy.ops.object.light_add(type='POINT', location=(-2.5, -1.5, 1.55))
lamp_light = bpy.context.active_object
lamp_light.name = "LampLight"
lamp_light.data.energy = 50
lamp_light.data.color = (1.0, 0.9, 0.7)
move_to_collection(lamp_light, lamp_collection)

# Add a sun light for ambient
bpy.ops.object.light_add(type='SUN', location=(0, 0, 5), rotation=(math.radians(45), 0, 0))
sun = bpy.context.active_object
sun.name = "Sun"
sun.data.energy = 2
move_to_collection(sun, living_room)

# Set up camera
bpy.ops.object.camera_add(location=(0, 3, 1.5), rotation=(math.radians(70), 0, 0))
camera = bpy.context.active_object
camera.name = "Camera"
bpy.context.scene.camera = camera
move_to_collection(camera, living_room)

print("Living room created with sofa, coffee table, floor lamp, and rug.")
```
