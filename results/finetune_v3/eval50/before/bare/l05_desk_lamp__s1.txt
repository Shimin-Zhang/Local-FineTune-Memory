```python
import bpy
import math
import mathutils

# Clear existing scene
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# Remove default lights
for obj in list(bpy.data.objects):
    if obj.type == 'LIGHT':
        bpy.data.objects.remove(obj, do_unlink=True)

# Set world to black (dark room)
world = bpy.data.worlds.get("World")
if world is None:
    world = bpy.data.worlds.new("World")
bpy.context.scene.world = world
world.use_nodes = True
bg_node = world.node_tree.nodes.get("Background")
if bg_node:
    bg_node.inputs[0].default_value = (0.0, 0.0, 0.0, 1.0)
    bg_node.inputs[1].default_value = 1.0

# Create materials
def create_material(name, color, metallic=0.0, roughness=0.5, emission=None, emission_strength=0.0):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get("Principled BSDF")
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Metallic"].default_value = metallic
    bsdf.inputs["Roughness"].default_value = roughness
    if emission is not None:
        bsdf.inputs["Emission Color"].default_value = emission
        bsdf.inputs["Emission Strength"].default_value = emission_strength
    return mat

wood_mat = create_material("Wood", (0.35, 0.2, 0.1, 1.0), metallic=0.0, roughness=0.6)
metal_mat = create_material("Metal", (0.4, 0.4, 0.45, 1.0), metallic=0.9, roughness=0.3)
shade_mat = create_material("LampShade", (0.9, 0.85, 0.7, 1.0), metallic=0.0, roughness=0.4, emission=(1.0, 0.9, 0.6, 1.0), emission_strength=2.0)
glass_mat = create_material("Glass", (1.0, 1.0, 1.0, 1.0), metallic=0.0, roughness=0.1)
bulb_mat = create_material("Bulb", (1.0, 1.0, 0.8, 1.0), metallic=0.0, roughness=0.2, emission=(1.0, 0.95, 0.7, 1.0), emission_strength=5.0)

# Create desk (table)
# Desk top
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0.75))
desk_top = bpy.context.active_object
desk_top.scale = (1.2, 0.6, 0.03)
desk_top.data.materials.append(wood_mat)
bpy.ops.object.transform_apply(scale=True)

# Desk legs (4 legs)
leg_positions = [(-1.1, -0.55), (1.1, -0.55), (-1.1, 0.55), (1.1, 0.55)]
for x, y in leg_positions:
    bpy.ops.mesh.primitive_cylinder_add(radius=0.03, depth=0.75, location=(x, y, 0.375))
    leg = bpy.context.active_object
    leg.data.materials.append(wood_mat)

# Create lamp on desk
lamp_base_x = -0.6
lamp_base_y = 0.2
lamp_z = 0.78  # top of desk

# Lamp base (cylinder)
bpy.ops.mesh.primitive_cylinder_add(radius=0.12, depth=0.04, location=(lamp_base_x, lamp_base_y, lamp_z + 0.02))
lamp_base = bpy.context.active_object
lamp_base.data.materials.append(metal_mat)

# Lamp arm (vertical cylinder)
arm_height = 0.5
bpy.ops.mesh.primitive_cylinder_add(radius=0.015, depth=arm_height, location=(lamp_base_x, lamp_base_y, lamp_z + 0.04 + arm_height/2))
lamp_arm = bpy.context.active_object
lamp_arm.data.materials.append(metal_mat)

# Lamp shade (cone)
shade_z = lamp_z + 0.04 + arm_height
bpy.ops.mesh.primitive_cone_add(radius1=0.15, radius2=0.05, depth=0.15, location=(lamp_base_x, lamp_base_y, shade_z + 0.075))
lamp_shade = bpy.context.active_object
lamp_shade.data.materials.append(shade_mat)

# Lamp bulb (sphere)
bulb_z = shade_z + 0.05
bpy.ops.mesh.primitive_uv_sphere_add(radius=0.04, location=(lamp_base_x, lamp_base_y, bulb_z))
lamp_bulb = bpy.context.active_object
lamp_bulb.data.materials.append(bulb_mat)

# Add point light inside the lamp shade
light_data = bpy.data.lights.new(name="LampLight", type='POINT')
light_data.energy = 50  # Watts
light_data.color = (1.0, 0.9, 0.6)
light_data.shadow_soft_size = 0.05
light_obj = bpy.data.objects.new(name="LampLightObj", object_data=light_data)
bpy.context.collection.objects.link(light_obj)
light_obj.location = (lamp_base_x, lamp_base_y, shade_z + 0.05)

# Create floor
bpy.ops.mesh.primitive_plane_add(size=20, location=(0, 0, 0))
floor = bpy.context.active_object
floor_mat = create_material("Floor", (0.1, 0.1, 0.1, 1.0), metallic=0.0, roughness=0.8)
floor.data.materials.append(floor_mat)

# Create walls (optional, to contain light)
# Back wall
bpy.ops.mesh.primitive_plane_add(size=20, location=(0, -5, 5))
back_wall = bpy.context.active_object
bpy.ops.object.transform_apply(rotation=True)
back_wall.rotation_euler = (math.radians(90), 0, 0)
wall_mat = create_material("Wall", (0.15, 0.15, 0.15, 1.0), metallic=0.0, roughness=0.9)
back_wall.data.materials.append(wall_mat)

# Set up camera
bpy.ops.object.camera_add(location=(2.5, 2.5, 1.5))
camera = bpy.context.active_object
bpy.context.scene.camera = camera

# Point camera at desk/lamp
target = mathutils.Vector((0, 0, 0.75))
direction = target - camera.location
rot_quat = direction.to_track_quat('-Z', 'Y')
camera.rotation_euler = rot_quat.to_euler()

# Set up render settings
scene = bpy.context.scene
scene.render.engine = 'CYCLES'
scene.cycles.samples = 128
scene.cycles.use_denoising = True
scene.render.resolution_x = 1920
scene.render.resolution_y = 1080
scene.render.resolution_percentage = 100

# Set film to not use transparent
scene.render.film_transparent = False

# Enable light linking or just rely on the single light source
# Make sure only the lamp light illuminates the scene
# Remove any other lights (already done)

print("Scene created: Desk with lamp in dark room, lit only by the lamp.")
```
