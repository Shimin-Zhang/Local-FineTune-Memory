import bpy, bmesh, math, random
from mathutils import Vector
random.seed(1325)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
def tone(c, e=0, r=.8):
    m = bpy.data.materials.new("t"); m.use_nodes = True
    n = next(x for x in m.node_tree.nodes if x.type == "BSDF_PRINCIPLED")
    n.inputs["Base Color"].default_value = (*c, 1); n.inputs["Roughness"].default_value = r
    if e: n.inputs["Emission Color"].default_value = (*c, 1); n.inputs["Emission Strength"].default_value = e
    return m
floor, wall, desk, drawer, lampm, shade, paper, book1, book2, book3, ink, brass = (tone((.12, .08, .06)), tone((.14, .12, .15)), tone((.28, .15, .08)), tone((.2, .1, .06)),
    tone((.1, .1, .12), r=.4), tone((.9, .55, .15), 3), tone((.7, .68, .6)), tone((.5, .1, .1)), tone((.1, .25, .4)), tone((.55, .45, .15)), tone((.02, .02, .03)), tone((.7, .5, .15), r=.3))
def make(bm, m, p=(0, 0, 0), r=(0, 0, 0)):
    me = bpy.data.meshes.new("p"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("p", me); o.location = p; o.rotation_euler = r
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def box(p, s, m, rz=0):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * s[0], v.co.y * s[1], (v.co.z + .5) * s[2]))
    return make(bm, m, p, (0, 0, rz))
def cone(p, r, h, n, m, r2=None, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return make(bm, m, p, rot)
def ball(p, r, m, s=(1, 1, 1)):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co = Vector((v.co.x * s[0], v.co.y * s[1], v.co.z * s[2]))
    return make(bm, m, p)
def lamp(kind, p, c, w, aim=None):
    L = bpy.data.lights.new("l", kind); L.color = c; L.energy = w
    o = bpy.data.objects.new("l", L); o.location = p; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
# floor slab and two walls
bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
for v in bm.verts:
    v.co = Vector((v.co.x * 15, v.co.y * 12, v.co.z * 2.2 - 1.1))
    if v.co.z < 0: v.co = Vector((v.co.x * .6 + random.uniform(-.8, .8), v.co.y * .6 + random.uniform(-.8, .8), v.co.z - random.uniform(0, 1.4)))
make(bm, floor)

for sx in (-2.2, 2.2):

# chair
box((-1.5, -3.6, 0), (2.2, 2.2, .5), book1); box((-1.5, -4.5, .5), (2.2, .3, 2.6), book1)
for sx in (-1, 1):
    for sy in (-1, 1): cone((-1.5 + sx * .9, -3.6 + sy * .9, 0), .12, .5, 5, lampm)
# the lamp: heavy base, stem, arm, tilted shade over the papers

cam = bpy.data.cameras.new("c"); cam.lens = 80; cam.clip_end = 400
co = bpy.data.objects.new("c", cam); sc.collection.objects.link(co)
t = Vector((0, 0, 2.5)); co.location = t + Vector((10, -36, 22))
co.rotation_euler = (t - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
