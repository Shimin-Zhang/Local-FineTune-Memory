import bpy, bmesh, random, math
from mathutils import Vector
random.seed(805)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
def tint(c, e=0):
    m = bpy.data.materials.new("t"); m.use_nodes = True
    n = next(x for x in m.node_tree.nodes if x.type == "BSDF_PRINCIPLED")
    n.inputs["Base Color"].default_value = (*c, 1); n.inputs["Roughness"].default_value = .8
    if e: n.inputs["Emission Color"].default_value = (*c, 1); n.inputs["Emission Strength"].default_value = e
    return m
sea, shore, plank, post, red, white, hull, hull2, crate, rope, lamp, glass, rock = (tint((.04, .13, .2)), tint((.2, .17, .12)), tint((.35, .22, .12)), tint((.18, .12, .08)), tint((.6, .08, .06)),
    tint((.7, .68, .62)), tint((.1, .25, .4)), tint((.5, .1, .08)), tint((.4, .28, .14)), tint((.5, .4, .25)), tint((1, .75, .3), 25), tint((1, .8, .4), 10), tint((.15, .15, .17)))
def make(bm, m, p=(0, 0, 0), r=(0, 0, 0)):
    me = bpy.data.meshes.new("m"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("m", me); o.location, o.rotation_euler = p, r
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def box(p, s, m, r=(0, 0, 0), top=1.0):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts:
        k = top if v.co.z > 0 else 1
        v.co = Vector((v.co.x * s[0] * k, v.co.y * s[1] * k, (v.co.z + .5) * s[2]))
    return make(bm, m, p, r)
def tube(p, r, h, n, m, r2=None, r_=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return make(bm, m, p, r_)
def line(a, b, t, m):
    a, b = Vector(a), Vector(b); d = b - a; q = d.to_track_quat('Z', 'Y')
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = q @ Vector((v.co.x * t, v.co.y * t, (v.co.z + .5) * d.length)) + a
    return make(bm, m)
def light(kind, p, c, w, aim=None):
    l = bpy.data.lights.new("l", kind); l.color = c; l.energy = w
    o = bpy.data.objects.new("l", l); o.location = p; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
# base: sea slab with a raised shore at the front
bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
for v in bm.verts:
    if v.co.z < 0: v.co.x *= .85; v.co.y *= .85; v.co.z -= random.uniform(0, .8)
    v.co = Vector((v.co.x * 26, v.co.y * 22, v.co.z * 3 - 1.5))
make(bm, sea)
box((0, 7.5, -.05), (26, 7, 1.0), shore, top=.96)
for i in range(6): box((-12 + i * 4.5, 3.9, 0), (3.5, .5, .12), plank)
# pier running out into the water with piles
box((0, -1, .1), (3.4, 11, .4), plank)
for i in range(10):
    for s in (-1, 1): tube((s * 1.6, 3.5 - i * 1.1, -.5), .18, 1.1, 5, post)
box((0, -6.4, .1), (3.8, 1.0, .4), plank)
for s in (-1, 1): tube((s * 1.8, -6.4, -.5), .2, 1.2, 5, post)
box((-1.6, -6.4, .5), (.12, 1.0, .12), post); box((1.6, -6.4, .5), (.12, 1.0, .12), post)
# two moored boats: hull, deck, cabin, mast
for x, y, m, a in ((-5.5, -3.5, hull, .15), (5.0, -6.0, hull2, -.2)):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts:
        v.co = Vector((v.co.x * 3.2, v.co.y * 6.2, v.co.z * 1.4))
        if v.co.z < 0: v.co.x *= .6
        if abs(v.co.x) < 1.7 and v.co.z > 0: v.co.z += .1
    make(bm, m, (x, y, .3), (0, 0, a))
    box((x, y + 1.2, 1.0), (2.2, 2.0, 1.2), white, (0, 0, a), .9); box((x, y + 1.2, 2.2), (2.4, 2.2, .15), post, (0, 0, a))
    box((x, y + 1.2, 1.6), (.1, .8, .5), glass, (0, 0, a))
    line((x, y - 1.2, 1.2), (x, y - 1.2, 5.0), .1, post); line((x, y - 1.2, 4.2), (x + 1.8, y - 1.2, 3.0), .06, rope)
    light("POINT", (x, y + 1.2, 1.9), (1, .7, .3), 120)
# crates and barrels on the shore and pier
for i, (x, y) in enumerate(((-8, 6), (-6.5, 6.5), (-7.2, 5), (-3, 7.5), (2, 7), (7, 5.5), (8.5, 6.5), (-1, -4.5), (1, -2.5))):
    box((x, y, 0 if y > 3 else .5), (1.3, 1.3, 1.1), crate, (0, 0, i * .4), .95)
    if i % 3 == 0: box((x + .1, y, 1.1), (1.0, 1.0, .9), crate, (0, 0, i * .4 + .5))
for x, y in ((-9, 4.5), (9.5, 4.5)): tube((x, y, 0), .6, 1.2, 8, post, .55)
# lighthouse on a rocky headland at the back right
tube((10, 8.5, 0), 3.0, 1.4, 7, rock, 2.4)
tube((10, 8.5, 1.2), 1.7, 8, 8, white, 1.1)
for z in (3.5, 6.0): tube((10, 8.5, z), 1.45 - z * .03, 1.2, 8, red, 1.35 - z * .03)
tube((10, 8.5, 9.2), 1.3, .3, 8, post); tube((10, 8.5, 9.5), 1.0, 1.4, 8, lamp)
tube((10, 8.5, 10.9), 1.4, .8, 8, red, .1)
light("POINT", (10, 8.5, 10.0), (1, .75, .3), 2500)
for i in range(5):
    a = i * 1.3; tube((10 + math.cos(a) * 4.2, 8.5 + math.sin(a) * 4.2, 0), .5, .7, 5, rock, .3)
# bollards with rope
for x in (-1, 1):
    tube((x * 2.8, 2.0, 0), .25, .6, 6, post); line((x * 2.8, 2.0, .5), (x * 1.6, 1.5, .55), .05, rope)
# a small net basket and a lantern on the pier end
tube((1.0, -5.5, .5), .4, .6, 6, rope, .35); box((-.8, -5.6, .5), (.4, .4, .5), lamp)
light("POINT", (-.8, -5.6, 1.2), (1, .7, .3), 150)
light("SUN", (-26, -16, 12), (.5, .65, 1), 4.5, (0, 0, 2))
light("SUN", (22, 22, 10), (1, .5, .4), 2.2, (0, 0, 2))
w = bpy.data.worlds.new("w"); sc.world = w; w.use_nodes = True
next(n for n in w.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.015, .02, .045, 1)
cam = bpy.data.cameras.new("c"); cam.lens = 80; cam.clip_end = 400
co = bpy.data.objects.new("c", cam); sc.collection.objects.link(co)
t = Vector((0, 1, 3)); co.location = t + Vector((-14, -66, 44))
co.rotation_euler = (t - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
