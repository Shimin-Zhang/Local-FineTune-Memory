import bpy, bmesh, math, random
from mathutils import Vector as V
random.seed(691)
for o in list(bpy.data.objects): bpy.data.objects.remove(o, do_unlink=True)
sc = bpy.context.scene
def leafy(c, em=0):
    m = bpy.data.materials.new("l"); m.use_nodes = True
    n = next(x for x in m.node_tree.nodes if x.type == "BSDF_PRINCIPLED")
    n.inputs["Base Color"].default_value = (*c, 1); n.inputs["Roughness"].default_value = .55
    if em: n.inputs["Emission Color"].default_value = (*c, 1); n.inputs["Emission Strength"].default_value = em
    return m
soil, soil2, rind, stripe, flesh, seed, stem, leaf, fruit, pink, bulb = (
    leafy((.12, .08, .05)), leafy((.09, .06, .04)), leafy((.05, .2, .08)), leafy((.12, .4, .1)), leafy((.9, .1, .2)), leafy((.03, .03, .03)), leafy((.2, .14, .07)),

def add(bm, m, loc=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("m"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("m", me); o.location = loc; o.rotation_euler = rot
    o.data.materials.append(m); sc.collection.objects.link(o); return o
def orb(r, m, loc, sq=1.0, jit=0):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=2, radius=r)
    for v in bm.verts:
        v.co.z *= sq
        if jit: v.co += V((random.uniform(-jit, jit), random.uniform(-jit, jit), random.uniform(-jit, jit)))
    return add(bm, m, loc)
def cyl(r, r2, h, n, m, loc, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return add(bm, m, loc, rot)
def wedge(r0, r1, z0, z1, a0, a1, m, jit=0):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=8, radius1=r0, radius2=r1, depth=z1 - z0)
    for v in bm.verts: v.co.z += (z0 + z1) / 2; v.co.x += random.uniform(-jit, jit); v.co.y += random.uniform(-jit, jit)
    bmesh.ops.bisect_plane(bm, geom=bm.verts[:] + bm.edges[:] + bm.faces[:], plane_co=(0, 0, 0), plane_no=(math.sin((a0 + a1) / 2), -math.cos((a0 + a1) / 2), 0), clear_inner=True)
    return add(bm, m)
def glim(kind, c, w, loc, aim=None):
    L = bpy.data.lights.new("g", kind); L.color = c; L.energy = w
    o = bpy.data.objects.new("g", L); o.location = loc; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (V(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
# a mound of dark soil, rough underneath
bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=10, radius1=5, radius2=11, depth=3)
for v in bm.verts:

    if v.co.z < -1: v.co += V((random.uniform(-1.2, 1.2), random.uniform(-1.2, 1.2), random.uniform(-1.3, .2)))
    else: v.co.z += random.uniform(-.12, .12)
add(bm, soil)

for k, (r0, r1, z0, z1, m) in enumerate(((3.4, 3.62, 0, 3.4, rind), (3.2, 3.4, 0, 3.4, stripe), (3.15, .05, 0, 3.4, flesh))):
    wedge(r0, r1, z0, z1, -math.pi / 2, -math.pi / 2 + math.pi / 2.3, m, .02)
for i in range(7):
    a = -math.pi / 2 + .25 + i * .22; r = .8 + (i % 3) * .85

for i in range(7):

co.rotation_euler = (t - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
