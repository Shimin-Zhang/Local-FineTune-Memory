import bpy, bmesh, random, math
from mathutils import Vector
random.seed(663)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
def stone(r, g, b, e=0):
    m = bpy.data.materials.new("s"); m.use_nodes = True
    p = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    p.inputs["Base Color"].default_value = (r, g, b, 1); p.inputs["Roughness"].default_value = .9
    if e: p.inputs["Emission Color"].default_value = (r, g, b, 1); p.inputs["Emission Strength"].default_value = e
    return m
def mk(bm, m, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("k"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("k", me); o.location = at; o.rotation_euler = rot
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def blk(w, d, h, m, at, rot=(0, 0, 0), top=1.0):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts:
        k = top if v.co.z > 0 else 1
        v.co = Vector((v.co.x * w * k, v.co.y * d * k, (v.co.z + .5) * h))
    return mk(bm, m, at, rot)
def cyl(r, r2, h, n, m, at, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return mk(bm, m, at, rot)
def lit(c, w, at):
    L = bpy.data.lights.new("l", "POINT"); L.color = c; L.energy = w
    o = bpy.data.objects.new("l", L); o.location = at; sc.collection.objects.link(o)
grass, wall, wall2, dark = stone(.07, .12, .08), stone(.34, .33, .36), stone(.25, .25, .29), stone(.04, .04, .06)
roof, water, bank, wood = stone(.3, .08, .07), stone(.05, .14, .25), stone(.14, .12, .1), stone(.22, .13, .07)
fire, win, gold, flag = stone(1, .5, .1, 16), stone(1, .65, .25, 8), stone(.8, .6, .15), stone(.5, .08, .08)
# castle island: grassy round, moat ring around it
cyl(15, 15, 1.0, 10, grass, (0, 0, -1.0)); cyl(15.2, 14, .2, 10, bank, (0, 0, -.2))
bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=10, radius1=15, radius2=15, depth=4.5)
for v in bm.verts:
    if v.co.z < 0: v.co += Vector((random.uniform(-1.5, 1.5), random.uniform(-1.5, 1.5), random.uniform(-1.5, .3)))
mk(bm, dark, (0, 0, -3.2))
cyl(10.2, 10.2, .25, 10, water, (0, 0, .02))
# curtain walls with merlons, gate gap in front
for i in range(4):
    a = i * math.pi / 2 + math.pi / 4
    blk(11.0, 1.2, 3.0, wall, (math.cos(a) * 7.0, math.sin(a) * 7.0, 0), (0, 0, a + math.pi / 2))
    for k in range(7):
        t = (k - 3) * 1.5
        blk(.8, 1.3, .8, wall2, (math.cos(a) * 7.0 - math.sin(a) * t, math.sin(a) * 7.0 + math.cos(a) * t, 3.0), (0, 0, a + math.pi / 2))
# four round towers with conical roofs
for a in (math.pi / 4, 3 * math.pi / 4, 5 * math.pi / 4, 7 * math.pi / 4):
    x, y = math.cos(a) * 7.0, math.sin(a) * 7.0
    cyl(1.9, 1.7, 5.5, 8, wall, (x, y, 0)); cyl(1.95, 1.95, .5, 8, wall2, (x, y, 4.2))
    cyl(2.5, .15, 3.2, 8, roof, (x, y, 5.5))
    blk(.5, .1, .9, win, (x - math.cos(a) * 1.75, y - math.sin(a) * 1.75, 3.6), (0, 0, a))
    lit((1, .6, .2), 120, (x - math.cos(a) * 3, y - math.sin(a) * 3, 3.6))
# gatehouse with arch, two towers and portcullis
for s in (-1, 1):
    cyl(1.5, 1.4, 5.0, 8, wall, (s * 3.2, -7.0, 0)); cyl(2.0, .15, 2.6, 8, roof, (s * 3.2, -7.0, 5.0))
    blk(.5, .1, 1.0, win, (s * 3.2, -8.45, 3.0)); lit((1, .6, .2), 120, (s * 3.2, -9.4, 3.0))
blk(3.6, 1.4, 4.0, wall2, (0, -7.0, 0)); blk(3.2, .2, 2.2, dark, (0, -7.8, 0)); blk(2.4, .15, 1.6, wood, (0, -7.9, 0))
blk(3.6, 1.8, .5, wall, (0, -7.0, 4.0)); blk(4.4, 2.0, .5, wall2, (0, -7.0, 5.0))
for s in (-1, 1): blk(.1, .1, 1.0, wood, (s * 1.2, -8.0, 1.6))
# drawbridge planks over the moat
for k in range(5): blk(3.0, .8, .12, wood, (0, -8.6 - k * .85, .35 - k * .02))
for s in (-1, 1): blk(.15, 4.6, .15, wood, (s * 1.5, -10.4, .5))
# keep with lit windows and flag
blk(5.0, 5.0, 7.0, wall, (0, 3.0, 0), (0, 0, 0), .9); blk(5.4, 5.4, .5, wall2, (0, 3.0, 7.0))
for z in (2.5, 4.5, 6.0): blk(.6, .1, 1.0, win, (0, .4, z))
blk(.1, .1, 2.5, wood, (0, 3.0, 7.4)); blk(1.4, .06, .8, flag, (.7, 3.0, 8.6))
lit((1, .6, .2), 450, (0, -.8, 3.0))
# courtyard: well, braziers, barrels, tent
cyl(.7, .7, .8, 8, wall2, (0, -1.0, 0)); blk(.1, .1, 1.0, wood, (-.5, -1.0, .8)); blk(.1, .1, 1.0, wood, (.5, -1.0, .8)); blk(1.2, .1, .1, wood, (0, -1.0, 1.8))
for x, y in ((-2.5, -3.5), (2.5, -3.5)):
    cyl(.15, .15, 1.2, 5, dark, (x, y, 0)); cyl(.5, .3, .4, 6, fire, (x, y, 1.2)); lit((1, .5, .1), 300, (x, y, 1.8))
for x, y in ((3.5, 1.5), (3.9, 2.4), (-3.5, 2.0)): cyl(.4, .4, .7, 7, wood, (x, y, 0))
# torches by the gate and a small dock
for s in (-1, 1): blk(.1, .1, 2.0, wood, (s * 2.0, -9.2, 0)); cyl(.2, .15, .4, 5, fire, (s * 2.0, -9.2, 2.0)); lit((1, .5, .1), 200, (s * 2.0, -9.2, 2.2))
blk(4.0, 1.2, .2, wood, (7.5, -10.5, .2), (0, 0, .4))
for c, w, p in (((.55, .65, 1), 4.5, (-24, -14, 10)), ((.8, .35, .7), 2.2, (22, 20, 12))):
    L = bpy.data.lights.new("s", "SUN"); L.color = c; L.energy = w
    o = bpy.data.objects.new("s", L); o.location = p; sc.collection.objects.link(o)
    o.rotation_euler = (Vector((0, 0, 2)) - o.location).to_track_quat('-Z', 'Y').to_euler()
w = bpy.data.worlds.new("w"); sc.world = w; w.use_nodes = True
next(n for n in w.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.02, .015, .04, 1)
cam = bpy.data.cameras.new("c"); cam.lens = 80; cam.clip_end = 500
co = bpy.data.objects.new("c", cam); sc.collection.objects.link(co)
t = Vector((0, 0, 2.5)); co.location = t + Vector((14, -60, 38))
co.rotation_euler = (t - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
