import bpy, bmesh, math, random
from mathutils import Vector as V
random.seed(757)
for o in list(bpy.data.objects): bpy.data.objects.remove(o, do_unlink=True)
sc = bpy.context.scene
def fabric(c, glow=0, r=.9):
    m = bpy.data.materials.new("f"); m.use_nodes = True
    n = next(x for x in m.node_tree.nodes if x.type == "BSDF_PRINCIPLED")
    n.inputs["Base Color"].default_value = (*c, 1); n.inputs["Roughness"].default_value = r
    if glow:
        n.inputs["Emission Color"].default_value = (*c, 1); n.inputs["Emission Strength"].default_value = glow
    return m
lawn, lawn2, red, cream = fabric((.1, .22, .08)), fabric((.07, .16, .06)), fabric((.6, .08, .08)), fabric((.7, .65, .5))
wicker, wicker2, cloth, apple = fabric((.5, .3, .1)), fabric((.35, .18, .06)), fabric((.1, .3, .55)), fabric((.8, .1, .08))
leaf, trunk, flame, jar = fabric((.12, .3, .1)), fabric((.2, .12, .07)), fabric((1, .5, .1), 16), fabric((.2, .5, .3), .4)
def build(bm, m, loc=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("b"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("b", me); o.location = loc; o.rotation_euler = rot
    o.data.materials.append(m); sc.collection.objects.link(o); return o
def box(m, x, y, z, w, d, h, rz=0):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = V((v.co.x * w, v.co.y * d, (v.co.z + .5) * h))
    return build(bm, m, (x, y, z), (0, 0, rz))
def cone(m, x, y, z, r, h, n=6, r2=None):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return build(bm, m, (x, y, z))
def orb(m, x, y, z, r, sq=1, jit=.12):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co = V((v.co.x, v.co.y, v.co.z * sq)) * random.uniform(1 - jit, 1 + jit)
    return build(bm, m, (x, y, z))
def lamp(kind, c, e, loc, aim=None):
    L = bpy.data.lights.new("l", kind); L.color = c; L.energy = e
    o = bpy.data.objects.new("l", L); o.location = loc; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (V(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()

bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=9, radius1=6, radius2=12, depth=3.2)
for v in bm.verts:

    v.co.z -= 1.6
build(bm, lawn)

for k in range(5): box(cloth, bx + 1.35 * math.cos(k * 1.256) * .9, by + 1.35 * math.sin(k * 1.256) * .9, 1.5, .3, .3, .25, k * 1.256)
box(cloth, bx + 1.4, by, 1.5, .15, .15, 1.0)

for (x, y) in ((2.4, -1.2), (3.4, .2)):
