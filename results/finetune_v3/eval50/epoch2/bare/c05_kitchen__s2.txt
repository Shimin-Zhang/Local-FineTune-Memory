import bpy, bmesh, math, random
from mathutils import Vector
random.seed(1230)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
def paint(r, g, b, em=0):
    m = bpy.data.materials.new("p"); m.use_nodes = True
    n = next(x for x in m.node_tree.nodes if x.type == "BSDF_PRINCIPLED")
    n.inputs["Base Color"].default_value = (r, g, b, 1); n.inputs["Roughness"].default_value = .7
    if em: n.inputs["Emission Color"].default_value = (r, g, b, 1); n.inputs["Emission Strength"].default_value = em
    return m
tileA, tileB, wall, wood = paint(.22, .24, .26), paint(.12, .13, .15), paint(.3, .22, .2), paint(.3, .17, .09)
cabinet, counter, steel, fridge = paint(.12, .22, .2), paint(.55, .52, .48), paint(.5, .52, .55), paint(.55, .57, .6)
stool, flame, bulb, jar = paint(.7, .2, .1), paint(1, .45, .08, 16), paint(1, .85, .5, 14), paint(.2, .5, .3)
def mk(bm, m, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("k"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("k", me); o.location = at; o.rotation_euler = rot
    o.data.materials.append(m); sc.collection.objects.link(o); return o
def slab(at, s, m, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * s[0], v.co.y * s[1], (v.co.z + .5) * s[2]))
    return mk(bm, m, at, rot)
def can(at, r, h, n, m, r2=None, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return mk(bm, m, at, rot)
def lamp(kind, c, w, at, aim=None):
    L = bpy.data.lights.new("l", kind); L.color = c; L.energy = w
    o = bpy.data.objects.new("l", L); o.location = at; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
# base: a tiled floor slab with a rough tapered underside
bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
for v in bm.verts:
    if v.co.z < 0: v.co.x *= .8; v.co.y *= .8; v.co += Vector((random.uniform(-.5, .5), random.uniform(-.5, .5), random.uniform(-1, 0)))
    v.co = Vector((v.co.x * 14, v.co.y * 12, v.co.z * 2.4 - 1.2))
mk(bm, tileB)
for i in range(7):
    for j in range(6): slab((-6 + i * 2, -5 + j * 2, 0), (1.96, 1.96, .06), tileA if (i + j) % 2 else tileB)
# two walls: back and left, with a window on the back wall
slab((0, 6, 0), (14, .5, 7), wall); slab((-7, 0, 0), (.5, 12, 7), wall)
slab((0, 5.7, 3.6), (4.4, .2, 2.6), paint(.1, .3, .55)); slab((0, 5.6, 3.6), (4.6, .3, .2), counter); slab((0, 5.6, 6.2), (4.6, .3, .2), counter)
slab((0, 5.6, 4.9), (4.6, .25, .12), counter); slab((0, 5.6, 4.9), (.12, .25, 2.6), counter)
# long counter along the back wall: base cabinets, worktop, three uppers
slab((-1, 4.7, 0), (9.4, 1.8, 1.5), cabinet); slab((-1, 4.7, 1.5), (9.8, 2.0, .18), counter)
for i in range(3): slab((-4.2 + i * 3.1, 4.7, 0), (.08, 1.82, 1.5), wood)
for x in (-3.5, -.5, 2.5): slab((x, 5.4, 3.6), (2.6, .9, 1.5), cabinet)
# sink with faucet and a stove with burners and a pot
slab((-3.5, 4.7, 1.68), (1.8, 1.2, .12), steel); slab((-3.5, 4.7, 1.8), (1.4, .9, .06), steel)
can((-3.5, 5.2, 1.68), .08, .8, 6, steel); can((-3.5, 5.0, 2.4), .08, .5, 6, steel, rot=(math.pi / 2, 0, 0))
slab((1.5, 4.7, 1.68), (2.6, 1.6, .1), steel)
for x, y in ((.9, 4.4), (2.1, 4.4), (.9, 5.0), (2.1, 5.0)): can((x, y, 1.78), .38, .05, 8, paint(.05, .05, .06))
for x, y in ((.9, 4.4), (2.1, 5.0)): can((x, y, 1.83), .3, .06, 8, flame)
can((2.1, 4.4, 1.78), .5, .6, 8, steel, .6); can((2.1, 4.4, 2.38), .55, .08, 8, steel)
lamp("POINT", (1, .45, .1), 250, (1.5, 4.2, 2.4))
# fridge at the left end of the run, with a handle and a freezer line
slab((-5.6, 4.7, 0), (2.0, 1.8, 4.4), fridge); slab((-5.6, 3.78, 2.6), (1.9, .05, .08), steel); slab((-5.6, 3.75, 1.2), (.12, .1, 1.6), steel)
# two stools on the near side of the counter, with round seats and a foot ring
for x in (-2.2, .2):
    can((x, 2.6, 1.0), .45, .15, 8, stool, .4)
    for k in range(4): can((x + .35 * math.cos(k * 1.57), 2.6 + .35 * math.sin(k * 1.57), 0), .06, 1.0, 5, steel)
    can((x, 2.6, .5), .3, .05, 8, steel, .3)
# pendant lamps over the counter, a shelf of jars on the left wall, a kettle
for x in (-2.5, 1.5):
    can((x, 3.8, 5.0), .03, 1.6, 4, steel); can((x, 3.8, 4.3), .55, .5, 7, bulb, .2)
lamp("POINT", (1, .85, .5), 600, (-2.5, 3.8, 4.3)); lamp("POINT", (1, .85, .5), 600, (1.5, 3.8, 4.3))
slab((-6.5, 0, 3.2), (.8, 6, .15), wood)
for i in range(5): can((-6.5, -1.8 + i * .9, 3.35), .22, .5, 6, jar if i % 2 else paint(.7, .5, .15))
can((-1, 5.2, 1.68), .35, .5, 7, steel, .25); can((-1, 5.2, 2.18), .1, .1, 6, steel)
lamp("SUN", (1, .75, .5), 4.5, (22, -14, 9), (0, 0, 1.5)); lamp("SUN", (.3, .6, .9), 2, (-14, 20, 10), (0, 0, 1.5))
w = bpy.data.worlds.new("w"); sc.world = w; w.use_nodes = True
next(n for n in w.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.02, .02, .035, 1)
cam = bpy.data.cameras.new("c"); cam.lens = 80; cam.clip_end = 400
co = bpy.data.objects.new("c", cam); sc.collection.objects.link(co)
t = Vector((0, 1, 2.2)); co.location = t + Vector((22, -36, 24))
co.rotation_euler = (t - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
