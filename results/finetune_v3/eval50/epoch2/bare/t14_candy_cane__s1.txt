import bpy, bmesh, random, math
from mathutils import Vector as V
random.seed(969)
for o in list(bpy.data.objects): bpy.data.objects.remove(o, do_unlink=True)
sc = bpy.context.scene
def candy(c, e=0):
    m = bpy.data.materials.new("c"); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*c, 1); b.inputs["Roughness"].default_value = .35
    if e: b.inputs["Emission Color"].default_value = (*c, 1); b.inputs["Emission Strength"].default_value = e
    return m
red, white, snow, ice, pine, gold, bulb1, bulb2, bulb3, sack = (candy((.7, .04, .05)), candy((.75, .75, .78)), candy((.5, .55, .68)), candy((.2, .4, .6)),
    candy((.03, .16, .1)), candy((.8, .55, .1)), candy((1, .1, .1), 14), candy((1, .8, .1), 14), candy((.2, .9, 1), 14), candy((.5, .08, .12)))
def part(bm, m, p=(0, 0, 0), r=(0, 0, 0)):
    me = bpy.data.meshes.new("p"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("p", me); o.location = p; o.rotation_euler = r
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def prism(p, s, m, top=1.0, rz=0):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts:
        k = top if v.co.z > 0 else 1
        v.co = V((v.co.x * s[0] * k, v.co.y * s[1] * k, (v.co.z + .5) * s[2]))
    return part(bm, m, p, (0, 0, rz))
def stick(a, b, r0, r1, m, n=8):
    a, b = V(a), V(b); d = b - a; q = d.to_track_quat('Z', 'Y')
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r0, radius2=r1, depth=d.length)
    for v in bm.verts: v.co = q @ V((v.co.x, v.co.y, v.co.z + d.length / 2)) + a
    return part(bm, m)
def lump(p, r, m, sq=.8):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co *= random.uniform(.85, 1.15); v.co.z *= sq
    return part(bm, m, p)
def shine(kind, c, w, p, aim=None):
    L = bpy.data.lights.new("s", kind); L.color = c; L.energy = w
    o = bpy.data.objects.new("s", L); o.location = p; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (V(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()

for v in bm.verts:

    if v.co.z < -1: v.co += V((random.uniform(-1.2, 1.2), random.uniform(-1.2, 1.2), random.uniform(-1.3, .2)))

for i in range(len(pts) - 1):
