import bpy, bmesh, math, random
from mathutils import Vector as V
random.seed(959)
for o in list(bpy.data.objects): bpy.data.objects.remove(o, do_unlink=True)
sc = bpy.context.scene
def stuff(rgb, em=0, rough=.5):
    m = bpy.data.materials.new("s"); m.use_nodes = True
    n = next(x for x in m.node_tree.nodes if x.type == "BSDF_PRINCIPLED")
    n.inputs["Base Color"].default_value = (*rgb, 1); n.inputs["Roughness"].default_value = rough
    if em: n.inputs["Emission Color"].default_value = (*rgb, 1); n.inputs["Emission Strength"].default_value = em
    return m
slab, slab2, marble, vein, gold, felt, lamp, glow = (stuff((.14, .1, .09)), stuff((.09, .07, .07)), stuff((.62, .63, .66)), stuff((.25, .27, .32)),
    stuff((.7, .5, .12), 0, .3), stuff((.05, .12, .1)), stuff((.05, .05, .06)), stuff((1, .7, .3), 22))
def mk(bm, m, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("m"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("m", me); o.location, o.rotation_euler = at, rot
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def lathe(m, at, pts, n=10, jit=0, rot=(0, 0, 0)):
    bm = bmesh.new(); rings = []
    for z, r in pts:
        rings.append([bm.verts.new((r * math.cos(i * 6.2832 / n) + random.uniform(-jit, jit), r * math.sin(i * 6.2832 / n) + random.uniform(-jit, jit), z)) for i in range(n)])
    for a, b in zip(rings, rings[1:]):
        for i in range(n): bm.faces.new((a[i], a[(i + 1) % n], b[(i + 1) % n], b[i]))
    bm.faces.new(rings[-1][::-1]); bm.faces.new(rings[0]); return mk(bm, m, at, rot)
def cube(m, at, s, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = V((v.co.x * s[0], v.co.y * s[1], (v.co.z + .5) * s[2]))
    return mk(bm, m, at, rot)
def lamp(at, w):
    L = bpy.data.lights.new("l", "POINT"); L.color, L.energy = (1, .7, .3), w
    o = bpy.data.objects.new("l", L); o.location = at; sc.collection.objects.link(o)
# marble base with a rough underside
bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=8, radius1=5, radius2=9.5, depth=2.2)
for v in bm.verts:
    if v.co.z < 0: v.co += V((random.uniform(-.8, .8), random.uniform(-.8, .8), random.uniform(-1, .2)))
    v.co.z -= 1.1
mk(bm, slab2)
for i in range(8): cube(slab if i % 2 else slab2, (0, 0, 0), (17 - i * .0, 17 - i * .0, .12), (0, 0, math.radians(i * 22.5 + 11)))
# chessboard of felt and marble squares, with the pawn's square glowing
for i in range(6):
    for j in range(6):
        x, y = -5 + i * 2, -5 + j * 2
        cube(felt if (i + j) % 2 else slab, (x, y, .12), (1.96, 1.96, .12))
cube(glow, (0, -1, .24), (1.9, 1.9, .02)); lamp((0, -1, 1.0), 160)
# the pawn: base, collar, tapering shaft, neck, head, collar, tip
pts = [(0, 3.0), (.35, 2.7), (2.1, 2.4), (2.3, 1.9), (1.5, 1.6), (1.2, 1.2), (1.0, .8), (.8, .5), (.75, .35), (1.3, .3), (1.7, .3), (1.9, 1.1), (1.8, 1.9), (1.1, 2.4), (1.0, 3.0), (.5, 3.3), (0, 3.4)]
lathe(marble, (0, -1, .24), pts, 10, .04)
# grey veins: thin curved strips laid along the shaft and head
for k, (z0, r0, z1, r1, a0) in enumerate(((.9, .85, 1.5, 1.2, .3), (1.4, 1.2, 2.2, 1.5, 2.2), (.5, .8, 1.0, .9, 4.0), (2.5, 1.6, 3.0, 1.0, 1.0), (1.1, .85, 1.7, 1.15, 5.2))):
    bm = bmesh.new(); a = a0
    for s in range(5):
        t = s / 4; z = z0 + (z1 - z0) * t; r = r0 + (r1 - r0) * t + .06
        a = a0 + t * 1.3
        bm.verts.new((r * math.cos(a), r * math.sin(a), z)); bm.verts.new((r * math.cos(a + .25), r * math.sin(a + .25), z + .1))
    bmesh.ops.convex_hull(bm, input=bm.verts); mk(bm, vein, (0, -1, .24))
# two small pawns fallen beside the board, a candle in a gold holder
for x, y, r in ((6.5, 2.5, 1.2), (-6.5, -2.5, 1.57)):
    lathe(marble, (x, y, .12), [(0, .9), (.1, .8), (.7, .75), (.8, .6), (.5, .5), (.4, .4), (.5, .3), (.6, .3), (.65, .7), (.55, 1.0), (.3, 1.1), (0, 1.15)], 8, 0, (math.pi / 2, 0, r))
lathe(gold, (6.2, -5.5, .12), [(0, 1.0), (.1, .9), (.5, .8), (.5, 1.0), (.9, .5), (1.0, .4)], 7)
lathe(marble, (6.2, -5.5, 1.0), [(0, .3), (.3, 1.3), (0, 1.5)], 6); lathe(glow, (6.2, -5.5, 1.5), [(0, .08), (.08, .3), (0, .35)], 5); lamp((6.2, -5.5, 2.2), 220)
for c, w, p in (((.6, .7, 1), 4.5, (-26, -14, 12)), ((1, .65, .35), 2.2, (22, 22, 10))):
    L = bpy.data.lights.new("s", "SUN"); L.color, L.energy = c, w
    o = bpy.data.objects.new("s", L); o.location = p; sc.collection.objects.link(o)
    o.rotation_euler = (V((0, 0, 1)) - o.location).to_track_quat('-Z', 'Y').to_euler()
sc.world = bpy.data.worlds.new("night"); sc.world.use_nodes = True
next(n for n in sc.world.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.01, .012, .03, 1)
cm = bpy.data.cameras.new("c"); cm.lens = 80; cm.clip_end = 500
co = bpy.data.objects.new("c", cm); sc.collection.objects.link(co)
t = V((0, -1, 1.5)); co.location = t + V((8, -38, 26))
co.rotation_euler = (t - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
