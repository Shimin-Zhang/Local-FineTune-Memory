def brick(c, e=0, r=.9):
    m = bpy.data.materials.new("b"); m.use_nodes = True
    n = next(x for x in m.node_tree.nodes if x.type == "BSDF_PRINCIPLED")
    n.inputs["Base Color"].default_value = n.inputs["Emission Color"].default_value = (*c, 1)
    n.inputs["Roughness"].default_value = r; n.inputs["Emission Strength"].default_value = e
    return m

def make(bm, m, p=(0, 0, 0), r=(0, 0, 0)):
    me = bpy.data.meshes.new("w"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("w", me); o.location, o.rotation_euler = p, r
    o.data.materials.append(m); sc.collection.objects.link(o); return o
def slab(m, x, y, z, w, d, h, rz=0, top=1):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts:
        k = top if v.co.z > 0 else 1
        v.co = Vector((v.co.x * w * k, v.co.y * d * k, (v.co.z + .5) * h))
    return make(bm, m, (x, y, z), (0, 0, rz))
def stem(m, x, y, z, r, h, n=5, r2=None):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return make(bm, m, (x, y, z))
def lamp(kind, c, e, p, aim=None):
    L = bpy.data.lights.new("l", kind); L.color = c; L.energy = e
    o = bpy.data.objects.new("l", L); o.location = p; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
