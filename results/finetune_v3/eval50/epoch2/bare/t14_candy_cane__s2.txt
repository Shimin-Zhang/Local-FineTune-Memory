import bpy, bmesh, random, math
from mathutils import Vector
random.seed(1843)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
def candy(c, e=0):
    m = bpy.data.materials.new("c"); m.use_nodes = True
    n = next(x for x in m.node_tree.nodes if x.type == "BSDF_PRINCIPLED")
    n.inputs["Base Color"].default_value = (*c, 1); n.inputs["Roughness"].default_value = .3
    if e: n.inputs["Emission Color"].default_value = (*c, 1); n.inputs["Emission Strength"].default_value = e
    return m
red, white, snow, ice = candy((.75, .04, .06)), candy((.85, .82, .8)), candy((.45, .5, .6)), candy((.3, .55, .8))
holly, berry, bulb = candy((.05, .3, .1)), candy((.8, .05, .1)), candy((1, .6, .2), 6)
def make(bm, m, p=(0, 0, 0), r=(0, 0, 0)):
    me = bpy.data.meshes.new("m"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("m", me); o.location = p; o.rotation_euler = r
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def ring(m, p, r0, r1, h, n=10, r_=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r0, radius2=r1, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return make(bm, m, p, r_)
def lump(m, p, s, r=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=1)
    for v in bm.verts: v.co = Vector((v.co.x * s[0], v.co.y * s[1], v.co.z * s[2]))
    return make(bm, m, p, r)
def light(kind, c, w, p, aim=None):
    L = bpy.data.lights.new("l", kind); L.color = c; L.energy = w
    o = bpy.data.objects.new("l", L); o.location = p; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
# snowy base with a rough underside
bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=10, radius1=5, radius2=10, depth=3)
for v in bm.verts:
    if v.co.z < 0: v.co += Vector((random.uniform(-1.2, 1.2), random.uniform(-1.2, 1.2), random.uniform(-1.5, .3)))
    else: v.co.z += random.uniform(-.1, .1)
    v.co.z -= 1.5
make(bm, snow)
# the candy cane: a J made of alternating red and white bands
R, th, w = 3.0, .5, .5
def cane(t):
    a = math.radians(t)
    return Vector((R * math.cos(a), 0, R * math.sin(a) + R))
pts = [cane(t) for t in range(0, 181, 2)] + [cane(180) + Vector((0, 0, 4.2))]
for i, (a, b) in enumerate(zip(pts, pts[1:])):
    d = b - a
    ring(red if i % 2 else white, a, w, w, d.length, 8, d.to_track_quat('Z', 'Y').to_euler())
# holly at the foot of the cane
for k in range(4):
    a = k * math.pi / 2 + .4
    lump(holly, (R * math.cos(a) * 1.0, R * math.sin(a) * 1.0 - .4, .35), (.9, .45, .12), (0, 0, a))
lump(berry, (R, -.5, .5), (.3, .3, .3)); lump(berry, (R - .5, -.9, .5), (.25, .25, .25))
# a candy-snowman beside it
for z, r in ((.8, 1.0), (2.4, .75)):
    for k in range(5):
        a = k * math.pi / 2.5
        lump(red if k % 2 else white, (7 * math.cos(a), 0, z + r * math.sin(a)), (.62, .62, .62))
lump(white, (7, 0, 3.9), (.55, .55, .5))
ring(red, (7, 0, 4.2), .5, .1, .9, 8); ring(white, (7, 0, 4.7), .3, .1, .5, 8)
# a candy-sled with striped runners
for s in (-1, 1): ring(white, (3.5, s * 1.1, .1), .12, .12, 6, 5, (0, math.pi / 2, 0))
for i in range(6): ring(red if i % 2 else white, (1.5 + i * .5, 0, .35), .4, .4, 2.2, 6, (math.pi / 2, 0, 0))
# a little candy-corn tree
ring(ice, (-6, 0, 0), .5, .5, 1.0, 6)
for k, (r, z) in enumerate(((2.2, 1.0), (1.7, 2.2), (1.2, 3.4), (.7, 4.5))): ring(red if k % 2 else white, (-6, 0, z), r, .1, 1.4, 7)
lump(bulb, (-6, 0, 5.8), (.4, .4, .4)); light("POINT", (1, .6, .2), 450, (-6, -1, 5.8))
# candy canes stuck in the snow
for x, y, h in ((-3, 4, 3.0), (-1, -4, 2.4), (4, 4.5, 2.8), (8, -3, 2.2)):
    for k in range(4): ring(red if k % 2 else white, (x, y, k * h / 4), .22, .22, h / 4, 6)
    lump(white, (x + .2, y, h + .1), (.2, .2, .2))
# string lights along a fence rail
for s in (-1, 1): ring(ice, (-8, s * 6, 0), .15, .15, 1.8, 5)
for i in range(7):
    x = -8 + i * 2.6
    for s in (-1, 1): ring(ice, (x, s * 6, 1.4), .1, .1, .5, 5)
    lump(bulb, (x, -6, 1.2), (.22, .22, .22)); lump(bulb, (x, 6, 1.2), (.22, .22, .22))
light("POINT", (1, .6, .2), 500, (0, -6, 1.5)); light("POINT", (1, .6, .2), 500, (0, 6, 1.5))
# a cold night: icy blue key from the left, magenta rim from behind
light("SUN", (.5, .7, 1), 5, (-22, -14, 12), (0, 0, 3)); light("SUN", (.9, .3, .7), 2.2, (20, 22, 10), (0, 0, 3))
w = bpy.data.worlds.new("w"); sc.world = w; w.use_nodes = True
next(n for n in w.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.01, .015, .04, 1)
cam = bpy.data.cameras.new("c"); cam.lens = 80; cam.clip_end = 400
co = bpy.data.objects.new("c", cam); sc.collection.objects.link(co)
t = Vector((0, 0, 3.5)); co.location = t + Vector((12, -46, 28))
co.rotation_euler = (t - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
