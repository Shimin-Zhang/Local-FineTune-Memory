import bpy, bmesh, math, random
from mathutils import Vector
random.seed(1046)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
col = {}
def paint(name, rgb, glow=0, rough=.85):
    m = bpy.data.materials.new(name); m.use_nodes = True
    p = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    p.inputs["Base Color"].default_value = (*rgb, 1); p.inputs["Roughness"].default_value = rough
    if glow:
        p.inputs["Emission Color"].default_value = (*rgb, 1); p.inputs["Emission Strength"].default_value = glow
    col[name] = m
def solid(bm, name, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new(name); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new(name, me); o.location = at; o.rotation_euler = rot
    sc.collection.objects.link(o); o.data.materials.append(col[name]); return o
def slab(at, size, name, rz=0, top=1):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts:
        k = top if v.co.z > 0 else 1
        v.co = Vector((v.co.x * size[0] * k, v.co.y * size[1] * k, (v.co.z + .5) * size[2]))
    return solid(bm, name, at, (0, 0, rz))
def drum(at, r, h, n, name, r2=None):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return solid(bm, name, at)
def lamp(kind, c, w, at, aim=None):
    d = bpy.data.lights.new(kind, kind); d.color = c; d.energy = w
    o = bpy.data.objects.new(kind, d); o.location = at; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
# street corner: asphalt slab with a rough tapered underside
bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
for v in bm.verts:
    if v.co.z < 0: v.co = Vector((v.co.x * 20 + random.uniform(-1.5, 1.5), v.co.y * 19 + random.uniform(-1.5, 1.5), -3.2 + random.uniform(-.8, .4)))
    else: v.co = Vector((v.co.x * 25, v.co.y * 24, 0))
solid(bm, "road", (0, 0, -1.6))
slab((0, 0, 0), (25, 24, .2), "road", 0, .98)

slab((0, 2.7, .3), (25, .35, .2), "kerb"); slab((-5.9, -2.6, .3), (.35, 13.2, .2), "kerb")
# crosswalk stripes across the street on the right
for i in range(6): slab((3.2 + i * 1.0, -2.6, .2), (.6, 6.0, .05), "paint")
# two buildings at the corner: a shop and a tall flat
slab((-4.5, 9.0, .3), (8.0, 6.0, 4.2), "shop", 0, .97)
slab((-4.5, 9.0, 4.5), (8.4, 6.4, .3), "roof")
slab((-4.5, 6.0, .3), (3.2, .2, 2.4), "door")
for i in range(3): slab((-6.8 + i * 2.3, 5.95, 1.4), (1.3, .15, 1.2), "window")
slab((-4.5, 5.8, 3.0), (4.0, .2, .8), "sign", 0, .95)
slab((3.5, 9.5, .3), (8.0, 7.0, 8.5), "flat", 0, .96)

        if (r, c) != (1, 1): slab((.9 + c * 2.6, 5.95, 1.6 + r * 2.5), (1.2, .15, 1.4), "window" if (r + c) % 2 else "dark")

slab((7.2, 11.0, 8.8), (1.2, 1.2, 1.0), "roof")
# street lamp: tall pole, arm, big glowing head
drum((-1.2, 4.3, .3), .16, 5.4, 6, "iron", .1)
slab((-1.2, 3.6, 5.5), (.16, 1.6, .16), "iron")

next(n for n in w.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.012, .014, .03, 1)

co = bpy.data.objects.new("cam", cam); sc.collection.objects.link(co)
t = Vector((0, 1, 3.5)); co.location = t + Vector((24, -58, 40))
co.rotation_euler = (t - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
