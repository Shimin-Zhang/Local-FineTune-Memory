import bpy, bmesh, random, math
from mathutils import Vector
random.seed(1216)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
S = {}
def paint(name, rgb, glow=0, rough=.5, metal=0):
    m = bpy.data.materials.new(name); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*rgb, 1); b.inputs["Roughness"].default_value = rough; b.inputs["Metallic"].default_value = metal
    if glow: b.inputs["Emission Color"].default_value = (*rgb, 1); b.inputs["Emission Strength"].default_value = glow
    S[name] = m
paint("floor", (.01, .01, .015), rough=.06); paint("floor2", (.02, .02, .03), rough=.06); paint("wall", (.03, .03, .045)); paint("car", (.75, .03, .04), rough=.15, metal=.3)
paint("dark", (.01, .01, .012), rough=.3); paint("trim", (.55, .55, .6), rough=.3, metal=.8); paint("glass", (.04, .08, .12), rough=.1)
paint("strip", (.8, .9, 1), glow=10); paint("head", (1, .95, .8), glow=40); paint("tail", (1, .05, .03), glow=30); paint("pool", (.5, .7, 1), glow=1.5); paint("tyre", (.02, .02, .02))
def put(bm, m, p=(0, 0, 0), r=(0, 0, 0)):
    me = bpy.data.meshes.new("p"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("p", me); o.location = p; o.rotation_euler = r
    sc.collection.objects.link(o); o.data.materials.append(S[m]); return o
def blk(m, x, y, z, w, d, h, rz=0):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * w, v.co.y * d, (v.co.z + .5) * h))
    return put(bm, m, (x, y, z), (0, 0, rz))
def cyl(m, x, y, z, r, h, n=8, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r, depth=h)
    return put(bm, m, (x, y, z), rot)
def bar(m, a, b, t):
    a, b = Vector(a), Vector(b); d = b - a; q = d.to_track_quat('Z', 'Y')
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = q @ Vector((v.co.x * t, v.co.y * t, (v.co.z + .5) * d.length)) + a
    return put(bm, m)
def light(kind, rgb, w, p, aim=None):
    L = bpy.data.lights.new("l", kind); L.color = rgb; L.energy = w
    o = bpy.data.objects.new("l", L); o.location = p; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
