import bpy, bmesh, random, math
from mathutils import Vector
random.seed(539)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
def mat(rgb, glow=0, rough=.6):
    m = bpy.data.materials.new("m"); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*rgb, 1); b.inputs["Roughness"].default_value = rough
    if glow: b.inputs["Emission Color"].default_value = (*rgb, 1); b.inputs["Emission Strength"].default_value = glow
    return m
oak, oak2, brass, velvet, orb, mist, mist2, rune, candle, wax = (mat((.16, .09, .05)), mat((.1, .055, .035)), mat((.7, .5, .15), rough=.35), mat((.3, .03, .12)),
    mat((.15, .3, .6), glow=.8, rough=.05), mat((.4, .7, 1), glow=3), mat((.7, .5, 1), glow=3), mat((.5, .9, 1), glow=10), mat((1, .7, .3), glow=14), mat((.7, .65, .55)))
def mesh(bm, m, at=(0, 0, 0), rot=(0, 0, 0)):
    me = bpy.data.meshes.new("p"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("p", me); o.location, o.rotation_euler = at, rot
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def disc(r, h, n, m, at, r2=None, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return mesh(bm, m, at, rot)
def bar(a, b, t, m):
    a, b = Vector(a), Vector(b); d = b - a; q = d.to_track_quat('Z', 'Y')
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = q @ Vector((v.co.x * t, v.co.y * t, (v.co.z + .5) * d.length)) + a
    return mesh(bm, m)
def gem(r, m, at, sq=.7):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=2, radius=r)
    for v in bm.verts: v.co.z *= sq
    return mesh(bm, m, at)
def lamp(kind, c, w, at, aim=None):
    L = bpy.data.lights.new("l", kind); L.color, L.energy = c, w
    o = bpy.data.objects.new("l", L); o.location = at; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
