import bpy, bmesh, random, math
from mathutils import Vector
random.seed(1106)
for obj in list(bpy.data.objects): bpy.data.objects.remove(obj, do_unlink=True)
sc = bpy.context.scene
W = {}
def stain(k, c, e=0, r=.9):
    m = bpy.data.materials.new(k); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*c, 1); b.inputs["Roughness"].default_value = r
    if e: b.inputs["Emission Color"].default_value = (*c, 1); b.inputs["Emission Strength"].default_value = e
    W[k] = m
for k, c in (("deck", (.16, .11, .07)), ("oak", (.38, .24, .12)), ("oak2", (.3, .18, .09)), ("oak3", (.44, .3, .16)), ("iron", (.06, .06, .07)),
             ("rope", (.45, .36, .2)), ("stone", (.17, .16, .15)), ("stain", (.05, .03, .02)), ("lamp", (1, .6, .2))): stain(k, c)
stain("candle", (1, .75, .3), 14); stain("ember", (1, .35, .05), 6)
def mk(bm, k, p=(0, 0, 0), r=(0, 0, 0)):
    me = bpy.data.meshes.new("m"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("o", me); o.location = p; o.rotation_euler = r
    sc.collection.objects.link(o); o.data.materials.append(W[k]); return o
def blk(p, s, k, r=(0, 0, 0), j=0):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * s[0], v.co.y * s[1], v.co.z * s[2])) + Vector((random.uniform(-j, j), random.uniform(-j, j), random.uniform(-j, j)))
    return mk(bm, k, p, r)
def rod(p, r, h, n, k, r2=None, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    return mk(bm, k, p, rot)
def lamp(kind, c, w, p, aim=None):
    l = bpy.data.lights.new("l", kind); l.color = c; l.energy = w
    o = bpy.data.objects.new("l", l); o.location = p; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()

    if v.co.z < 0: v.co = Vector((v.co.x * .55 + random.uniform(-1, 1), v.co.y * .55 + random.uniform(-1, 1), v.co.z - random.uniform(0, 1.8)))
    v.co.z -= 1.6
mk(bm, "stone")

for z in (1.0, 5.0):
    for y in (-3.15, 3.15): blk(at(0, y, z), (6.4, .4, .35), "iron")

blk(at(0, 0, 0.2), (6.0, 6.0, .4), "oak2")
blk(at(0, 0, 6.1), (6.4, 6.4, .45), "oak3", (0, 0, 0)); blk(at(0, 0, 6.4), (5.6, 5.6, .1), "oak")
blk(at(0, -3.3, 6.7), (6.4, .4, .35), "iron"); blk(at(0, 3.3, 6.7), (6.4, .4, .35), "iron")
