def skin(c, e=0, r=.85):
    m = bpy.data.materials.new("s"); m.use_nodes = True
    n = next(x for x in m.node_tree.nodes if x.type == "BSDF_PRINCIPLED")
    n.inputs["Base Color"].default_value = (*c, 1); n.inputs["Roughness"].default_value = r
    if e: n.inputs["Emission Color"].default_value = (*c, 1); n.inputs["Emission Strength"].default_value = e
    return m

def mesh(bm, m, p=(0, 0, 0), r=(0, 0, 0)):
    me = bpy.data.meshes.new("b"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("b", me); o.location = p; o.rotation_euler = r
    sc.collection.objects.link(o); o.data.materials.append(m); return o
def slab(x, y, z, w, d, h, m, rz=0):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = V((v.co.x * w, v.co.y * d, (v.co.z + .5) * h))
    return mesh(bm, m, (x, y, z), (0, 0, rz))
def tube(x, y, z, r, h, n, m, r2=None):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return mesh(bm, m, (x, y, z))
def ball(x, y, z, r, m, tilt):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=3, radius=r)
    return mesh(bm, m, (x, y, z), tilt)
def shine(kind, p, c, w, aim=None):
    L = bpy.data.lights.new("s", kind); L.color = c; L.energy = w
    o = bpy.data.objects.new("s", L); o.location = p; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (V(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()

def seam(pts, w):
    bm = bmesh.new()

        a, b = V(pts[i]), V(pts[i + 1]); d = b - a; q = d.to_track_quat('Z', 'Y')

    return mesh(bm, black)
def at(u, v, r=R): return V((r * math.cos(u) * math.cos(v), r * math.cos(u) * math.sin(v), r * math.sin(u)))
