import bpy
import math
import random
from mathutils import Vector

# Fix camera - it's pointing wrong direction
camera = bpy.data.objects.get("Camera")
if camera:
    # Position camera at street level, looking down the street
    camera.location = (0, 2, 25)
    # Look towards -Z direction (down the street)
    camera.rotation_euler = (math.radians(85), 0, 0)

# Add more neon signs closer to camera for better visibility
mat_neon_pink = bpy.data.materials.get("Neon_Pink")
mat_neon_cyan = bpy.data.materials.get("Neon_Cyan")
mat_neon_yellow = bpy.data.materials.get("Neon_Yellow")
mat_neon_green = bpy.data.materials.get("Neon_Green")
mat_neon_purple = bpy.data.materials.get("Neon_Purple")
mat_neon_red = bpy.data.materials.get("Neon_Red")
mat_screen = bpy.data.materials.get("Screen")
mat_screen2 = bpy.data.materials.get("Screen2")

def create_neon_sign_simple(x, y, z, width, height, mat):
    bpy.ops.mesh.primitive_plane_add(size=1, location=(x, y, z))
    sign = bpy.context.active_object
    sign.scale = (width/2, height/2, 1)
    sign.rotation_euler = (math.pi/2, 0, 0)
    sign.data.materials.append(mat)
    return sign

# Add more signs closer to camera
create_neon_sign_simple(-8, 5, 15, 3, 1, mat_neon_pink)
create_neon_sign_simple(8, 6, 15, 3, 1, mat_neon_cyan)
create_neon_sign_simple(-8, 8, 20, 4, 1.5, mat_neon_yellow)
create_neon_sign_simple(8, 9, 20, 4, 1.5, mat_neon_green)
create_neon_sign_simple(-8, 12, 25, 5, 2, mat_neon_purple)
create_neon_sign_simple(8, 13, 25, 5, 2, mat_neon_red)

# Add more advertising screens
def create_ad_screen_simple(x, y, z, width, height, mat):
    bpy.ops.mesh.primitive_plane_add(size=1, location=(x, y, z))
    screen = bpy.context.active_object
    screen.scale = (width/2, height/2, 1)
    screen.rotation_euler = (0, 0, 0)
    screen.data.materials.append(mat)
    return screen

create_ad_screen_simple(-12, 30, 20, 6, 4, mat_screen)
create_ad_screen_simple(12, 35, 20, 6, 4, mat_screen2)
create_ad_screen_simple(-12, 40, 30, 8, 5, mat_screen)
create_ad_screen_simple(12, 45, 30, 8, 5, mat_screen2)

# Add more people with umbrellas closer to camera
mat_umbrella = bpy.data.materials.get("Umbrella")
mat_umbrella2 = bpy.data.materials.get("Umbrella2")
mat_umbrella3 = bpy.data.materials.get("Umbrella3")
mat_person = bpy.data.materials.get("Person")
mat_person2 = bpy.data.materials.get("Person2")
mat_metal = bpy.data.materials.get("Metal")

def create_person_simple(x, z, umbrella_mat):
    bpy.ops.mesh.primitive_cylinder_add(radius=0.2, depth=1.5, location=(x, 0.75, z))
    body = bpy.context.active_object
    body.data.materials.append(random.choice([mat_person, mat_person2]))
    
    bpy.ops.mesh.primitive_uv_sphere_add(radius=0.15, location=(x, 1.7, z))
    head = bpy.context.active_object
    head.data.materials.append(mat_person)
    
    bpy.ops.mesh.primitive_cone_add(radius1=0.8, radius2=0.1, depth=0.5, location=(x, 2.2, z))
    umbrella = bpy.context.active_object
    umbrella.data.materials.append(umbrella_mat)
    
    bpy.ops.mesh.primitive_cylinder_add(radius=0.02, depth=1, location=(x, 1.7, z))
    handle = bpy.context.active_object
    handle.data.materials.append(mat_metal)

# Add more people closer to camera
for i in range(15):
    x = random.uniform(-15, 15)
    z = random.uniform(5, 25)
    if abs(x) < 3:
        x += 4 * (1 if x > 0 else -1)
    create_person_simple(x, z, random.choice([mat_umbrella, mat_umbrella2, mat_umbrella3]))

# Add more neon lights
neon_light_data = [
    (-8, 5, 15, (1, 0.1, 0.3)),
    (8, 6, 15, (0.1, 0.8, 1)),
    (-8, 8, 20, (1, 0.9, 0.1)),
    (8, 9, 20, (0.1, 1, 0.3)),
    (-8, 12, 25, (0.6, 0.1, 1)),
    (8, 13, 25, (1, 0.1, 0.1)),
]

for x, y, z, color in neon_light_data:
    bpy.ops.object.light_add(type='POINT', location=(x, y, z))
    light = bpy.context.active_object
    light.data.energy = 30
    light.data.color = color
    light.data.shadow_soft_size = 0.3

# Add street lamps
mat_street_light = bpy.data.materials.new("Street_Light")
mat_street_light.use_nodes = True
nodes = mat_street_light.node_tree.nodes
links = mat_street_light.node_tree.links
nodes.clear()
out = nodes.new('ShaderNodeOutputMaterial')
out.location = (300, 0)
bsdf = nodes.new('ShaderNodeBsdfPrincipled')
bsdf.location = (0, 0)
bsdf.inputs['Base Color'].default_value = (0.8, 0.8, 0.9, 1)
bsdf.inputs['Emission Color'].default_value = (0.8, 0.8, 0.9, 1)
bsdf.inputs['Emission Strength'].default_value = 2.0
links.new(bsdf.outputs['BSDF'], out.inputs['Surface'])

for z in range(10, 30, 10):
    # Lamp post
    bpy.ops.mesh.primitive_cylinder_add(radius=0.05, depth=5, location=(-12, 2.5, z))
    post = bpy.context.active_object
    post.data.materials.append(bpy.data.materials.get("Metal"))
    
    # Lamp head
    bpy.ops.mesh.primitive_cube_add(size=1, location=(-12, 5, z))
    lamp = bpy.context.active_object
    lamp.scale = (0.5, 0.3, 0.5)
    lamp.data.materials.append(mat_street_light)
    
    # Light
    bpy.ops.object.light_add(type='POINT', location=(-12, 5, z))
    light = bpy.context.active_object
    light.data.energy = 20
    light.data.color = (0.8, 0.8, 0.9)
    light.data.shadow_soft_size = 0.5

print("Added more detail to scene!")
