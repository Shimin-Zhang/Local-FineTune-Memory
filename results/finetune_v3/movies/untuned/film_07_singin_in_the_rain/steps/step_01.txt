def create_material(name, base_color, metallic=0.0, roughness=0.5, emission=0.0, emission_color=(0,0,0,1)):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    links = mat.node_tree.links
    nodes.clear()
    bsdf = nodes.new('ShaderNodeBsdfPrincipled')
    out = nodes.new('ShaderNodeOutputMaterial')
    out.location = (300, 0)
    bsdf.location = (0, 0)
    bsdf.inputs['Base Color'].default_value = base_color
    bsdf.inputs['Metallic'].default_value = metallic
    bsdf.inputs['Roughness'].default_value = roughness
    if emission > 0:
        bsdf.inputs['Emission Color'].default_value = emission_color
        bsdf.inputs['Emission Strength'].default_value = emission
    links.new(bsdf.outputs['BSDF'], out.inputs['Surface'])
    return mat

def create_building(x, z, width, depth, height, mat, num_floors=4, windows_per_floor=4):
    """Create a simple box building with windows."""
    # Main structure
    mesh = bpy.data.meshes.new(f"Building_{x}_{z}")
    obj = bpy.data.objects.new(f"Building_{x}_{z}", mesh)
    bpy.context.collection.objects.link(obj)
    bm = bmesh.new()
    # Box
    x0, x1 = x - width/2, x + width/2
    z0, z1 = z - depth/2, z + depth/2
    y0, y1 = 0, height
    verts = [
        bm.verts.new((x0, y0, z0)),
        bm.verts.new((x1, y0, z0)),
        bm.verts.new((x1, y0, z1)),
        bm.verts.new((x0, y0, z1)),
        bm.verts.new((x0, y1, z0)),
        bm.verts.new((x1, y1, z0)),
        bm.verts.new((x1, y1, z1)),
        bm.verts.new((x0, y1, z1)),
    ]
    bm.faces.new([verts[0], verts[1], verts[2], verts[3]])
    bm.faces.new([verts[4], verts[7], verts[6], verts[5]])
    bm.faces.new([verts[0], verts[4], verts[5], verts[1]])
    bm.faces.new([verts[2], verts[6], verts[7], verts[3]])
    bm.faces.new([verts[1], verts[5], verts[6], verts[2]])
    bm.faces.new([verts[0], verts[3], verts[7], verts[4]])
    bm.to_mesh(mesh)
    bm.free()
    obj.data.materials.append(mat)
    
    # Windows on front face (facing street, z = z1)
    floor_h = height / num_floors
    win_w = width / (windows_per_floor * 2)
    win_h = floor_h * 0.5
    
    for floor in range(num_floors):
        for w in range(windows_per_floor):
            wx = x0 + win_w * (w * 2 + 1)
            wy = y0 + floor_h * floor + floor_h * 0.25
            wz = z1 + 0.01
            wm = bpy.data.meshes.new(f"Win_{x}_{z}_{floor}_{w}")
            wo = bpy.data.objects.new(f"Win_{x}_{z}_{floor}_{w}", wm)
            bpy.context.collection.objects.link(wo)
            bm = bmesh.new()
            ww = win_w * 0.8
            wh = win_h * 0.8
            v = [
                bm.verts.new((wx - ww/2, wy - wh/2, wz)),
                bm.verts.new((wx + ww/2, wy - wh/2, wz)),
                bm.verts.new((wx + ww/2, wy + wh/2, wz)),
                bm.verts.new((wx - ww/2, wy + wh/2, wz)),
            ]
            bm.faces.new(v)
            bm.to_mesh(wm)
            bm.free()
            # Randomly lit or dark
            if random.random() > 0.3:
                wo.data.materials.append(mat_window_lit)
            else:
                wo.data.materials.append(mat_window_dark)
            
            # Window frame
            fm = bpy.data.meshes.new(f"WinF_{x}_{z}_{floor}_{w}")
            fo = bpy.data.objects.new(f"WinF_{x}_{z}_{floor}_{w}", fm)
            bpy.context.collection.objects.link(fo)
            bm = bmesh.new()
            fw = ww + 0.05
            fh = wh + 0.05
            v = [
                bm.verts.new((wx - fw/2, wy - fh/2, wz - 0.01)),
                bm.verts.new((wx + fw/2, wy - fh/2, wz - 0.01)),
                bm.verts.new((wx + fw/2, wy + fh/2, wz - 0.01)),
                bm.verts.new((wx - fw/2, wy + fh/2, wz - 0.01)),
            ]
            bm.faces.new(v)
            bm.to_mesh(fm)
            bm.free()
            fo.data.materials.append(mat_window_frame)
    
    return obj

def create_storefront(x, z, width, depth, mat_wall, mat_glass, mat_frame):
    """Create a storefront with large glass windows at street level."""
    # Storefront wall
    mesh = bpy.data.meshes.new(f"Storefront_{x}_{z}")
    obj = bpy.data.objects.new(f"Storefront_{x}_{z}", mesh)
    bpy.context.collection.objects.link(obj)
    bm = bmesh.new()
    x0, x1 = x - width/2, x + width/2
    z0, z1 = z - depth/2, z + depth/2
    h = 4  # Storefront height
    # Front face
    v = [
        bm.verts.new((x0, 0, z1)),
        bm.verts.new((x1, 0, z1)),
        bm.verts.new((x1, h, z1)),
        bm.verts.new((x0, h, z1)),
    ]
    bm.faces.new(v)
    # Top
    v = [
        bm.verts.new((x0, h, z0)),
        bm.verts.new((x1, h, z0)),
        bm.verts.new((x1, h, z1)),
        bm.verts.new((x0, h, z1)),
    ]
    bm.faces.new(v)
    bm.to_mesh(mesh)
    bm.free()
    obj.data.materials.append(mat_wall)
    
    # Large glass window
    gm = bpy.data.meshes.new(f"StoreGlass_{x}_{z}")
    go = bpy.data.objects.new(f"StoreGlass_{x}_{z}", gm)
    bpy.context.collection.objects.link(go)
    bm = bmesh.new()
    gw = width * 0.8
    gh = 3
    v = [
        bm.verts.new((x - gw/2, 0.3, z1 + 0.02)),
        bm.verts.new((x + gw/2, 0.3, z1 + 0.02)),
        bm.verts.new((x + gw/2, 0.3 + gh, z1 + 0.02)),
        bm.verts.new((x - gw/2, 0.3 + gh, z1 + 0.02)),
    ]
    bm.faces.new(v)
    bm.to_mesh(gm)
    bm.free()
    go.data.materials.append(mat_glass)
    
    # Frame
    fm = bpy.data.meshes.new(f"StoreFrame_{x}_{z}")
    fo = bpy.data.objects.new(f"StoreFrame_{x}_{z}", fm)
    bpy.context.collection.objects.link(fo)
    bm = bmesh.new()
    fw = gw + 0.1
    fh = gh + 0.1
    v = [
        bm.verts.new((x - fw/2, 0.2, z1 + 0.01)),
        bm.verts.new((x + fw/2, 0.2, z1 + 0.01)),
        bm.verts.new((x + fw/2, 0.2 + fh, z1 + 0.01)),
        bm.verts.new((x - fw/2, 0.2 + fh, z1 + 0.01)),
    ]
    bm.faces.new(v)
    bm.to_mesh(fm)
    bm.free()
    fo.data.materials.append(mat_frame)
    
    # Awning
    am = bpy.data.meshes.new(f"Awning_{x}_{z}")
    ao = bpy.data.objects.new(f"Awning_{x}_{z}", am)
    bpy.context.collection.objects.link(ao)
    bm = bmesh.new()
    aw = width * 0.85
    ah = 1.5
    v = [
        bm.verts.new((x - aw/2, 0.3 + gh + 0.1, z1 + 0.02)),
        bm.verts.new((x + aw/2, 0.3 + gh + 0.1, z1 + 0.02)),
        bm.verts.new((x + aw/2, 0.3 + gh + 0.1, z1 + 0.02 + ah)),
        bm.verts.new((x - aw/2, 0.3 + gh + 0.1, z1 + 0.02 + ah)),
    ]
    bm.faces.new(v)
    bm.to_mesh(am)
    bm.free()
    ao.data.materials.append(mat_awning)

def create_lamp_post(x, z):
    """Create a classic street lamp post."""
    # Base
    bm = bmesh.new()
    # Octagonal base
    import math
    for i in range(8):
        angle = i * math.pi / 4
        r = 0.3
        bm.verts.new((x + r * math.cos(angle), 0, z + r * math.sin(angle)))
    for i in range(8):
        angle = i * math.pi / 4
        r = 0.25
        bm.verts.new((x + r * math.cos(angle), 0.2, z + r * math.sin(angle)))
    bm.to_mesh(bpy.data.meshes.new(f"LampBase_{x}_{z}"))
    # Actually let's do a simpler cylinder-like base
    base_mesh = bpy.data.meshes.new(f"LampBase_{x}_{z}")
    base_obj = bpy.data.objects.new(f"LampBase_{x}_{z}", base_mesh)
    bpy.context.collection.objects.link(base_obj)
    bm = bmesh.new()
    # Create a simple tapered cylinder base
    segments = 12
    bottom_r = 0.35
    top_r = 0.2
    h = 0.3
    bottom_verts = []
    top_verts = []
    for i in range(segments):
        angle = i * 2 * math.pi / segments
        bottom_verts.append(bm.verts.new((x + bottom_r * math.cos(angle), 0, z + bottom_r * math.sin(angle))))
        top_verts.append(bm.verts.new((x + top_r * math.cos(angle), h, z + top_r * math.sin(angle))))
    for i in range(segments):
        j = (i + 1) % segments
        bm.faces.new([bottom_verts[i], bottom_verts[j], top_verts[j], top_verts[i]])
    bm.to_mesh(base_mesh)
    bm.free()
    base_obj.data.materials.append(mat_lamp_base)
    
    # Pole
    pole_mesh = bpy.data.meshes.new(f"LampPole_{x}_{z}")
    pole_obj = bpy.data.objects.new(f"LampPole_{x}_{z}", pole_mesh)
    bpy.context.collection.objects.link(pole_obj)
    bm = bmesh.new()
    pole_h = 4.5
    pr = 0.08
    bv = []
    tv = []
    for i in range(segments):
        angle = i * 2 * math.pi / segments
        bv.append(bm.verts.new((x + pr * math.cos(angle), h, z + pr * math.sin(angle))))
        tv.append(bm.verts.new((x + pr * math.cos(angle), h + pole_h, z + pr * math.sin(angle))))
    for i in range(segments):
        j = (i + 1) % segments
        bm.faces.new([bv[i], bv[j], tv[j], tv[i]])
    bm.to_mesh(pole_mesh)
    bm.free()
    pole_obj.data.materials.append(mat_lamp_post)
    
    # Lamp head (globe)
    lamp_mesh = bpy.data.meshes.new(f"LampHead_{x}_{z}")
    lamp_obj = bpy.data.objects.new(f"LampHead_{x}_{z}", lamp_mesh)
    bpy.context.collection.objects.link(lamp_obj)
    bpy.ops.mesh.primitive_uv_sphere_add(radius=0.25, location=(x, h + pole_h + 0.25, z), segments=16, ring_count=8)
    lamp_obj = bpy.context.active_object
    lamp_obj.name = f"LampHead_{x}_{z}"
    lamp_obj.data.materials.append(mat_lamp_glass)
    
    # Decorative arm
    arm_mesh = bpy.data.meshes.new(f"LampArm_{x}_{z}")
    arm_obj = bpy.data.objects.new(f"LampArm_{x}_{z}", arm_mesh)
    bpy.context.collection.objects.link(arm_obj)
    bm = bmesh.new()
    # Curved arm from top of pole
    arm_start = (x, h + pole_h, z)
    arm_end = (x, h + pole_h + 0.1, z)
    v = [
        bm.verts.new((x - 0.05, h + pole_h, z)),
        bm.verts.new((x + 0.05, h + pole_h, z)),
        bm.verts.new((x + 0.05, h + pole_h + 0.15, z)),
        bm.verts.new((x - 0.05, h + pole_h + 0.15, z)),
    ]
    bm.faces.new(v)
    bm.to_mesh(arm_mesh)
    bm.free()
    arm_obj.data.materials.append(mat_lamp_post)
