import bpy, math, random

# Remove the rainbow torus that appears as a vertical line
for obj in list(bpy.data.objects):
    if obj.type == 'MESH' and obj.dimensions.z > 20 and obj.dimensions.x < 2:
        bpy.data.objects.remove(obj, do_unlink=True)

# Remove remaining fence rails crossing road
for obj in list(bpy.data.objects):
    if obj.type == 'MESH' and obj.dimensions.x > 4 and obj.dimensions.y < 1.5 and obj.dimensions.z < 1.5:
        if abs(obj.location.z) < 2 and abs(obj.location.x) < 5:
            bpy.data.objects.remove(obj, do_unlink=True)

# Add a proper rainbow using curved arcs (half torus)
m_r1 = bpy.data.materials.new("RB1"); m_r1.use_nodes=True
m_r1.node_tree.nodes['Principled BSDF'].inputs['Base Color'].default_value=(1,.3,.3,1)
m_r1.node_tree.nodes['Principled BSDF'].inputs['Emission Strength'].default_value=.4
m_r2 = bpy.data.materials.new("RB2"); m_r2.use_nodes=True
m_r2.node_tree.nodes['Principled BSDF'].inputs['Base Color'].default_value=(1,.8,.3,1)
m_r2.node_tree.nodes['Principled BSDF'].inputs['Emission Strength'].default_value=.4
m_r3 = bpy.data.materials.new("RB3"); m_r3.use_nodes=True
m_r3.node_tree.nodes['Principled BSDF'].inputs['Base Color'].default_value=(.3,.8,.3,1)
m_r3.node_tree.nodes['Principled BSDF'].inputs['Emission Strength'].default_value=.4
m_r4 = bpy.data.materials.new("RB4"); m_r4.use_nodes=True
m_r4.node_tree.nodes['Principled BSDF'].inputs['Base Color'].default_value=(.3,.5,1,1)
m_r4.node_tree.nodes['Principled BSDF'].inputs['Emission Strength'].default_value=.4

# Rainbow arcs in the distance
for i, (mat, r) in enumerate([(m_r1,25),(m_r2,23),(m_r3,21),(m_r4,19)]):
    bpy.ops.mesh.primitive_torus_add(major_radius=r, minor_radius=.4, location=(0,70,0), major_segments=48, minor_segments=6)
    arc = bpy.context.active_object
    arc.rotation_euler = (0, math.radians(90), 0)
    arc.scale = (1, 1, .5)
    arc.data.materials.append(mat)

# Add a small cottage in the poppy field
m_cottage_wall = bpy.data.materials.new("CottageWall"); m_cottage_wall.use_nodes=True
m_cottage_wall.node_tree.nodes['Principled BSDF'].inputs['Base Color'].default_value=(.9,.85,.7,1)
m_cottage_wall.node_tree.nodes['Principled BSDF'].inputs['Roughness'].default_value=.7
m_cottage_roof = bpy.data.materials.new("CottageRoof"); m_cottage_roof.use_nodes=True
m_cottage_roof.node_tree.nodes['Principled BSDF'].inputs['Base Color'].default_value=(.6,.2,.1,1)
m_cottage_roof.node_tree.nodes['Principled BSDF'].inputs['Roughness'].default_value=.6

bpy.ops.mesh.primitive_cube_add(size=1, location=(-12,-25,1))
ct = bpy.context.active_object; ct.scale=(2,2,1); ct.data.materials.append(m_cottage_wall)
bpy.ops.mesh.primitive_cone_add(radius1=3, radius2=0, depth=2, location=(-12,-25,3))
cr = bpy.context.active_object; cr.scale=(1,1,1); cr.data.materials.append(m_cottage_roof)
# Door
bpy.ops.mesh.primitive_cube_add(size=.8, location=(-12,-23,0.5))
dr = bpy.context.active_object; dr.scale=(.5,.1,.5); dr.data.materials.append(bpy.data.materials.get("Fence"))
# Window
bpy.ops.mesh.primitive_cube_add(size=.5, location=(-11,-23,1.2))
wn = bpy.context.active_object; wn.scale=(.3,.1,.3); wn.data.materials.append(bpy.data.materials.get("Win"))

# Add a second small cottage
bpy.ops.mesh.primitive_cube_add(size=1, location=(14,-30,1))
ct2 = bpy.context.active_object; ct2.scale=(1.5,1.5,1); ct2.data.materials.append(m_cottage_wall)
bpy.ops.mesh.primitive_cone_add(radius1=2.5, radius2=0, depth=1.5, location=(14,-30,2.5))
cr2 = bpy.context.active_object; cr2.data.materials.append(m_cottage_roof)

# Add a small well
m_stone = bpy.data.materials.get("Rock")
if m_stone:
    bpy.ops.mesh.primitive_cylinder_add(radius=.8, depth=1, location=(5,-20,.5))
    wl = bpy.context.active_object; wl.data.materials.append(m_stone)
    bpy.ops.mesh.primitive_cylinder_add(radius=.7, depth=.1, location=(5,-20,1.05))
    wt = bpy.context.active_object; wt.data.materials.append(m_stone)
    # Well roof posts
    for px in [-.6,.6]:
        bpy.ops.mesh.primitive_cylinder_add(radius=.05, depth=1.2, location=(5+px,-20,1.6))
        bpy.context.active_object.data.materials.append(bpy.data.materials.get("Fence"))
    # Well roof
    bpy.ops.mesh.primitive_cone_add(radius1=1, radius2=0, depth=.6, location=(5,-20,2.4))
    bpy.context.active_object.data.materials.append(m_cottage_roof)

# Add a path leading from road to cottage
m_path = bpy.data.materials.new("DirtPath"); m_path.use_nodes=True
m_path.node_tree.nodes['Principled BSDF'].inputs['Base Color'].default_value=(.6,.5,.3,1)
m_path.node_tree.nodes['Principled BSDF'].inputs['Roughness'].default_value=.9

bpy.ops.mesh.primitive_plane_add(size=1, location=(-7,-25,.01))
pp = bpy.context.active_object; pp.scale=(1.5,4,1); pp.data.materials.append(m_path)

# Add a few butterflies (simple)
m_wing = bpy.data.materials.new("Wing"); m_wing.use_nodes=True
m_wing.node_tree.nodes['Principled BSDF'].inputs['Base Color'].default_value=(1,.5,.8,1)
m_wing.node_tree.nodes['Principled BSDF'].inputs['Emission Strength'].default_value=.2

random.seed(55)
for i in range(6):
    bx = random.uniform(-8,8); by = random.uniform(-20,-5); bz = random.uniform(.5,1.5)
    bpy.ops.mesh.primitive_plane_add(size=.1, location=(bx,by,bz))
    w1 = bpy.context.active_object; w1.rotation_euler=(random.uniform(0,3),random.uniform(0,3),random.uniform(0,3)); w1.data.materials.append(m_wing)
    bpy.ops.mesh.primitive_plane_add(size=.1, location=(bx+.05,by,bz))
    w2 = bpy.context.active_object; w2.rotation_euler=(random.uniform(0,3),random.uniform(0,3),random.uniform(0,3)); w2.data.materials.append(m_wing)

# Final camera adjustment
cam = bpy.data.objects.get("Cam")
if cam:
    cam.location = (0, -28, 4.5)
    cam.rotation_euler = (math.radians(85), 0, 0)
    cam.data.lens = 38
