import bpy, bmesh, math, random
from mathutils import Vector
random.seed(1125)
sc = bpy.context.scene
mats = {}
def tone(name, rgb, glow=0, rough=.9):
    m = bpy.data.materials.new(name); m.use_nodes = True
    b = next(n for n in m.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
    b.inputs["Base Color"].default_value = (*rgb, 1); b.inputs["Roughness"].default_value = rough
    if glow:
        b.inputs["Emission Color"].default_value = (*rgb, 1); b.inputs["Emission Strength"].default_value = glow
    mats[name] = m
for n, c, g in [("soil", (.07, .05, .04), 0), ("paving", (.22, .22, .24), 0), ("trunk", (.14, .09, .06), 0), ("pine", (.04, .13, .09), 0),
        ("pine2", (.06, .17, .11), 0), ("frame", (.55, .08, .06), 0), ("chrome", (.7, .72, .75), 0), ("tyre", (.03, .03, .03), 0),
        ("skin", (.5, .55, .6), 0), ("skin2", (.35, .4, .45), 0), ("jeans", (.1, .15, .4), 0), ("shirt", (.5, .3, .1), 0),
        ("hair", (.3, .15, .05), 0), ("basket", (.45, .3, .12), 0), ("moon", (.9, .92, 1), 1.2), ("crater", (.55, .6, .75), .4),
        ("lamp", (1, .85, .5), 20), ("tail", (1, .1, .05), 12), ("star", (1, 1, 1), 10)]:
    tone(n, c, g)
def put(bm, m, p=(0, 0, 0), r=(0, 0, 0)):
    me = bpy.data.meshes.new("m"); bm.to_mesh(me); bm.free()
    o = bpy.data.objects.new("m", me); o.location = p; o.rotation_euler = r
    sc.collection.objects.link(o); o.data.materials.append(mats[m]); return o
def cube(p, s, m, rz=0):
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = Vector((v.co.x * s[0], v.co.y * s[1], (v.co.z + .5) * s[2]))
    return put(bm, m, p, (0, 0, rz))
def tube(p, r, h, n, m, r2=None, rot=(0, 0, 0)):
    bm = bmesh.new(); bmesh.ops.create_cone(bm, cap_ends=True, segments=n, radius1=r, radius2=r if r2 is None else r2, depth=h)
    for v in bm.verts: v.co.z += h / 2
    return put(bm, m, p, rot)
def ball(p, r, m, sq=1):
    bm = bmesh.new(); bmesh.ops.create_icosphere(bm, subdivisions=1, radius=r)
    for v in bm.verts: v.co.z *= sq
    return put(bm, m, p)
def bar(a, b, t, m):
    a, b = Vector(a), Vector(b); d = b - a; q = d.to_track_quat('Z', 'Y')
    bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
    for v in bm.verts: v.co = q @ Vector((v.co.x * t, v.co.y * t, (v.co.z + .5) * d.length)) + a
    return put(bm, m)
def shine(kind, c, w, p, aim=None):
    L = bpy.data.lights.new("l", kind); L.color = c; L.energy = w
    o = bpy.data.objects.new("l", L); o.location = p; sc.collection.objects.link(o)
    if aim: o.rotation_euler = (Vector(aim) - o.location).to_track_quat('-Z', 'Y').to_euler()
# floating slab: rough tapered underside, grass and a strip of road
bm = bmesh.new(); bmesh.ops.create_cube(bm, size=1)
for v in bm.verts:
    if v.co.z < 0: v.co.x *= .55; v.co.y *= .55; v.co.z -= random.uniform(0, .8); v.co.x += random.uniform(-.1, .1)
    v.co = Vector((v.co.x * 28, v.co.y * 22, v.co.z * 3 - 1.5))
put(bm, "soil")
cube((0, -6.5, 0), (26, 3.4, .1), "paving"); cube((0, -6.5, .1), (26, .12, .05), "chrome")
for i in range(14): cube((-12 + i * 1.85, -6.5, .1), (.9, .12, .05), "lamp")
# the full moon, huge, low behind the trees, with craters
ball((-2, 9.5, 8), 6.5, "moon")
for x, y, z, r in [(-4, 9, 9.5, 1.2), (-.5, 9, 6, .9), (1.5, 9.5, 10, .8), (-3.5, 9, 5.5, .6), (0, 9, 10.5, .5)]:
    ball((x, y, z), r, "crater")
# forest: layered cones on trunks, both banks and behind
for i in range(16):
    x = random.uniform(-12.5, 12.5); y = random.choice([random.uniform(-2.5, 1.5), random.uniform(3, 6)])
    if abs(x) < 3 and y < 0: continue
    h = random.uniform(2.5, 4.2); tube((x, y, 0), .22, h * .4, 5, "trunk")
    for k in range(3): tube((x, y, h * .3 + k * h * .22), 1.5 - k * .4, h * .4, 6, random.choice(["pine", "pine2"]), .1)
# the bicycle, enlarged, banked toward us, E.T. in the basket
bx, by, bz, tilt = -1.5, -6.5, 4.2, .12
def bike(m): return Vector((bx + m[0] * math.cos(tilt), by + m[0] * math.sin(tilt), bz + m[1]))
R = 1.1
for wx in (-2.0, 2.0):
    for k in range(10):
        a = k * math.pi / 5
        bar(bike((wx, R, 0)) + Vector((math.cos(a) * R, 0, math.sin(a) * R)), bike((wx, R, 0)) + Vector((math.cos(a + math.pi) * R, 0, math.sin(a + math.pi) * R)), .1, "tyre")
    bar(bike((wx - .0, R, 0)), bike((wx, R + .0, 0)), .22, "chrome")
bar(bike((-2, R, 0)), bike((0, R + 1.1, 0)), .13, "frame"); bar(bike((0, R + 1.1, 0)), bike((2, R, 0)), .13, "frame")
bar(bike((0, R + 1.1, 0)), bike((-.4, R + 2.4, 0)), .12, "frame"); bar(bike((2, R, 0)), bike((2.3, R + 2.2, 0)), .12, "frame")
bar(bike((-.4, R + 2.4, 0)), bike((-2.4, R + 2.4, 0)), .1, "chrome"); bar(bike((2.3, R + 2.2, 0)), bike((2.3, R + 2.9, 0)), .1, "chrome")
cube(bike((-.2, R + 1.2, 0)), (.9, .35, .25), "tyre")
cube(bike((2.0, R, 0)), (.3, .25, .4), "lamp"); shine("POINT", (1, .85, .5), 900, bike((2.0, R, 0)) + Vector((0, -1.5, .4)))
cube(bike((-2.0, R, .45)), (.25, .2, .25), "tail"); shine("POINT", (1, .1, .05), 200, bike((-2.0, R, .45)) + Vector((0, 1, 0)))
# the basket with E.T.
cube(bike((1.3, R + 1.4, 0)), (1.5, 1.3, .8), "basket")
for sx in (-1, 1): bar(bike((1.3 + sx * .7, R + 1.4, .8)), bike((1.3 + sx * .7, R + 1.4, 1.5)), .06, "basket")
bar(bike((1.3, R + .75, .8)), bike((1.3, R + .75, 1.5)), .06, "basket")
et = bike((1.3, R + 1.4, .8))
ball(et + Vector((0, 0, .7)), .7, "skin", 1.1); ball(et + Vector((0, -.3, .95)), .38, "skin2", 1.2)
for s in (-1, 1): ball(et + Vector((s * .22, -.62, 1.0)), .12, "tyre")
for s in (-1, 1): bar(et + Vector((s * .5, 0, 1.0)), et + Vector((s * .9, -.3, 1.5)), .12, "skin")
for s in (-1, 1): bar(et + Vector((s * .3, .5, .5)), et + Vector((s * .5, .9, .1)), .14, "skin2")
# the boy riding
pp = bike((0, R + 1.1, 0))
tube(pp + Vector((0, 0, .1)), .38, 1.0, 6, "jeans", .3)
tube(pp + Vector((0, 0, 1.0)), .45, .9, 6, "shirt", .35)
ball(pp + Vector((0, 0, 2.15)), .36, "skin"); tube(pp + Vector((0, 0, 2.25)), .38, .22, 6, "hair", .2)
bar(pp + Vector((0, 0, 1.6)), bike((2.3, R + 2.2, 0)) + Vector((0, 0, .1)), .13, "shirt")
bar(pp + Vector((.2, 0, .1)), bike((2, R, 0)) + Vector((0, 0, .5)), .14, "jeans"); bar(pp + Vector((-.2, 0, .1)), bike((2, R, 0)) + Vector((0, 0, .5)), .14, "jeans")
# a few stars
for i in range(12):
    x, z = random.uniform(-12, 12), random.uniform(9, 13.5)
    ball((x, 10.5, z), .12, "star")

shine("SUN", (.6, .7, 1), 5, (-6, 14, 10), (0, 0, 3)); shine("SUN", (.7, .4, .8), 1.8, (14, -10, 6), (0, 0, 3))

next(n for n in w.node_tree.nodes if n.type == "BACKGROUND").inputs["Color"].default_value = (.01, .015, .04, 1)
cam = bpy.data.cameras.new("c"); cam.lens = 80; cam.clip_end = 500
co = bpy.data.objects.new("c", cam); sc.collection.objects.link(co)
t = Vector((0, 1, 5.5)); co.location = t + Vector((6, -62, 30))
co.rotation_euler = (t - co.location).to_track_quat('-Z', 'Y').to_euler(); sc.camera = co
